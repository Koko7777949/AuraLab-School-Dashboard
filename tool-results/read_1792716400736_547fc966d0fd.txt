     1→'use client';
     2→
     3→import { useCallback, useState, useRef, useEffect, useMemo } from 'react';
     4→import {
     5→  Search,
     6→  FileDown,
     7→  Download,
     8→  Upload,
     9→  Globe,
    10→  Settings,

    12→  Users,

    20→} from 'lucide-react';
    21→import {

    25→} from '@/components/ui/tooltip';
    26→import {

    30→  DropdownMenuLabel,
    31→  DropdownMenuItem,
    32→  DropdownMenuSeparator,
    33→} from '@/components/ui/dropdown-menu';

    38→

    43→  icon: Icon,

    47→  icon: React.ElementType;
    48→  label: string;
    49→  onClick?: () => void;

    52→    <Tooltip>

    55→          onClick={onClick}

    63→        >

    65→        </button>
    66→      </TooltipTrigger>
    67→      <TooltipContent
    68→        sideOffset={8}
    69→        className="rounded-lg border border-[#2A2A38] bg-[#1A1A24] px-3 py-1.5 text-xs text-[#C4C4D4] shadow-lg"
    70→      >
    71→        {label}
    72→      </TooltipContent>
    73→    </Tooltip>
    74→  );
    75→}
    76→
    77→// ---------------------------------------------------------------------------
    78→// Search suggestions (visual)
    79→// ---------------------------------------------------------------------------
    80→const SEARCH_SUGGESTIONS: { text: string; section: ActiveSection; icon: React.ElementType }[] = [
    81→  { text: 'بحث في الطلاب...', section: 'students', icon: Users },
    82→  { text: 'عرض الحصص', section: 'sessions', icon: Settings },
    83→  { text: 'الامتحانات', section: 'exams', icon: Settings },
    84→];
    85→
    86→// ---------------------------------------------------------------------------
    87→// Header Component
    88→// ---------------------------------------------------------------------------
    89→export default function Header() {
    90→  const { searchQuery, setSearchQuery, setActiveSection, sidebarOpen, notifications, students, teachers, locale, toggleLocale } = useAppStore();
    91→  const [showSuggestions, setShowSuggestions] = useState(false);
    92→  const [filteredSuggestions, setFilteredSuggestions] = useState<typeof SEARCH_SUGGESTIONS>([]);
    93→  const searchRef = useRef<HTMLDivElement>(null);
    94→
    95→  // Data refresh indicator (initialize empty to avoid hydration mismatch)
    96→  const [lastRefreshed, setLastRefreshed] = useState('');
    97→  const [isRefreshing, setIsRefreshing] = useState(false);
    98→  const [refreshFlash, setRefreshFlash] = useState(false);
    99→
   100→  // Locale-aware time formatter
   101→  const timeLocale = locale === 'ar' ? 'ar-SA' : 'en-US';
   102→
   103→  // Set initial refresh time client-side
   104→  useEffect(() => {
   105→    // eslint-disable-next-line react-hooks/set-state-in-effect -- client-only init to avoid hydration mismatch
   106→    setLastRefreshed(new Date().toLocaleTimeString(timeLocale, { hour: '2-digit', minute: '2-digit' }));
   107→  }, [timeLocale]);
   108→
   109→  const handleRefresh = useCallback(() => {
   110→    setIsRefreshing(true);
   111→    setRefreshFlash(true);
   112→    setTimeout(() => {
   113→      setIsRefreshing(false);
   114→      setLastRefreshed(new Date().toLocaleTimeString(timeLocale, { hour: '2-digit', minute: '2-digit' }));
   115→    }, 800);
   116→    setTimeout(() => setRefreshFlash(false), 1200);
   117→  }, [timeLocale]);
   118→
   119→  // Auto-refresh every 5 minutes
   120→  useEffect(() => {
   121→    const interval = setInterval(() => {
   122→      handleRefresh();
   123→    }, 5 * 60 * 1000);
   124→    return () => clearInterval(interval);
   125→  }, [handleRefresh]);
   126→
   127→  // Dynamic unread notification count
   128→  const unreadCount = notifications.filter((n) => !n.isRead).length;
   129→
   130→  // Live clock
   131→  const [currentTime, setCurrentTime] = useState('');
   132→  const [currentDate, setCurrentDate] = useState('');
   133→  const [hijriDate, setHijriDate] = useState('');
   134→  const [greeting, setGreeting] = useState('');
   135→
   136→  useEffect(() => {
   137→    const updateClock = () => {
   138→      const now = new Date();
   139→      setCurrentTime(now.toLocaleTimeString(timeLocale, { hour: '2-digit', minute: '2-digit', second: '2-digit' }));
   140→      setCurrentDate(now.toLocaleDateString(timeLocale, { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' }));
   141→
   142→      // Hijri date approximation (only for Arabic)
   143→      if (locale === 'ar') {
   144→        try {
   145→          setHijriDate(now.toLocaleDateString('ar-SA-u-ca-islamic', { year: 'numeric', month: 'long', day: 'numeric' }));
   146→        } catch {
   147→          setHijriDate('');
   148→        }
   149→      } else {
   150→        setHijriDate('');
   151→      }
   152→
   153→      // Time-based greeting
   154→      const hour = now.getHours();
   155→      if (hour >= 5 && hour < 12) {
   156→        setGreeting(t('greeting.morning', locale));
   157→      } else if (hour >= 12 && hour < 17) {
   158→        setGreeting(t('greeting.afternoon', locale));
   159→      } else {
   160→        setGreeting(t('greeting.evening', locale));
   161→      }
   162→    };
   163→    updateClock();
   164→    const interval = setInterval(updateClock, 1000);
   165→    return () => clearInterval(interval);
   166→  }, [locale, timeLocale]);
   167→
   168→  const timeParts = useMemo(() => {
   169→    if (!currentTime) return { time: '', seconds: '' };
   170→    const parts = currentTime.split(':');
   171→    return { time: `${parts[0]}:${parts[1]}`, seconds: parts[2] || '' };
   172→  }, [currentTime]);
   173→
   174→// locale toggle is handled by store toggleLocale
   175→
   176→  const handleSearchKeyDown = useCallback(
   177→    (e: React.KeyboardEvent<HTMLInputElement>) => {
   178→      if (e.key === 'Enter' && searchQuery.trim()) {
   179→        const q = searchQuery.trim().toLowerCase();
   180→        // Search across students and teachers
   181→        const matchedStudent = students.find((s) => s.name.toLowerCase().includes(q));
   182→        const matchedTeacher = teachers.find((t) => t.name.toLowerCase().includes(q));
   183→        if (matchedStudent) {
   184→          setActiveSection('students');
   185→        } else if (matchedTeacher) {
   186→          setActiveSection('teachers');
   187→        } else {
   188→          setActiveSection('students');
   189→        }
   190→        setShowSuggestions(false);
   191→        (e.target as HTMLInputElement).blur();
   192→      }
   193→    },
   194→    [searchQuery, students, teachers, setActiveSection],
   195→  );
   196→
   197→  const handleSearchChange = useCallback(
   198→    (e: React.ChangeEvent<HTMLInputElement>) => {
   199→      const value = e.target.value;
   200→      setSearchQuery(value);
   201→      if (value.trim()) {
   202→        const q = value.trim().toLowerCase();
   203→        // Build dynamic suggestions from store data
   204→        const dynamicSuggestions: typeof SEARCH_SUGGESTIONS = [];
   205→        const matchedStudents = students.filter((s) => s.name.toLowerCase().includes(q)).slice(0, 3);
   206→        const matchedTeachers = teachers.filter((t) => t.name.toLowerCase().includes(q)).slice(0, 3);
   207→        for (const s of matchedStudents) {
   208→          dynamicSuggestions.push({ text: s.name, section: 'students' as const, icon: GraduationCap });
   209→        }
   210→        for (const t of matchedTeachers) {
   211→          dynamicSuggestions.push({ text: t.name, section: 'teachers' as const, icon: Users });
   212→        }
   213→        // Fallback to static suggestions if no matches
   214→        setFilteredSuggestions(
   215→          dynamicSuggestions.length > 0
   216→            ? dynamicSuggestions
   217→            : SEARCH_SUGGESTIONS.filter((s) => s.text.includes(value.trim()))
   218→        );
   219→        setShowSuggestions(true);
   220→      } else {
   221→        setShowSuggestions(false);
   222→      }
   223→    },
   224→    [setSearchQuery, students, teachers],
   225→  );
   226→
   227→  const handleSuggestionClick = useCallback(
   228→    (section: typeof SEARCH_SUGGESTIONS[number]['section']) => {
   229→      setActiveSection(section);
   230→      setShowSuggestions(false);
   231→    },
   232→    [setActiveSection],
   233→  );
   234→
   235→  // Close suggestions on click outside
   236→  useEffect(() => {
   237→    const handleClickOutside = (e: MouseEvent) => {
   238→      if (searchRef.current && !searchRef.current.contains(e.target as Node)) {
   239→        setShowSuggestions(false);
   240→      }
   241→    };
   242→    document.addEventListener('mousedown', handleClickOutside);
   243→    return () => document.removeEventListener('mousedown', handleClickOutside);
   244→  }, []);
   245→
   246→  return (
   247→    <>
   248→      <header
   249→        dir="rtl"
   250→        className={`
   251→          fixed top-0 inset-x-0 z-30
   252→          flex h-16 items-center justify-between gap-3 px-3 sm:px-6
   253→          border-b border-[#2A2A38]/60
   254→          bg-[#121218]/60 backdrop-blur-2xl
   255→          transition-all duration-300 ease-in-out
   256→          ${sidebarOpen ? 'lg:mr-60' : 'lg:mr-[72px]'}
   257→        `}
   258→      >
   259→        {/* Subtle gradient overlay (amber-to-transparent) */}
   260→        <div
   261→          className="absolute inset-0 pointer-events-none rounded-none"
   262→          style={{
   263→            background: 'linear-gradient(to bottom, rgba(212,165,116,0.03), transparent)',
   264→          }}
   265→        />
   266→        {/* ----------------------------------------------------------------- */}
   267→        {/* Right side (RTL start) — Search bar */}
   268→        {/* ----------------------------------------------------------------- */}
   269→        <div className="relative w-full max-w-xs min-w-0 sm:max-w-sm" ref={searchRef}>
   270→          <Search
   271→            size={18}
   272→            strokeWidth={1.8}
   273→            className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-[#8B8B9E]"
   274→          />
   275→          <input
   276→            type="text"
   277→            value={searchQuery}
   278→            onChange={handleSearchChange}
   279→            onKeyDown={handleSearchKeyDown}
   280→            onFocus={() => {
   281→              if (searchQuery.trim()) setShowSuggestions(true);
   282→            }}
   283→            placeholder={t('header.search', locale)}
   284→            className="
   285→              h-9 w-full rounded-lg border border-[#2A2A38] bg-[#1A1A24]/80
   286→              pr-10 pl-4 text-sm text-[#E4E4EC] placeholder-[#8B8B9E]
   287→              outline-none backdrop-blur-sm
   288→              transition-all duration-200
   289→              focus:border-[#D4A574]/50
   290→              focus:shadow-[0_0_12px_2px_rgba(212,165,116,0.15),0_0_4px_0px_rgba(212,165,116,0.1)]
   291→              focus:ring-0
   292→            "
   293→          />
   294→
   295→          {/* Search suggestions dropdown */}
   296→          {showSuggestions && (
   297→            <div
   298→              className="absolute top-full right-0 left-0 mt-1.5 z-50 rounded-lg border border-[#2A2A38] bg-[#1A1A24] py-1.5 shadow-xl animate-fade-in-up overflow-hidden"
   299→              style={{ animationDuration: '0.2s' }}
   300→            >
   301→              {filteredSuggestions.length > 0 ? (
   302→                filteredSuggestions.map((suggestion, i) => {
   303→                  const Icon = suggestion.icon;
   304→                  return (
   305→                    <button
   306→                      key={i}
   307→                      onClick={() => handleSuggestionClick(suggestion.section)}
   308→                      className="flex w-full items-center gap-2.5 px-3 py-2 text-sm text-[#C4C4D4] transition-colors duration-150 hover:bg-[#22222E] hover:text-[#D4A574]"
   309→                    >
   310→                      <Icon size={14} strokeWidth={1.5} className="text-[#8B8B9E]" />
   311→                      <span>{suggestion.text}</span>
   312→                      <span className="mr-auto text-[11px] text-[#8B8B9E]/60">
   313→                        Enter ↵
   314→                      </span>
   315→                    </button>
   316→                  );
   317→                })
   318→              ) : (
   319→                <div className="flex items-center gap-2 px-3 py-2 text-sm text-[#8B8B9E]">
   320→                  <Search size={14} strokeWidth={1.5} />
   321→                  <span>{t('header.searchStudents', locale)}</span>
   322→                </div>
   323→              )}
   324→            </div>
   325→          )}
   326→        </div>
   327→
   328→        {/* ----------------------------------------------------------------- */}
   329→        {/* Center — Live Clock + App name */}
   330→        {/* ----------------------------------------------------------------- */}
   331→        <div className="hidden sm:flex flex-col items-center justify-center gap-0.5 min-w-0">
   332→          {/* Greeting */}
   333→          {greeting && (
   334→            <span className="text-[10px] font-medium text-[#D4A574]/70">{greeting}</span>
   335→          )}
   336→          {/* Live clock */}
   337→          <div className="flex items-center gap-2 select-none">
   338→            <Clock size={13} strokeWidth={1.8} className="text-[#D4A574]/60" />
   339→            <span className="text-sm font-semibold tabular-nums tracking-wider text-[#C4C4D4]" dir="ltr">
   340→              {timeParts.time}
   341→            </span>
   342→            <span className="text-xs font-medium tabular-nums text-[#8B8B9E] min-w-[18px] text-center animate-pulse" dir="ltr">
   343→              {timeParts.seconds}
   344→            </span>
   345→          </div>
   346→          {/* App name */}
   347→          <h1 className="text-sm font-bold tracking-tight select-none">
   348→            <span
   349→              className="bg-clip-text text-transparent"
   350→              style={{
   351→                backgroundImage:
   352→                  'linear-gradient(135deg, #D4A574 0%, #E8C9A0 50%, #D4A574 100%)',
   353→              }}
   354→            >
   355→              AuraLab
   356→            </span>
   357→          </h1>
   358→          {/* Gregorian date */}
   359→          <span className="text-[10px] text-[#5A5A6E] truncate max-w-[180px]">
   360→            {currentDate}
   361→          </span>
   362→          {/* Hijri date */}
   363→          {hijriDate && (
   364→            <span className="text-[9px] text-[#5A5A6E]/60 truncate max-w-[180px]">
   365→              {hijriDate}
   366→            </span>
   367→          )}
   368→        </div>
   369→
   370→        {/* ----------------------------------------------------------------- */}
   371→        {/* Left side (RTL end) — Action buttons */}
   372→        {/* ----------------------------------------------------------------- */}
   373→        <div className="flex items-center gap-2 sm:gap-3">
   374→          {/* Last Refreshed indicator + Refresh button */}
   375→          <div className={`hidden sm:flex items-center gap-1.5 text-[10px] text-[#8B8B9E] transition-all duration-300 ${refreshFlash ? 'opacity-100 scale-105' : 'opacity-70'}`}>
   376→            <span>{t('header.lastRefresh', locale)}</span>
   377→            <span className="tabular-nums font-medium" dir="ltr">{lastRefreshed}</span>
   378→            <button
   379→              onClick={handleRefresh}
   380→              aria-label={t('header.refreshData', locale)}
   381→              className="flex items-center justify-center h-5 w-5 rounded text-[#8B8B9E] hover:text-[#D4A574] hover:bg-[#D4A574]/10 transition-all duration-200"
   382→            >
   383→              <RefreshCw
   384→                size={12}
   385→                strokeWidth={2}
   386→                className={isRefreshing ? 'animate-spin' : ''}
   387→              />
   388→            </button>
   389→          </div>
   390→
   391→          {/* PDF Export → analytics */}
   392→          <HeaderButton
   393→            icon={FileDown}
   394→            label={t('header.exportPdf', locale)}
   395→            onClick={() => setActiveSection('analytics')}
   396→          />
   397→
   398→          {/* Export → analytics */}
   399→          <div className="hidden sm:block">
   400→            <HeaderButton
   401→              icon={Download}
   402→              label={t('header.export', locale)}
   403→              onClick={() => setActiveSection('analytics')}
   404→            />
   405→          </div>
   406→
   407→          {/* Import → students */}
   408→          <div className="hidden sm:block">
   409→            <HeaderButton
   410→              icon={Upload}
   411→              label={t('header.import', locale)}
   412→              onClick={() => setActiveSection('students')}
   413→            />
   414→          </div>
   415→
   416→          {/* Divider */}
   417→          <div className="mx-1 hidden h-5 w-px bg-[#2A2A38] sm:block" />
   418→
   419→          {/* Language toggle */}
   420→          <Tooltip>
   421→            <TooltipTrigger asChild>
   422→              <button
   423→                onClick={() => {
   424→                  toggleLocale();
   425→                  toast.success(locale === 'ar' ? t('lang.changedToEn', locale) : t('lang.changedToAr', locale));
   426→                }}
   427→                aria-label={t('header.changeLanguage', locale)}
   428→                className="flex h-9 items-center gap-1.5 rounded-lg
   429→                  border border-[#2A2A38] bg-[#1A1A24]/80 px-2.5
   430→                  text-xs font-semibold text-[#C4C4D4]
   431→                  transition-all duration-200
   432→                  hover:scale-[1.02] hover:border-[#D4A574]/40 hover:text-[#D4A574]
   433→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#D4A574]/50
   434→                  active:scale-[0.98]"
   435→              >
   436→                <Globe size={16} strokeWidth={1.8} />
   437→                <span className="hidden sm:inline">{locale === 'ar' ? 'AR' : 'EN'}</span>
   438→              </button>
   439→            </TooltipTrigger>
   440→            <TooltipContent
   441→              sideOffset={8}
   442→              className="rounded-lg border border-[#2A2A38] bg-[#1A1A24] px-3 py-1.5 text-xs text-[#C4C4D4] shadow-lg"
   443→            >
   444→              {locale === 'ar' ? 'English' : 'العربية'}
   445→            </TooltipContent>
   446→          </Tooltip>
   447→
   448→          {/* Settings → settings */}
   449→          <HeaderButton
   450→            icon={Settings}
   451→            label={t('header.settings', locale)}
   452→            onClick={() => setActiveSection('settings')}
   453→          />
   454→
   455→          {/* Divider */}
   456→          <div className="mx-1 hidden h-5 w-px bg-[#2A2A38] sm:block" />
   457→
   458→          {/* Notification bell → notifications */}
   459→          <Tooltip>
   460→            <TooltipTrigger asChild>
   461→              <button
   462→                onClick={() => setActiveSection('notifications')}
   463→                aria-label={t('header.notifications', locale)}
   464→                className="relative ripple-effect glow-amber-sm flex h-9 w-9 shrink-0 items-center justify-center rounded-lg
   465→                  border border-[#2A2A38] bg-[#1A1A24]/80 text-[#C4C4D4]
   466→                  transition-all duration-200
   467→                  hover:scale-[1.02] hover:border-[#D4A574]/40 hover:text-[#D4A574]
   468→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#D4A574]/50
   469→                  active:scale-[0.98]"
   470→              >
   471→                <span className={unreadCount > 0 ? 'animate-bell-ring' : ''}>
   472→                  <Bell size={18} strokeWidth={1.8} />
   473→                </span>
   474→                {/* Dynamic notification count badge with pulse ring */}
   475→                {unreadCount > 0 && (
   476→                  <span className="absolute -top-1 -left-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-rose-500 text-[9px] font-bold text-white px-1 tabular-nums shadow-lg animate-badge-pulse-ring">
   477→                    {unreadCount > 99 ? '99+' : unreadCount}
   478→                  </span>
   479→                )}
   480→              </button>
   481→            </TooltipTrigger>
   482→            <TooltipContent
   483→              sideOffset={8}
   484→              className="rounded-lg border border-[#2A2A38] bg-[#1A1A24] px-3 py-1.5 text-xs text-[#C4C4D4] shadow-lg"
   485→            >
   486→              {t('header.notifications', locale)} {unreadCount > 0 && `(${unreadCount})`}
   487→            </TooltipContent>
   488→          </Tooltip>
   489→
   490→          {/* ----------------------------------------------------------------- */}
   491→          {/* User Profile Dropdown */}
   492→          {/* ----------------------------------------------------------------- */}
   493→          <DropdownMenu dir={locale === 'ar' ? 'rtl' : 'ltr'}>
   494→            <DropdownMenuTrigger asChild>
   495→              <button
   496→                aria-label={t('header.userMenu', locale)}
   497→                className="flex items-center gap-2 rounded-lg
   498→                  border border-[#2A2A38] bg-[#1A1A24]/80 py-1.5 pl-2 pr-3
   499→                  transition-all duration-200
   500→                  hover:scale-[1.02] hover:border-[#D4A574]/40
   501→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#D4A574]/50
   502→                  active:scale-[0.98]"
   503→              >
   504→                {/* Avatar circle */}
   505→                <span
   506→                  className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold text-[#1A1A24]"
   507→                  style={{
   508→                    background:
   509→                      'linear-gradient(135deg, #D4A574 0%, #E8C9A0 60%, #D4A574 100%)',
   510→                  }}
   511→                >
   512→                  {locale === 'ar' ? 'أم' : 'AM'}
   513→                </span>
   514→
   515→                {/* Name (hidden on small screens) */}
   516→                <span className="hidden text-sm font-medium text-[#E4E4EC] sm:block">
   517→                  {locale === 'ar' ? 'أحمد محمد' : 'Ahmed Mohamed'}
   518→                </span>
   519→
   520→                {/* Dropdown chevron */}
   521→                <ChevronDown
   522→                  size={14}
   523→                  strokeWidth={2}
   524→                  className="hidden text-[#8B8B9E] sm:block"
   525→                />
   526→              </button>
   527→            </DropdownMenuTrigger>
   528→            <DropdownMenuContent
   529→              sideOffset={8}
   530→              align="start"
