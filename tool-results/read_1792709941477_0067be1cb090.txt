   130→function getScoreColor(score: number | null, maxScore: number): { bar: string; text: string; bg: string } {
   131→  if (score === null) return { bar: 'bg-slate-500', text: 'text-slate-400', bg: '' };
   132→  const pct = (score / maxScore) * 100;
   133→  if (pct >= 70) return { bar: 'bg-emerald-500', text: 'text-emerald-400', bg: 'bg-emerald-500/10' };
   134→  if (pct >= 50) return { bar: 'bg-amber-500', text: 'text-amber-400', bg: 'bg-amber-500/10' };
   135→  return { bar: 'bg-rose-500', text: 'text-rose-400', bg: 'bg-rose-500/10' };
   136→}
   137→
   138→function getScoreLabel(score: number | null, maxScore: number): string {
   139→  if (score === null) return 'لم يُحدد';
   140→  const pct = (score / maxScore) * 100;
   141→  if (pct >= 90) return 'ممتاز';
   142→  if (pct >= 70) return 'جيد جداً';
   143→  if (pct >= 50) return 'مقبول';
   144→  return 'يحتاج تحسين';
   145→}

   151→interface SubjectAvg {
   152→  subject: string;

   158→interface AnalysisStats {
   159→  highest: { pct: number; subject: string };
   160→  lowest: { pct: number; subject: string };
   161→  overallAvg: number;
   162→  passRate: number;
   163→}
   164→
   165→function getBarColorHex(avg: number): string {
   166→  if (avg >= 85) return '#10b981';
   167→  if (avg >= 70) return '#f59e0b';
   168→  if (avg >= 50) return '#0ea5e9';
   169→  return '#f43f5e';
   170→}
   171→
   172→function computeSubjectAverages(exams: Exam[]): SubjectAvg[] {
   173→  const groups: Record<string, { total: number; count: number }> = {};
   174→
   175→  for (const exam of exams) {
   176→    if (exam.score === null) continue;
   177→    const pct = (exam.score / Math.max(0.001, exam.maxScore)) * 100;
   178→    if (!groups[exam.subject]) {
   179→      groups[exam.subject] = { total: 0, count: 0 };
   180→    }
   181→    groups[exam.subject] = {
   182→      total: groups[exam.subject].total + pct,
   183→      count: groups[exam.subject].count + 1,
   184→    };
   185→  }
   186→
   187→  return Object.entries(groups)
   188→    .map(([subject, data]) => ({
   189→      subject,
   190→      avg: Math.round((data.total / Math.max(0.001, data.count)) * 10) / 10,
   191→      count: data.count,
   192→      color: getBarColorHex(data.total / Math.max(0.001, data.count)),
   193→    }))
   194→    .sort((a, b) => b.avg - a.avg);
   195→}
   196→
   197→function computeAnalysisStats(exams: Exam[]): AnalysisStats {
   198→  const scoredExams = exams.filter((e) => e.score !== null);
   199→
   200→  if (scoredExams.length === 0) {
   201→    return {
   202→      highest: { pct: 0, subject: '—' },
   203→      lowest: { pct: 0, subject: '—' },
   204→      overallAvg: 0,
   205→      passRate: 0,
   206→    };
   207→  }
   208→
   209→  const result = scoredExams.reduce(
   210→    (acc, exam) => {
   211→      const pct = (exam.score! / Math.max(0.001, exam.maxScore)) * 100;
   212→      return {
   213→        totalPct: acc.totalPct + pct,
   214→        passCount: acc.passCount + (pct >= 50 ? 1 : 0),
   215→        highestPct: Math.max(acc.highestPct, pct),
   216→        highestSubject: pct > acc.highestPct ? exam.subject : acc.highestSubject,
   217→        lowestPct: Math.min(acc.lowestPct, pct),
   218→        lowestSubject: pct < acc.lowestPct ? exam.subject : acc.lowestSubject,
   219→      };
   220→    },
   221→    { totalPct: 0, passCount: 0, highestPct: -1, highestSubject: '', lowestPct: 101, lowestSubject: '' },
   222→  );
   223→
   224→  const count = Math.max(0.001, scoredExams.length);
   225→  return {
   226→    highest: { pct: Math.round(result.highestPct * 10) / 10, subject: result.highestSubject },
   227→    lowest: { pct: Math.round(result.lowestPct * 10) / 10, subject: result.lowestSubject },
   228→    overallAvg: Math.round((result.totalPct / count) * 10) / 10,
   229→    passRate: Math.round(((result.passCount / count) * 100) * 10) / 10,
   230→  };
   231→}
