   971→   971→    recipientRole: Message['recipientRole'];
   972→   972→    subject: string;
   973→   973→    content: string;
   974→   974→    priority: Message['priority'];
   975→   975→  }) => void;
   976→   976→  initialValues?: ComposeInitialValues;
   977→   977→}) {

   982→   982→      const found = RECIPIENT_OPTIONS.find((r) => r.id === initialValues.recipientId)
   983→   983→        ?? GROUP_RECIPIENT_OPTIONS.find((r) => r.id === initialValues.recipientId);

   985→   985→    }

   992→   992→  const [priority, setPriority] = useState<Message['priority']>(initialValues?.priority ?? 'normal');
   993→   993→  const [isSending, setIsSending] = useState(false);

   995→   995→  const [showGroupSelector, setShowGroupSelector] = useState(false);
   996→   996→

   998→   998→    () => [...GROUP_RECIPIENT_OPTIONS, ...RECIPIENT_OPTIONS.map((r) => ({ ...r, isGroup: false }))],
   999→   999→    [],
  1000→  1000→  );

  1004→  1004→      return showGroupSelector ? GROUP_RECIPIENT_OPTIONS : RECIPIENT_OPTIONS.map((r) => ({ ...r, isGroup: false }));
