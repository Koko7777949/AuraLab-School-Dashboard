     1→'use client';
     2→
     3→import { useCallback, useState, useRef, useEffect, useMemo } from 'react';
     4→import {
     5→  Search,
     6→  FileDown,
     7→  Download,
     8→  Upload,
     9→  Globe,
    10→  Settings,
    11→  ChevronDown,
    12→  Users,

    20→} from 'lucide-react';
    21→import {

    25→} from '@/components/ui/tooltip';
    26→import {

    31→  DropdownMenuItem,
    32→  DropdownMenuSeparator,
    33→} from '@/components/ui/dropdown-menu';

    37→

    46→  icon: React.ElementType;

    62→        >
    63→          <Icon size={18} strokeWidth={1.8} />
    64→        </button>
    65→      </TooltipTrigger>

    67→        sideOffset={8}

    69→      >
    70→        {label}
    71→      </TooltipContent>
    72→    </Tooltip>
    73→  );
    74→}
    75→
    76→// ---------------------------------------------------------------------------
    77→// Search suggestions (visual)
    78→// ---------------------------------------------------------------------------
    79→const SEARCH_SUGGESTIONS: { text: string; section: ActiveSection; icon: React.ElementType }[] = [
    80→  { text: 'بحث في الطلاب...', section: 'students', icon: Users },
    81→  { text: 'عرض الحصص', section: 'sessions', icon: Settings },
    82→  { text: 'الامتحانات', section: 'exams', icon: Settings },
    83→];
    84→
    85→// ---------------------------------------------------------------------------
    86→// Header Component
    87→// ---------------------------------------------------------------------------
    88→export default function Header() {
    89→  const { searchQuery, setSearchQuery, setActiveSection, sidebarOpen, notifications, students, teachers, locale, toggleLocale } = useAppStore();
    90→  const [showSuggestions, setShowSuggestions] = useState(false);
    91→  const [filteredSuggestions, setFilteredSuggestions] = useState<typeof SEARCH_SUGGESTIONS>([]);
    92→  const searchRef = useRef<HTMLDivElement>(null);
    93→
    94→  // Data refresh indicator (initialize empty to avoid hydration mismatch)
    95→  const [lastRefreshed, setLastRefreshed] = useState('');
    96→  const [isRefreshing, setIsRefreshing] = useState(false);
    97→  const [refreshFlash, setRefreshFlash] = useState(false);
    98→
    99→  // Locale-aware time formatter
   100→  const timeLocale = locale === 'ar' ? 'ar-SA' : 'en-US';
   101→
   102→  // Set initial refresh time client-side
   103→  useEffect(() => {
   104→    // eslint-disable-next-line react-hooks/set-state-in-effect -- client-only init to avoid hydration mismatch
   105→    setLastRefreshed(new Date().toLocaleTimeString(timeLocale, { hour: '2-digit', minute: '2-digit' }));
   106→  }, [timeLocale]);
   107→
   108→  const handleRefresh = useCallback(() => {
   109→    setIsRefreshing(true);
   110→    setRefreshFlash(true);
   111→    setTimeout(() => {
   112→      setIsRefreshing(false);
   113→      setLastRefreshed(new Date().toLocaleTimeString(timeLocale, { hour: '2-digit', minute: '2-digit' }));
   114→    }, 800);
   115→    setTimeout(() => setRefreshFlash(false), 1200);
   116→  }, [timeLocale]);
   117→
   118→  // Auto-refresh every 5 minutes
   119→  useEffect(() => {
   120→    const interval = setInterval(() => {
   121→      handleRefresh();
   122→    }, 5 * 60 * 1000);
   123→    return () => clearInterval(interval);
   124→  }, [handleRefresh]);
   125→
   126→  // Dynamic unread notification count
   127→  const unreadCount = notifications.filter((n) => !n.isRead).length;
   128→
   129→  // Live clock
   130→  const [currentTime, setCurrentTime] = useState('');
   131→  const [currentDate, setCurrentDate] = useState('');
   132→  const [hijriDate, setHijriDate] = useState('');
   133→  const [greeting, setGreeting] = useState('');
   134→
   135→  useEffect(() => {
   136→    const updateClock = () => {
   137→      const now = new Date();
   138→      setCurrentTime(now.toLocaleTimeString(timeLocale, { hour: '2-digit', minute: '2-digit', second: '2-digit' }));
   139→      setCurrentDate(now.toLocaleDateString(timeLocale, { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' }));
   140→
   141→      // Hijri date approximation (only for Arabic)
   142→      if (locale === 'ar') {
   143→        try {
   144→          setHijriDate(now.toLocaleDateString('ar-SA-u-ca-islamic', { year: 'numeric', month: 'long', day: 'numeric' }));
   145→        } catch {
   146→          setHijriDate('');
   147→        }
   148→      } else {
   149→        setHijriDate('');
   150→      }
   151→
   152→      // Time-based greeting
   153→      const hour = now.getHours();
   154→      if (hour >= 5 && hour < 12) {
   155→        setGreeting(t('greeting.morning', locale));
   156→      } else if (hour >= 12 && hour < 17) {
   157→        setGreeting(t('greeting.afternoon', locale));
   158→      } else {
   159→        setGreeting(t('greeting.evening', locale));
   160→      }
   161→    };
   162→    updateClock();
   163→    const interval = setInterval(updateClock, 1000);
   164→    return () => clearInterval(interval);
   165→  }, [locale, timeLocale]);
   166→
   167→  const timeParts = useMemo(() => {
   168→    if (!currentTime) return { time: '', seconds: '' };
   169→    const parts = currentTime.split(':');
   170→    return { time: `${parts[0]}:${parts[1]}`, seconds: parts[2] || '' };
   171→  }, [currentTime]);
   172→
   173→// locale toggle is handled by store toggleLocale
   174→
   175→  const handleSearchKeyDown = useCallback(
   176→    (e: React.KeyboardEvent<HTMLInputElement>) => {
   177→      if (e.key === 'Enter' && searchQuery.trim()) {
   178→        const q = searchQuery.trim().toLowerCase();
   179→        // Search across students and teachers
   180→        const matchedStudent = students.find((s) => s.name.toLowerCase().includes(q));
   181→        const matchedTeacher = teachers.find((t) => t.name.toLowerCase().includes(q));
   182→        if (matchedStudent) {
   183→          setActiveSection('students');
   184→        } else if (matchedTeacher) {
   185→          setActiveSection('teachers');
   186→        } else {
   187→          setActiveSection('students');
   188→        }
   189→        setShowSuggestions(false);
   190→        (e.target as HTMLInputElement).blur();
   191→      }
   192→    },
   193→    [searchQuery, students, teachers, setActiveSection],
   194→  );
   195→
   196→  const handleSearchChange = useCallback(
   197→    (e: React.ChangeEvent<HTMLInputElement>) => {
   198→      const value = e.target.value;
   199→      setSearchQuery(value);
   200→      if (value.trim()) {
   201→        const q = value.trim().toLowerCase();
   202→        // Build dynamic suggestions from store data
   203→        const dynamicSuggestions: typeof SEARCH_SUGGESTIONS = [];
   204→        const matchedStudents = students.filter((s) => s.name.toLowerCase().includes(q)).slice(0, 3);
   205→        const matchedTeachers = teachers.filter((t) => t.name.toLowerCase().includes(q)).slice(0, 3);
   206→        for (const s of matchedStudents) {
   207→          dynamicSuggestions.push({ text: s.name, section: 'students' as const, icon: GraduationCap });
   208→        }
   209→        for (const t of matchedTeachers) {
   210→          dynamicSuggestions.push({ text: t.name, section: 'teachers' as const, icon: Users });
   211→        }
   212→        // Fallback to static suggestions if no matches
   213→        setFilteredSuggestions(
   214→          dynamicSuggestions.length > 0
   215→            ? dynamicSuggestions
   216→            : SEARCH_SUGGESTIONS.filter((s) => s.text.includes(value.trim()))
   217→        );
   218→        setShowSuggestions(true);
   219→      } else {
   220→        setShowSuggestions(false);
   221→      }
   222→    },
   223→    [setSearchQuery, students, teachers],
   224→  );
   225→
   226→  const handleSuggestionClick = useCallback(
   227→    (section: typeof SEARCH_SUGGESTIONS[number]['section']) => {
   228→      setActiveSection(section);
   229→      setShowSuggestions(false);
   230→    },
   231→    [setActiveSection],
   232→  );
   233→
   234→  // Close suggestions on click outside
   235→  useEffect(() => {
   236→    const handleClickOutside = (e: MouseEvent) => {
   237→      if (searchRef.current && !searchRef.current.contains(e.target as Node)) {
   238→        setShowSuggestions(false);
   239→      }
   240→    };
   241→    document.addEventListener('mousedown', handleClickOutside);
   242→    return () => document.removeEventListener('mousedown', handleClickOutside);
   243→  }, []);
   244→
   245→  return (
   246→    <>
   247→      <header
   248→        dir="rtl"
   249→        className={`
   250→          fixed top-0 inset-x-0 z-30
   251→          flex h-16 items-center justify-between gap-4 px-6
   252→          border-b border-[#2A2A38]/60
   253→          bg-[#121218]/60 backdrop-blur-2xl
   254→          transition-all duration-300 ease-in-out
   255→          ${sidebarOpen ? 'mr-60' : 'mr-[72px]'}
   256→        `}
   257→      >
   258→        {/* Subtle gradient overlay (amber-to-transparent) */}
   259→        <div
   260→          className="absolute inset-0 pointer-events-none rounded-none"
   261→          style={{
   262→            background: 'linear-gradient(to bottom, rgba(212,165,116,0.03), transparent)',
   263→          }}
   264→        />
   265→        {/* ----------------------------------------------------------------- */}
   266→        {/* Right side (RTL start) — Search bar */}
   267→        {/* ----------------------------------------------------------------- */}
   268→        <div className="relative w-full max-w-xs min-w-0 sm:max-w-sm" ref={searchRef}>
   269→          <Search
   270→            size={18}
   271→            strokeWidth={1.8}
   272→            className="pointer-events-none absolute right-3 top-1/2 -translate-y-1/2 text-[#8B8B9E]"
   273→          />
   274→          <input
   275→            type="text"
   276→            value={searchQuery}
   277→            onChange={handleSearchChange}
   278→            onKeyDown={handleSearchKeyDown}
   279→            onFocus={() => {
   280→              if (searchQuery.trim()) setShowSuggestions(true);
   281→            }}
   282→            placeholder={t('header.search', locale)}
   283→            className="
   284→              h-9 w-full rounded-lg border border-[#2A2A38] bg-[#1A1A24]/80
   285→              pr-10 pl-4 text-sm text-[#E4E4EC] placeholder-[#8B8B9E]
   286→              outline-none backdrop-blur-sm
   287→              transition-all duration-200
   288→              focus:border-[#D4A574]/50
   289→              focus:shadow-[0_0_12px_2px_rgba(212,165,116,0.15),0_0_4px_0px_rgba(212,165,116,0.1)]
   290→              focus:ring-0
   291→            "
   292→          />
   293→
   294→          {/* Search suggestions dropdown */}
   295→          {showSuggestions && (
   296→            <div
   297→              className="absolute top-full right-0 left-0 mt-1.5 z-50 rounded-lg border border-[#2A2A38] bg-[#1A1A24] py-1.5 shadow-xl animate-fade-in-up overflow-hidden"
   298→              style={{ animationDuration: '0.2s' }}
   299→            >
   300→              {filteredSuggestions.length > 0 ? (
   301→                filteredSuggestions.map((suggestion, i) => {
   302→                  const Icon = suggestion.icon;
   303→                  return (
   304→                    <button
   305→                      key={i}
   306→                      onClick={() => handleSuggestionClick(suggestion.section)}
   307→                      className="flex w-full items-center gap-2.5 px-3 py-2 text-sm text-[#C4C4D4] transition-colors duration-150 hover:bg-[#22222E] hover:text-[#D4A574]"
   308→                    >
   309→                      <Icon size={14} strokeWidth={1.5} className="text-[#8B8B9E]" />
   310→                      <span>{suggestion.text}</span>
   311→                      <span className="mr-auto text-[11px] text-[#8B8B9E]/60">
   312→                        Enter ↵
   313→                      </span>
   314→                    </button>
   315→                  );
   316→                })
   317→              ) : (
   318→                <div className="flex items-center gap-2 px-3 py-2 text-sm text-[#8B8B9E]">
   319→                  <Search size={14} strokeWidth={1.5} />
   320→                  <span>{t('header.searchStudents', locale)}</span>
   321→                </div>
   322→              )}
   323→            </div>
   324→          )}
   325→        </div>
   326→
   327→        {/* ----------------------------------------------------------------- */}
   328→        {/* Center — Live Clock + App name */}
   329→        {/* ----------------------------------------------------------------- */}
   330→        <div className="hidden sm:flex flex-col items-center justify-center gap-0.5 min-w-0">
   331→          {/* Greeting */}
   332→          {greeting && (
   333→            <span className="text-[10px] font-medium text-[#D4A574]/70">{greeting}</span>
   334→          )}
   335→          {/* Live clock */}
   336→          <div className="flex items-center gap-2 select-none">
   337→            <Clock size={13} strokeWidth={1.8} className="text-[#D4A574]/60" />
   338→            <span className="text-sm font-semibold tabular-nums tracking-wider text-[#C4C4D4]" dir="ltr">
   339→              {timeParts.time}
   340→            </span>
   341→            <span className="text-xs font-medium tabular-nums text-[#8B8B9E] min-w-[18px] text-center animate-pulse" dir="ltr">
   342→              {timeParts.seconds}
   343→            </span>
   344→          </div>
   345→          {/* App name */}
   346→          <h1 className="text-sm font-bold tracking-tight select-none">
   347→            <span
   348→              className="bg-clip-text text-transparent"
   349→              style={{
   350→                backgroundImage:
   351→                  'linear-gradient(135deg, #D4A574 0%, #E8C9A0 50%, #D4A574 100%)',
   352→              }}
   353→            >
   354→              AuraLab
   355→            </span>
   356→          </h1>
   357→          {/* Gregorian date */}
   358→          <span className="text-[10px] text-[#5A5A6E] truncate max-w-[180px]">
   359→            {currentDate}
   360→          </span>
   361→          {/* Hijri date */}
   362→          {hijriDate && (
   363→            <span className="text-[9px] text-[#5A5A6E]/60 truncate max-w-[180px]">
   364→              {hijriDate}
   365→            </span>
   366→          )}
   367→        </div>
   368→
   369→        {/* ----------------------------------------------------------------- */}
   370→        {/* Left side (RTL end) — Action buttons */}
   371→        {/* ----------------------------------------------------------------- */}
   372→        <div className="flex items-center gap-2 sm:gap-3">
   373→          {/* Last Refreshed indicator + Refresh button */}
   374→          <div className={`hidden sm:flex items-center gap-1.5 text-[10px] text-[#8B8B9E] transition-all duration-300 ${refreshFlash ? 'opacity-100 scale-105' : 'opacity-70'}`}>
   375→            <span>{t('header.lastRefresh', locale)}</span>
   376→            <span className="tabular-nums font-medium" dir="ltr">{lastRefreshed}</span>
   377→            <button
   378→              onClick={handleRefresh}
   379→              aria-label={t('header.refreshData', locale)}
   380→              className="flex items-center justify-center h-5 w-5 rounded text-[#8B8B9E] hover:text-[#D4A574] hover:bg-[#D4A574]/10 transition-all duration-200"
   381→            >
   382→              <RefreshCw
   383→                size={12}
   384→                strokeWidth={2}
   385→                className={isRefreshing ? 'animate-spin' : ''}
   386→              />
   387→            </button>
   388→          </div>
   389→
   390→          {/* PDF Export → analytics */}
   391→          <HeaderButton
   392→            icon={FileDown}
   393→            label={t('header.exportPdf', locale)}
   394→            onClick={() => setActiveSection('analytics')}
   395→          />
   396→
   397→          {/* Export → analytics */}
   398→          <div className="hidden sm:block">
   399→            <HeaderButton
   400→              icon={Download}
   401→              label={t('header.export', locale)}
   402→              onClick={() => setActiveSection('analytics')}
   403→            />
   404→          </div>
   405→
   406→          {/* Import → students */}
   407→          <div className="hidden sm:block">
   408→            <HeaderButton
   409→              icon={Upload}
   410→              label={t('header.import', locale)}
   411→              onClick={() => setActiveSection('students')}
   412→            />
   413→          </div>
   414→
   415→          {/* Divider */}
   416→          <div className="mx-1 hidden h-5 w-px bg-[#2A2A38] sm:block" />
   417→
   418→          {/* Language toggle */}
   419→          <Tooltip>
   420→            <TooltipTrigger asChild>
   421→              <button
   422→                onClick={toggleLocale}
   423→                aria-label={t('header.changeLanguage', locale)}
   424→                className="flex h-9 items-center gap-1.5 rounded-lg
   425→                  border border-[#2A2A38] bg-[#1A1A24]/80 px-2.5
   426→                  text-xs font-semibold text-[#C4C4D4]
   427→                  transition-all duration-200
   428→                  hover:scale-[1.02] hover:border-[#D4A574]/40 hover:text-[#D4A574]
   429→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#D4A574]/50
   430→                  active:scale-[0.98]"
   431→              >
   432→                <Globe size={16} strokeWidth={1.8} />
   433→                <span className="hidden sm:inline">{locale === 'ar' ? 'AR' : 'EN'}</span>
   434→              </button>
   435→            </TooltipTrigger>
   436→            <TooltipContent
   437→              sideOffset={8}
   438→              className="rounded-lg border border-[#2A2A38] bg-[#1A1A24] px-3 py-1.5 text-xs text-[#C4C4D4] shadow-lg"
   439→            >
   440→              {locale === 'ar' ? 'English' : 'العربية'}
   441→            </TooltipContent>
   442→          </Tooltip>
   443→
   444→          {/* Settings → settings */}
   445→          <HeaderButton
   446→            icon={Settings}
   447→            label={t('header.settings', locale)}
   448→            onClick={() => setActiveSection('settings')}
   449→          />
   450→
   451→          {/* Divider */}
   452→          <div className="mx-1 hidden h-5 w-px bg-[#2A2A38] sm:block" />
   453→
   454→          {/* Notification bell → notifications */}
   455→          <Tooltip>
   456→            <TooltipTrigger asChild>
   457→              <button
   458→                onClick={() => setActiveSection('notifications')}
   459→                aria-label={t('header.notifications', locale)}
   460→                className="relative ripple-effect glow-amber-sm flex h-9 w-9 shrink-0 items-center justify-center rounded-lg
   461→                  border border-[#2A2A38] bg-[#1A1A24]/80 text-[#C4C4D4]
   462→                  transition-all duration-200
   463→                  hover:scale-[1.02] hover:border-[#D4A574]/40 hover:text-[#D4A574]
   464→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#D4A574]/50
   465→                  active:scale-[0.98]"
   466→              >
   467→                <span className={unreadCount > 0 ? 'animate-bell-ring' : ''}>
   468→                  <Bell size={18} strokeWidth={1.8} />
   469→                </span>
   470→                {/* Dynamic notification count badge with pulse ring */}
   471→                {unreadCount > 0 && (
   472→                  <span className="absolute -top-1 -left-1 flex h-4 min-w-4 items-center justify-center rounded-full bg-rose-500 text-[9px] font-bold text-white px-1 tabular-nums shadow-lg animate-badge-pulse-ring">
   473→                    {unreadCount > 99 ? '99+' : unreadCount}
   474→                  </span>
   475→                )}
   476→              </button>
   477→            </TooltipTrigger>
   478→            <TooltipContent
   479→              sideOffset={8}
   480→              className="rounded-lg border border-[#2A2A38] bg-[#1A1A24] px-3 py-1.5 text-xs text-[#C4C4D4] shadow-lg"
   481→            >
   482→              {t('header.notifications', locale)} {unreadCount > 0 && `(${unreadCount})`}
   483→            </TooltipContent>
   484→          </Tooltip>
   485→
   486→          {/* ----------------------------------------------------------------- */}
   487→          {/* User Profile Dropdown */}
   488→          {/* ----------------------------------------------------------------- */}
   489→          <DropdownMenu dir={locale === 'ar' ? 'rtl' : 'ltr'}>
   490→            <DropdownMenuTrigger asChild>
   491→              <button
   492→                aria-label={t('header.userMenu', locale)}
   493→                className="flex items-center gap-2 rounded-lg
   494→                  border border-[#2A2A38] bg-[#1A1A24]/80 py-1.5 pl-2 pr-3
   495→                  transition-all duration-200
   496→                  hover:scale-[1.02] hover:border-[#D4A574]/40
   497→                  focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-[#D4A574]/50
   498→                  active:scale-[0.98]"
   499→              >
   500→                {/* Avatar circle */}
   501→                <span
   502→                  className="flex h-7 w-7 shrink-0 items-center justify-center rounded-full text-xs font-bold text-[#1A1A24]"
   503→                  style={{
   504→                    background:
   505→                      'linear-gradient(135deg, #D4A574 0%, #E8C9A0 60%, #D4A574 100%)',
   506→                  }}
   507→                >
   508→                  {locale === 'ar' ? 'أم' : 'AM'}
   509→                </span>
   510→
   511→                {/* Name (hidden on small screens) */}
   512→                <span className="hidden text-sm font-medium text-[#E4E4EC] sm:block">
   513→                  {locale === 'ar' ? 'أحمد محمد' : 'Ahmed Mohamed'}
   514→                </span>
   515→
   516→                {/* Dropdown chevron */}
   517→                <ChevronDown
   518→                  size={14}
   519→                  strokeWidth={2}
   520→                  className="hidden text-[#8B8B9E] sm:block"
   521→                />
   522→              </button>
   523→            </DropdownMenuTrigger>
   524→            <DropdownMenuContent
   525→              sideOffset={8}
   526→              align="start"
   527→              className="w-60 rounded-xl border border-[#2A2A38] bg-[#1A1A24]/95 backdrop-blur-xl p-1.5 shadow-2xl shadow-black/40"
   528→            >
   529→              {/* User info header */}
   530→              <DropdownMenuLabel className="px-3 py-2.5">
