   249→interface QuickStatItem {
   250→  label: string;
   251→  value: number;
   252→  suffix?: string;
   253→  icon: React.ElementType;
   254→  color: string;
   255→  gradientFrom: string;
   256→  gradientTo: string;
   257→  trend?: string;
   258→}
   259→
   260→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   261→  const [count, setCount] = useState(0);
   262→  const ref = useRef<HTMLSpanElement>(null);
   263→  const started = useRef(false);
   264→
   265→  useEffect(() => {
   266→    if (started.current) return;
   267→    started.current = true;
   268→    const start = performance.now();
   269→    function step(now: number) {
   270→      const progress = Math.min((now - start) / duration, 1);
   271→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
   272→      setCount(Math.round(eased * target));
   273→      if (progress < 1) requestAnimationFrame(step);
   274→    }
   275→    requestAnimationFrame(step);
   276→  }, [target, duration]);
   277→
   278→  return <span ref={ref} className="tabular-nums">{count}</span>;
   279→}
