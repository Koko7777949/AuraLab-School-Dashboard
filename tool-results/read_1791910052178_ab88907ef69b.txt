   423→function AttendanceDistribution({ apiAttendance }: { apiAttendance: { rate: number; present: number; absent: number; late: number; excused: number; total: number } | null }) {
   424→  const total = apiAttendance?.total || 32;
   425→  const presentPct = apiAttendance && total > 0 ? Math.round((apiAttendance.present / total) * 100) : 85;
   426→  const latePct = apiAttendance && total > 0 ? Math.round((apiAttendance.late / total) * 100) : 8;
   427→  const absentPct = apiAttendance && total > 0 ? Math.round((apiAttendance.absent / total) * 100) : 7;
   428→
   429→  const localAttendanceData = [
   430→    { label: 'حاضر', percentage: presentPct, color: 'bg-emerald-500', textColor: 'text-emerald-400' },
   431→    { label: 'متأخر', percentage: latePct, color: 'bg-amber-500', textColor: 'text-amber-400' },
   432→    { label: 'غائب', percentage: absentPct, color: 'bg-rose-500', textColor: 'text-rose-400' },
   433→  ];
   434→
   435→  return (
   436→    <div className="flex flex-col gap-5">
   437→      {localAttendanceData.map((item) => (
   438→        <div key={item.label} className="flex flex-col gap-2">
   439→          <div className="flex items-center justify-between">
   440→            <span className="text-sm font-medium text-slate-300">{item.label}</span>
   441→            <span className={`text-sm font-semibold tabular-nums ${item.textColor}`}>
   442→              {item.percentage}%
   443→            </span>
   444→          </div>
   445→          {/* Bar track */}
   446→          <div className="h-3 w-full rounded-full bg-white/[0.06] overflow-hidden">
   447→            <div
   448→              className={`h-full rounded-full ${item.color} transition-all duration-700 ease-out`}
   449→              style={{ width: `${item.percentage}%` }}
   450→            />
   451→          </div>
   452→        </div>
   453→      ))}
   454→
   455→      {/* Summary */}
   456→      <div
   457→        className="mt-1 rounded-lg p-3"
   458→        style={{
   459→          backgroundColor: 'rgba(255,255,255,0.02)',
   460→          border: '1px solid rgba(255,255,255,0.06)',
   461→        }}
   462→      >
   463→        <div className="flex items-center justify-between">
   464→          <span className="text-xsB0B0C0">إجمالي الطلاب المسجلين</span>
   465→          <span className="text-sm font-bold text-white tabular-nums">{total} طالب</span>
   466→        </div>
   467→      </div>
   468→    </div>
   469→  );
   470→}
