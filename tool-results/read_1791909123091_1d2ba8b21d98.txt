   309→function getRelativeTime(dateString: string): string {
   310→  const now = Date.now();
   311→  const date = new Date(dateString).getTime();
   312→  const diffMs = now - date;
   313→  const diffSeconds = Math.floor(diffMs / 1000);
   314→  const diffMinutes = Math.floor(diffSeconds / 60);
   315→  const diffHours = Math.floor(diffMinutes / 60);
   316→  const diffDays = Math.floor(diffHours / 24);
   317→
   318→  if (diffSeconds < 60) return 'منذ لحظات';
   319→  if (diffMinutes < 60) return `منذ ${diffMinutes} دقيقة`;
   320→  if (diffHours < 24) return `منذ ${diffHours} ساعة`;
   321→  if (diffDays < 7) return `منذ ${diffDays} يوم`;
   322→  return new Date(dateString).toLocaleDateString('ar-SA');
   323→}
   324→
   325→function isExpired(expiresAt: string | null): boolean {
   326→  if (!expiresAt) return false;
   327→  return new Date(expiresAt) < new Date();
   328→}

   935→export default function AnnouncementsSection() {
   936→  // Local state for announcements (managed in-memory like other sections)
   937→  const [announcements, setAnnouncements] = useState<Announcement[]>([]);
   938→
   939→  // UI state

   942→  const [currentPage, setCurrentPage] = useState(1);
   943→  const [formOpen, setFormOpen] = useState(false);

   947→
   948→  // Initialize mock data once (useRef guard pattern)
   949→  const initialized = useRef(false);
   950→  useEffect(() => {
   951→    if (!initialized.current && announcements.length === 0) {
   952→      initialized.current = true;
   953→      requestAnimationFrame(() => setAnnouncements(INITIAL_ANNOUNCEMENTS));
   954→    }
   955→  }, []);
   956→
   957→  // Split into pinned and unpinned
   958→  const { pinned, unpinned } = useMemo(() => {
   959→    const allFiltered = announcements.filter((a) => {
   960→      // Category filter
   961→      if (categoryFilter !== 'all' && a.category !== categoryFilter) return false;
   962→      // Search filter
   963→      if (localSearch.trim()) {
   964→        const query = localSearch.trim().toLowerCase();
   965→        if (
   966→          !a.title.toLowerCase().includes(query) &&
   967→          !a.content.toLowerCase().includes(query)
   968→        ) {
   969→          return false;
   970→        }
   971→      }
   972→      return true;
   973→    });
   974→
   975→    return {
   976→      pinned: allFiltered.filter((a) => a.pinned),
   977→      unpinned: allFiltered.filter((a) => !a.pinned),
   978→    };
   979→  }, [announcements, categoryFilter, localSearch]);
   980→
   981→  // Pagination — only for unpinned (pinned always shown)
   982→  const totalUnpinned = unpinned.length;
   983→  const totalPages = Math.max(1, Math.ceil(totalUnpinned / ITEMS_PER_PAGE));
   984→  const paginatedUnpinned = unpinned.slice(
   985→    (currentPage - 1) * ITEMS_PER_PAGE,
   986→    currentPage * ITEMS_PER_PAGE,
   987→  );
   988→
   989→  // Reset to page 1 when filters change
   990→  const prevFiltersRef = useRef({ categoryFilter, localSearch });
   991→  useEffect(() => {
