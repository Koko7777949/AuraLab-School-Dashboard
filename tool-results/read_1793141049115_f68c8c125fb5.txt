   433→function AttendanceDistribution({ apiAttendance }: { apiAttendance: { rate: number; present: number; absent: number; late: number; excused: number; total: number } | null }) {
   434→  const total = apiAttendance?.total || 32;
   435→  const presentPct = apiAttendance && total > 0 ? Math.round((apiAttendance.present / total) * 100) : 85;
   436→  const latePct = apiAttendance && total > 0 ? Math.round((apiAttendance.late / total) * 100) : 8;
   437→  const absentPct = apiAttendance && total > 0 ? Math.round((apiAttendance.absent / total) * 100) : 7;
   438→
   439→  const localAttendanceData = [
   440→    { label: 'حاضر', percentage: presentPct, color: 'bg-emerald-500', textColor: 'text-emerald-400' },
   441→    { label: 'متأخر', percentage: latePct, color: 'bg-amber-500', textColor: 'text-amber-400' },
   442→    { label: 'غائب', percentage: absentPct, color: 'bg-rose-500', textColor: 'text-rose-400' },
   443→  ];
   444→
   445→  return (
   446→    <div className="flex flex-col gap-5">
   447→      {localAttendanceData.map((item) => (
   448→        <div key={item.label} className="flex flex-col gap-2">
   449→          <div className="flex items-center justify-between">
   450→            <span className="text-sm font-medium text-slate-300">{item.label}</span>
   451→            <span className={`text-sm font-semibold tabular-nums ${item.textColor}`}>
   452→              {item.percentage}%
   453→            </span>
   454→          </div>
   455→          {/* Bar track */}
   456→          <div className="h-3 w-full rounded-full bg-white/[0.06] overflow-hidden">
   457→            <div
   458→              className={`h-full rounded-full ${item.color} transition-all duration-700 ease-out`}
   459→              style={{ width: `${item.percentage}%` }}
   460→            />
   461→          </div>
   462→        </div>
   463→      ))}
   464→
   465→      {/* Summary */}
   466→      <div
   467→        className="mt-1 rounded-lg p-3"
   468→        style={{
   469→          backgroundColor: 'rgba(255,255,255,0.02)',
   470→          border: '1px solid rgba(255,255,255,0.06)',
   471→        }}
   472→      >
   473→        <div className="flex items-center justify-between">
   474→          <span className="text-xsB0B0C0">إجمالي الطلاب المسجلين</span>
   475→          <span className="text-sm font-bold text-white tabular-nums">{total} طالب</span>
   476→        </div>
   477→      </div>
   478→    </div>
   479→  );
   480→}
