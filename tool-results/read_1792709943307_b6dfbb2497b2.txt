   147→function getCapacityColor(percentage: number): { bar: string; text: string } {
   148→  if (percentage >= 90) {
   149→    return { bar: 'bg-rose-500', text: 'text-rose-400' };
   150→  }
   151→  if (percentage >= 70) {
   152→    return { bar: 'bg-amber-500', text: 'text-amber-400' };
   153→  }
   154→  return { bar: 'bg-emerald-500', text: 'text-emerald-400' };
   155→}

   315→function ClassFormContent({
   316→  editingClass,
   317→  onClose,
   318→}: {
   319→  editingClass: ClassRoom | null;
   320→  onClose: () => void;
   321→}) {
   322→  const { addClass, updateClass } = useAppStore();
   323→  const isEditing = !!editingClass;
   324→
   325→  // Form state
   326→  const [name, setName] = useState(editingClass?.name ?? '');
   327→  const [grade, setGrade] = useState(editingClass?.grade ?? 'الأول');
   328→  const [room, setRoom] = useState(editingClass?.room ?? '');
   329→  const [notes, setNotes] = useState(editingClass?.notes ?? '');
   330→
   331→  const handleSubmit = (e: React.FormEvent) => {
   332→    e.preventDefault();
   333→    if (!name.trim() || !room.trim()) return;
   334→
   335→    const now = new Date().toISOString();
   336→
   337→    if (isEditing && editingClass) {
   338→      updateClass(editingClass.id, {
   339→        name: name.trim(),
   340→        grade,
   341→        room: room.trim(),
   342→        notes: notes.trim(),
   343→        updatedAt: now,
   344→      });
   345→      toast.success('تم تحديث بيانات الصف');
   346→    } else {
   347→      const newClass: ClassRoom = {
   348→        id: `cls-${Date.now()}`,
   349→        name: name.trim(),
   350→        grade,
   351→        room: room.trim(),
   352→        notes: notes.trim(),
   353→        createdAt: now,
   354→        updatedAt: now,
   355→        studentCount: 0,
   356→        teacherNames: [],
   357→      };
   358→      addClass(newClass);
   359→      toast.success('تم إضافة الصف بنجاح');
   360→    }
   361→
   362→    onClose();
   363→  };
   364→
   365→  return (
   366→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   367→      {/* Class name */}
   368→      <div className="flex flex-col gap-2">
   369→        <Label className="text-slate-300 text-sm">اسم الصف</Label>
   370→        <Input
   371→          value={name}
   372→          onChange={(e) => setName(e.target.value)}
   373→          placeholder="مثال: الصف الأول أ"
   374→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   375→          required
   376→        />
   377→      </div>
   378→
   379→      {/* Grade & Room row */}
   380→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   381→        {/* Grade dropdown */}
   382→        <div className="flex flex-col gap-2">
   383→          <Label className="text-slate-300 text-sm">المرحلة الدراسية</Label>
   384→          <Select value={grade} onValueChange={setGrade}>
   385→            <SelectTrigger className="w-full bg-white/[0.03] border-white/[0.08] text-slate-200 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20">
   386→              <SelectValue placeholder="اختر المرحلة" />
   387→            </SelectTrigger>
   388→            <SelectContent className="bg-[#1A1A24] border-white/[0.08]">
   389→              <SelectItem value="الأول" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   390→                الصف الأول
   391→              </SelectItem>
   392→              <SelectItem value="الثاني" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   393→                الصف الثاني
   394→              </SelectItem>
   395→              <SelectItem value="الثالث" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   396→                الصف الثالث
   397→              </SelectItem>
   398→            </SelectContent>
   399→          </Select>
   400→        </div>
   401→
   402→        {/* Room */}
   403→        <div className="flex flex-col gap-2">
   404→          <Label className="text-slate-300 text-sm">رقم القاعة</Label>
   405→          <Input
   406→            value={room}
   407→            onChange={(e) => setRoom(e.target.value)}

   415→      {/* Notes */}

   498→/** Delete confirmation dialog */
   499→function DeleteClassDialog({
   500→  cls,
   501→  open,
   502→  onOpenChange,
   503→}: {
   504→  cls: ClassRoom | null;
   505→  open: boolean;
   506→  onOpenChange: (open: boolean) => void;
   507→}) {
   508→  const { deleteClass, addToTrash } = useAppStore();
   509→
   510→  const handleDelete = () => {
   511→    if (cls) {
   512→      addToTrash({
   513→        id: `trash-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   514→        itemType: 'class',
   515→        itemId: cls.id,
   516→        itemName: cls.name,
   517→        deletedBy: 'المدير',
   518→        dataJson: JSON.stringify(cls),
   519→        restoredAt: null,
   520→        createdAt: new Date().toISOString(),
   521→      });
   522→      deleteClass(cls.id);
   523→      toast.success('تم حذف الصف');
   524→      onOpenChange(false);
   525→    }
   526→  };
   527→
   528→  return (
   529→    <AlertDialog open={open} onOpenChange={onOpenChange}>
   530→      <AlertDialogContent
   531→        dir="rtl"
   532→        style={{
   533→          backgroundColor: '#1A1A24',
   534→          border: '1px solid rgba(255,255,255,0.06)',
   535→        }}
   536→      >
   537→        <AlertDialogHeader>
   538→          <AlertDialogTitle className="text-white text-lg">
   539→            تأكيد حذف الصف
   540→          </AlertDialogTitle>
   541→          <AlertDialogDescription className="text-muted-foreground leading-relaxed">
   542→            هل أنت متأكد من حذف الصف{' '}
