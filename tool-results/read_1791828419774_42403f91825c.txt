   222→function getClassName(classId: string | null): string {

   228→function formatDate(dateStr: string): string {
   229→  const date = new Date(dateStr);
   230→  return date.toLocaleDateString('ar-SA', {
   231→    year: 'numeric',
   232→    month: 'short',
   233→    day: 'numeric',
   234→  });
   235→}

   238→  const date = new Date(dateStr);

   455→function EditFormContent({
   456→  record,
   457→  onClose,
   458→}: {
   459→  record: AttendanceRecord | null;
   460→  onClose: () => void;
   461→}) {
   462→  const { updateAttendanceRecord } = useAttendanceStore();
   463→  const [status, setStatus] = useState<AttendanceStatus>(record?.status ?? 'present');
   464→  const [notes, setNotes] = useState(record?.notes ?? '');
   465→
   466→  const handleSubmit = (e: React.FormEvent) => {
   467→    e.preventDefault();
   468→    if (!record) return;
   469→
   470→    updateAttendanceRecord(record.id, { status, notes });
   471→    toast.success('تم تحديث سجل الحضور');
   472→    onClose();
   473→  };
   474→
   475→  return (
   476→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   477→      {/* Status radio buttons */}
   478→      <div className="flex flex-col gap-2">
   479→        <Label className="text-slate-300 text-sm">حالة الحضور</Label>
   480→        <div className="grid grid-cols-2 gap-2">
   481→          {STATUS_OPTIONS.map((opt) => {
   482→            const config = STATUS_CONFIG[opt.value];
   483→            const Icon = config.icon;
   484→            const isSelected = status === opt.value;
   485→            return (
   486→              <button
   487→                key={opt.value}
   488→                type="button"
   489→                onClick={() => setStatus(opt.value)}
   490→                className={cn(
   491→                  'flex items-center justify-center gap-2 rounded-xl py-3 px-4 border transition-all duration-200',
   492→                  isSelected
   493→                    ? config.bgClass + ' ' + config.color + ' font-semibold'
   494→                    : 'border-white/[0.06] text-slate-400 hover:bg-white/[0.04] hover:text-slate-300'
   495→                )}
   496→              >
   497→                <Icon className="h-4 w-4" />
   498→                <span className="text-sm">{opt.label}</span>
   499→              </button>
   500→            );
   501→          })}
   502→        </div>
   503→      </div>
   504→
   505→      {/* Notes */}
   506→      <div className="flex flex-col gap-2">
   507→        <Label className="text-slate-300 text-sm">ملاحظات</Label>
   508→        <Textarea
   509→          value={notes}
   510→          onChange={(e) => setNotes(e.target.value)}
   511→          placeholder="أضف ملاحظات حول الحضور..."
   512→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20 min-h-[80px] resize-none"
   513→        />
   514→      </div>
   515→
   516→      <DialogFooter className="flex-row gap-3 sm:justify-start pt-2">
   517→        <Button
   518→          type="submit"
   519→          className="bg-[#D4A574] hover:bg-[#B8895A] text-slate-900 font-semibold rounded-lg px-6 gap-2"
   520→        >
   521→          <Save className="h-4 w-4" />
   522→          حفظ التغييرات
   523→        </Button>
   524→        <Button
   525→          type="button"
   526→          variant="outline"
   527→          onClick={onClose}
   528→          className="border-white/[0.08] text-slate-300 hover:bg-white/[0.05] hover:text-white rounded-lg px-6"

   815→function AttendanceHeatmap({
   816→  records,
   817→  currentMonth,
   818→  currentYear,
   819→  onMonthChange,
   820→  onYearChange,
   821→}: {
   822→  records: AttendanceRecord[];
   823→  currentMonth: number;
   824→  currentYear: number;
   825→  onMonthChange: (month: number) => void;
   826→  onYearChange: (year: number) => void;
   827→}) {
   828→  const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
   829→  const firstDayOfMonth = new Date(currentYear, currentMonth, 1).getDay();
   830→
   831→  // Group records by date for this month
   832→  const dateMap = new Map<string, AttendanceRecord[]>();
   833→  for (const rec of records) {
   834→    const d = new Date(rec.date);
   835→    if (d.getMonth() === currentMonth && d.getFullYear() === currentYear) {
   836→      const key = d.getDate().toString();
   837→      if (!dateMap.has(key)) dateMap.set(key, []);
   838→      dateMap.get(key)!.push(rec);
   839→    }
   840→  }
   841→
   842→  const getCellColor = (day: number) => {
   843→    const dayRecords = dateMap.get(day.toString());
   844→    if (!dayRecords || dayRecords.length === 0) return 'bg-white/[0.02]';
   845→    const presentCount = dayRecords.filter((r) => r.status === 'present').length;
   846→    const ratio = presentCount / dayRecords.length;
   847→    if (ratio >= 0.9) return 'bg-emerald-500/40';
   848→    if (ratio >= 0.7) return 'bg-emerald-500/25';
   849→    if (ratio >= 0.5) return 'bg-amber-500/25';
   850→    if (ratio > 0) return 'bg-rose-500/25';
   851→    return 'bg-rose-500/40';
   852→  };
   853→
   854→  const getDaySummary = (day: number) => {
   855→    const dayRecords = dateMap.get(day.toString());
   856→    if (!dayRecords || dayRecords.length === 0) return null;
   857→    const present = dayRecords.filter((r) => r.status === 'present').length;
   858→    const absent = dayRecords.filter((r) => r.status === 'absent').length;
   859→    const late = dayRecords.filter((r) => r.status === 'late').length;
   860→    const excused = dayRecords.filter((r) => r.status === 'excused').length;
   861→    return { total: dayRecords.length, present, absent, late, excused };
   862→  };
   863→
   864→  const handlePrevMonth = () => {
   865→    if (currentMonth === 0) {
   866→      onMonthChange(11);
   867→      onYearChange(currentYear - 1);
   868→    } else {
   869→      onMonthChange(currentMonth - 1);
   870→    }
   871→  };
   872→
   873→  const handleNextMonth = () => {
   874→    if (currentMonth === 11) {
   875→      onMonthChange(0);
   876→      onYearChange(currentYear + 1);
   877→    } else {
   878→      onMonthChange(currentMonth + 1);
   879→    }
   880→  };
   881→
   882→  // Build calendar grid (6 rows × 7 columns)
   883→  const cells: (number | null)[] = [];
   884→  // Add empty cells for offset (Saturday-first week: shift by (firstDay + 1) % 7)
   885→  const offset = (firstDayOfMonth + 1) % 7;
   886→  for (let i = 0; i < offset; i++) cells.push(null);
   887→  for (let d = 1; d <= daysInMonth; d++) cells.push(d);
   888→  while (cells.length % 7 !== 0) cells.push(null);
   889→
   890→  return (
   891→    <div
   892→      className="rounded-xl p-4"
   893→      style={{
   894→        backgroundColor: 'rgba(255,255,255,0.02)',
   895→        border: '1px solid rgba(255,255,255,0.06)',
   896→      }}
   897→    >
   898→      {/* Month navigation */}
   899→      <div className="flex items-center justify-between mb-4">
   900→        <h3 className="text-sm font-semibold text-slate-300 flex items-center gap-2">
   901→          <BarChart3 className="h-4 w-4 text-[#D4A574]" />
   902→          خريطة الحضور الشهرية
   903→        </h3>
   904→        <div className="flex items-center gap-2">
   905→          <Button
   906→            variant="ghost"
   907→            size="icon"
   908→            onClick={handlePrevMonth}
   909→            className="h-7 w-7 text-slate-400 hover:text-white hover:bg-white/[0.05] rounded-lg"
   910→          >
   911→            <ChevronRight className="h-4 w-4" />
   912→          </Button>
   913→          <span className="text-sm font-medium text-slate-200 min-w-[120px] text-center">
   914→            {ARABIC_MONTHS[currentMonth]} {currentYear}
   915→          </span>
   916→          <Button
   917→            variant="ghost"
   918→            size="icon"
   919→            onClick={handleNextMonth}
   920→            className="h-7 w-7 text-slate-400 hover:text-white hover:bg-white/[0.05] rounded-lg"
   921→          >
   922→            <ChevronLeft className="h-4 w-4" />
   923→          </Button>
   924→        </div>
   925→      </div>
   926→
   927→      {/* Day headers */}
   928→      <div className="grid grid-cols-7 gap-1 mb-1">
   929→        {ARABIC_DAYS.map((day) => (
   930→          <div key={day} className="text-center text-[10px] font-semibold text-muted-foreground/60 py-1">
   931→            {day}
   932→          </div>
   933→        ))}
   934→      </div>
   935→
   936→      {/* Calendar cells */}
   937→      <div className="grid grid-cols-7 gap-1">
   938→        {cells.map((day, i) => {
   939→          if (day === null) {
   940→            return <div key={`empty-${i}`} className="aspect-square" />;
   941→          }
   942→
   943→          const summary = getDaySummary(day);
   944→          const isToday =
   945→            day === new Date().getDate() &&
   946→            currentMonth === new Date().getMonth() &&
   947→            currentYear === new Date().getFullYear();
   948→
   949→          return (
   950→            <div
   951→              key={day}
   952→              className={cn(
   953→                'aspect-square rounded-lg flex flex-col items-center justify-center text-xs relative transition-all duration-200 cursor-default',
   954→                getCellColor(day),
   955→                isToday && 'ring-1 ring-[#D4A574]/60 ring-offset-1 ring-offset-[#121218]'
   956→              )}
   957→              title={summary ? `حاضر: ${summary.present}, غائب: ${summary.absent}, متأخر: ${summary.late}, مستأذن: ${summary.excused}` : 'لا توجد بيانات'}
   958→            >
   959→              <span className={cn(
   960→                'font-medium tabular-nums',
   961→                summary ? 'text-slate-200' : 'text-muted-foreground/40'
   962→              )}>
   963→                {day}
   964→              </span>
   965→              {summary && summary.total > 0 && (
   966→                <span className="text-[9px] text-muted-foreground/50 mt-0.5">
   967→                  {summary.total}
   968→                </span>
   969→              )}
   970→            </div>
   971→          );
   972→        })}
   973→      </div>
   974→
   975→      {/* Legend */}
   976→      <div className="flex items-center justify-center gap-4 mt-3 pt-3 border-t border-white/[0.04]">
   977→        <div className="flex items-center gap-1.5">
   978→          <div className="h-3 w-3 rounded bg-emerald-500/40" />
   979→          <span className="text-[10px] text-muted-foreground">ممتاز</span>
   980→        </div>
   981→        <div className="flex items-center gap-1.5">
   982→          <div className="h-3 w-3 rounded bg-emerald-500/25" />
   983→          <span className="text-[10px] text-muted-foreground">جيد</span>
   984→        </div>
   985→        <div className="flex items-center gap-1.5">
   986→          <div className="h-3 w-3 rounded bg-amber-500/25" />
   987→          <span className="text-[10px] text-muted-foreground">متوسط</span>
   988→        </div>

   990→          <div className="h-3 w-3 rounded bg-rose-500/25" />
