   191→function formatCurrency(amount: number): string {
   192→  return new Intl.NumberFormat('ar-SA', {
   193→    style: 'currency',
   194→    currency: 'SAR',
   195→    minimumFractionDigits: 0,
   196→    maximumFractionDigits: 0,
   197→  }).format(amount);
   198→}
   199→
   200→function formatDate(dateStr: string): string {
   201→  if (!dateStr) return '—';
   202→  try {
   203→    return new Date(dateStr).toLocaleDateString('ar-SA', {
   204→      year: 'numeric',
   205→      month: 'short',
   206→      day: 'numeric',
   207→    });
   208→  } catch {
   209→    return dateStr;
   210→  }
   211→}
