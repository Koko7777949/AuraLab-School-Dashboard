   190→function formatCurrency(amount: number): string {
   191→  return new Intl.NumberFormat('ar-SA', {
   192→    style: 'currency',
   193→    currency: 'SAR',
   194→    minimumFractionDigits: 0,
   195→    maximumFractionDigits: 0,
   196→  }).format(amount);
   197→}
   198→
   199→function formatDate(dateStr: string): string {
   200→  if (!dateStr) return '—';
   201→  try {
   202→    return new Date(dateStr).toLocaleDateString('ar-SA', {
   203→      year: 'numeric',
   204→      month: 'short',
   205→      day: 'numeric',
   206→    });
   207→  } catch {
   208→    return dateStr;
   209→  }
   210→}
