   154→function SparklineChart({ data, width = 200, height = 50 }: { data: number[]; width?: number; height?: number }) {
   155→  if (data.length < 2) {
   156→    return (
   157→      <svg width={width} height={height} className="block">
   158→        <text x={width / 2} y={height / 2} textAnchor="middle" dominantBaseline="central" fill="#64748b" fontSize="10">
   159→          لا توجد بيانات كافية
   160→        </text>
   161→      </svg>
   162→    );
   163→  }
   164→
   165→  const padding = 4;
   166→  const chartW = width - padding * 2;
   167→  const chartH = height - padding * 2;
   168→  const min = Math.min(...data);
   169→  const max = Math.max(...data);
   170→  const range = max - min || 1;
   171→
   172→  const points = data.map((v, i) => ({
   173→    x: padding + (i / (data.length - 1)) * chartW,
   174→    y: padding + chartH - ((v - min) / range) * chartH,
   175→  }));
   176→
   177→  // Build smooth path using catmull-rom to bezier
   178→  const pathD = points.reduce((acc, point, idx) => {
   179→    if (idx === 0) return `M ${point.x} ${point.y}`;
   180→    const prev = points[idx - 1];
   181→    const cpx1 = prev.x + (point.x - prev.x) * 0.4;
   182→    const cpx2 = prev.x + (point.x - prev.x) * 0.6;
   183→    return `${acc} C ${cpx1} ${prev.y}, ${cpx2} ${point.y}, ${point.x} ${point.y}`;
   184→  }, '');
   185→
   186→  // Area fill path
   187→  const areaD = `${pathD} L ${points[points.length - 1].x} ${height - padding} L ${points[0].x} ${height - padding} Z`;
   188→
   189→  const gradientId = `sparkline-grad-${Math.random().toString(36).slice(2, 7)}`;
   190→
   191→  return (
   192→    <svg width={width} height={height} className="block" preserveAspectRatio="xMidYMid meet">
   193→      <defs>
   194→        <linearGradient id={gradientId} x1="0" y1="0" x2="0" y2="1">
   195→          <stop offset="0%" stopColor="#D4A574" stopOpacity="0.3" />
   196→          <stop offset="100%" stopColor="#D4A574" stopOpacity="0" />
   197→        </linearGradient>
   198→      </defs>
   199→      {/* Area fill */}
   200→      <path d={areaD} fill={`url(#${gradientId})`} />
   201→      {/* Line */}
   202→      <path d={pathD} fill="none" stroke="#D4A574" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round" />
   203→      {/* End dot */}
   204→      <circle cx={points[points.length - 1].x} cy={points[points.length - 1].y} r="3" fill="#D4A574" />
   205→      <circle cx={points[points.length - 1].x} cy={points[points.length - 1].y} r="5" fill="#D4A574" opacity="0.3" />
   206→    </svg>
   207→  );
   208→}

   214→function GradeDistributionBar({ grades }: { grades: GradeRecord[] }) {
   215→  const distribution = useMemo(() => {
   216→    const counts = { excellent: 0, veryGood: 0, good: 0, pass: 0, fail: 0 };
   217→    for (const g of grades) {
   218→      const pct = (g.score / g.maxScore) * 100;
   219→      if (pct >= 90) counts.excellent++;
   220→      else if (pct >= 80) counts.veryGood++;
   221→      else if (pct >= 70) counts.good++;
   222→      else if (pct >= 60) counts.pass++;
   223→      else counts.fail++;
   224→    }
   225→    return counts;
   226→  }, [grades]);
   227→
   228→  const total = grades.length || 1;
   229→  const segments = [
   230→    { label: 'ممتاز', count: distribution.excellent, color: '#34d399' },
   231→    { label: 'جيد جداً', count: distribution.veryGood, color: '#D4A574' },
   232→    { label: 'جيد', count: distribution.good, color: '#fbbf24' },
   233→    { label: 'مقبول', count: distribution.pass, color: '#fb923c' },
   234→    { label: 'ضعيف', count: distribution.fail, color: '#fb7185' },
   235→  ];
   236→
   237→  return (
   238→    <div className="flex flex-col gap-2">
   239→      <div className="flex rounded-full overflow-hidden h-3 w-full">
   240→        {segments.map((seg) => (
   241→          <div
   242→            key={seg.label}
   243→            style={{
   244→              width: `${(seg.count / total) * 100}%`,
   245→              backgroundColor: seg.color,
   246→              minWidth: seg.count > 0 ? 4 : 0,
   247→            }}
   248→            className="transition-all duration-500"
   249→            title={`${seg.label}: ${seg.count}`}
   250→          />
   251→        ))}
   252→      </div>
   253→      <div className="flex flex-wrap gap-x-4 gap-y-1">
   254→        {segments.map((seg) => (
   255→          <div key={seg.label} className="flex items-center gap-1.5 text-xs">
   256→            <div className="h-2 w-2 rounded-full" style={{ backgroundColor: seg.color }} />
   257→            <span className="text-muted-foreground">{seg.label}</span>
   258→            <span className="text-slate-300 font-semibold tabular-nums">{seg.count}</span>
   259→          </div>
   260→        ))}
   261→      </div>
   262→    </div>
   263→  );
   264→}

   270→function AttendanceHeatmap({ records }: { records: Array<{ date: string; status: string }> }) {
   271→  const today = new Date();
   272→  const days: Array<{ date: string; dayOfWeek: number; day: number; status: string | null }> = [];
   273→
   274→  for (let i = 29; i >= 0; i--) {
   275→    const d = new Date(today);
   276→    d.setDate(d.getDate() - i);
   277→    const dateStr = d.toISOString().split('T')[0];
   278→    const record = records.find((r) => r.date === dateStr);
   279→    days.push({
   280→      date: dateStr,
   281→      dayOfWeek: d.getDay(),
   282→      day: d.getDate(),
   283→      status: record?.status ?? null,
   284→    });
   285→  }
   286→
   287→  const cellSize = 16;
   288→  const gap = 2;
   289→  const cols = 7; // Mon-Sun (Sat-first for Arabic, but 7 cols grid)
   290→
   291→  return (
   292→    <div className="glass-card-elevated rounded-xl p-4">
   293→      <h3 className="text-sm font-semibold text-slate-300 mb-3 flex items-center gap-2">
   294→        <Calendar className="h-4 w-4 text-[#D4A574]" />
   295→        خريطة الحضور - آخر 30 يوم
   296→      </h3>
   297→      <div className="flex gap-1 justify-center flex-wrap">
   298→        {days.map((day) => {
   299→          const isWeekend = day.dayOfWeek === 5 || day.dayOfWeek === 6;
   300→          const statusInfo = day.status ? ATTENDANCE_STATUS_MAP[day.status] : null;
   301→          const bgColor = isWeekend
   302→            ? 'rgba(255,255,255,0.02)'
   303→            : statusInfo
   304→              ? `${statusInfo.color}30`
   305→              : 'rgba(255,255,255,0.04)';
   306→          const dotColor = statusInfo?.color ?? (isWeekend ? 'transparent' : 'rgba(255,255,255,0.08)');
   307→
   308→          return (
   309→            <div
   310→              key={day.date}
   311→              className="flex flex-col items-center gap-0.5"
   312→              title={`${day.date}: ${statusInfo?.label ?? (isWeekend ? 'عطلة' : 'لا بيانات')}`}
   313→            >
   314→              <span className="text-[9px] text-muted-foreground/50 tabular-nums">{day.day}</span>
   315→              <div
   316→                className="rounded-sm flex items-center justify-center"
   317→                style={{
   318→                  width: cellSize,
   319→                  height: cellSize,
   320→                  backgroundColor: bgColor,
   321→                  border: '1px solid rgba(255,255,255,0.04)',
   322→                }}
   323→              >
   324→                {!isWeekend && (
   325→                  <div
   326→                    className="rounded-full"
   327→                    style={{
   328→                      width: 6,
   329→                      height: 6,
   330→                      backgroundColor: dotColor,
   331→                    }}
   332→                  />
   333→                )}
   334→              </div>
   335→            </div>
   336→          );
   337→        })}
   338→      </div>
   339→      {/* Legend */}
   340→      <div className="flex items-center justify-center gap-4 mt-3">
   341→        {Object.values(ATTENDANCE_STATUS_MAP).map((s) => (
   342→          <div key={s.label} className="flex items-center gap-1 text-[10px] text-muted-foreground">
   343→            <div className="h-2.5 w-2.5 rounded-sm" style={{ backgroundColor: `${s.color}30` }} />
   344→            <span>{s.label}</span>
   345→          </div>
   346→        ))}
   347→      </div>
   348→    </div>
   349→  );
   350→}
