   206→function getDayLabel(day: DayOfWeek): string {
   207→  return DAYS.find((d) => d.value === day)?.label ?? day;
   208→}
   209→
   210→function getPeriodLabel(period: number): string {
   211→  const p = PERIODS.find((pr) => pr.value === period);
   212→  return p ? `${p.label} (${p.time})` : `الحصة ${period}`;
   213→}
   214→
   215→function getPeriodTime(period: number): string {
   216→  const p = PERIODS.find((pr) => pr.value === period);
   217→  return p ? p.time : '';
   218→}

   338→function AddEditBlockDialog({
   339→  block,
   340→  open,
   341→  onOpenChange,
   342→  defaultDay,
   343→  defaultPeriod,
   344→}: {
   345→  block: ScheduleBlock | null;
   346→  open: boolean;
   347→  onOpenChange: (open: boolean) => void;
   348→  defaultDay?: DayOfWeek;
   349→  defaultPeriod?: number;
   350→}) {
   351→  const { addScheduleBlock, updateScheduleBlock, teachers, classes } = useAppStore();
   352→  const isEdit = !!block;
   353→
   354→  const [day, setDay] = useState<DayOfWeek>(defaultDay ?? 'sat');
   355→  const [period, setPeriod] = useState<number>(defaultPeriod ?? 1);
   356→  const [subject, setSubject] = useState('');
   357→  const [teacherId, setTeacherId] = useState<string>('');
   358→  const [classId, setClassId] = useState<string>('');
   359→  const [room, setRoom] = useState('');
   360→  const [startTime, setStartTime] = useState('');
   361→  const [endTime, setEndTime] = useState('');
   362→  const [color, setColor] = useState<string>(COLOR_PALETTE[0].value);
   363→  const [notes, setNotes] = useState('');
   364→  const [subjectSuggestionsOpen, setSubjectSuggestionsOpen] = useState(false);
   365→
   366→  // Populate fields when editing or when dialog opens for adding
   367→  const prevOpenRef = useRef(false);
   368→  useEffect(() => {
   369→    if (open && !prevOpenRef.current) {
   370→      if (block) {
   371→        requestAnimationFrame(() => {
   372→          setDay(block.dayOfWeek);
   373→          setPeriod(block.period);
   374→          setSubject(block.subject);
   375→          setTeacherId(block.teacherId ?? '');
   376→          setClassId(block.classId ?? '');
   377→          setRoom(block.room);
   378→          setStartTime(block.startTime);
   379→          setEndTime(block.endTime);
   380→          setColor(block.color);
   381→          setNotes(block.notes);
   382→        });
   383→      } else {
   384→        const defDay = defaultDay ?? 'sat';
   385→        const defPeriod = defaultPeriod ?? 1;
   386→        const pDef = PERIODS.find((p) => p.value === defPeriod);
   387→        const [s, e] = pDef ? pDef.time.split(' - ') : ['7:30', '8:15'];
   388→        requestAnimationFrame(() => {
   389→          setDay(defDay);
   390→          setPeriod(defPeriod);
   391→          setSubject('');
   392→          setTeacherId('');
   393→          setClassId('');
   394→          setRoom('');
   395→          setStartTime(s);
   396→          setEndTime(e);
   397→          setColor(COLOR_PALETTE[0].value);
   398→          setNotes('');
   399→        });
   400→      }
   401→      requestAnimationFrame(() => setSubjectSuggestionsOpen(false));
   402→    }
   403→    prevOpenRef.current = open;
   404→  }, [open, block, defaultDay, defaultPeriod]);
   405→
   406→  // Update times when period changes
   407→  const handlePeriodChange = useCallback((val: string) => {
   408→    const num = parseInt(val, 10);
   409→    setPeriod(num);
   410→    const pDef = PERIODS.find((p) => p.value === num);
   411→    if (pDef) {
   412→      const [s, e] = pDef.time.split(' - ');
   413→      setStartTime(s);
   414→      setEndTime(e);
   415→    }
   416→  }, []);
   417→
   418→  // Filter subject suggestions
   419→  const subjectSuggestions = useMemo(() => {
   420→    if (!subject.trim() || subjectSuggestionsOpen === false) return [];
   421→    return SUBJECTS.filter((s) => s.includes(subject.trim())).slice(0, 6);
   422→  }, [subject, subjectSuggestionsOpen]);
   423→
   424→  const handleSubmit = useCallback(() => {
   425→    if (!subject.trim()) {
   426→      toast.error('يرجى إدخال اسم المادة');
   427→      return;
   428→    }
   429→
   430→    const teacher = teachers.find((t) => t.id === teacherId);
   431→    const cls = classes.find((c) => c.id === classId);
   432→
   433→    if (isEdit && block) {
   434→      updateScheduleBlock(block.id, {
   435→        dayOfWeek: day,
   436→        period,
   437→        subject: subject.trim(),
   438→        teacherId: teacherId || null,
   439→        teacherName: teacher?.name,
   440→        classId: classId || null,
   441→        className: cls?.name,
   442→        room: room.trim() || '—',
   443→        startTime,
   444→        endTime,
   445→        color,
   446→        notes: notes.trim(),
   447→      });
   448→      toast.success('تم تحديث الحصة بنجاح');
   449→    } else {
   450→      addScheduleBlock({
   451→        id: `sch-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   452→        dayOfWeek: day,
   453→        period,
   454→        subject: subject.trim(),
   455→        teacherId: teacherId || null,
   456→        teacherName: teacher?.name,
   457→        classId: classId || null,
   458→        className: cls?.name,
   459→        room: room.trim() || '—',
   460→        startTime,
   461→        endTime,
   462→        color,
   463→        notes: notes.trim(),
   464→      });
   465→      toast.success('تمت إضافة الحصة بنجاح');
   466→    }
   467→    onOpenChange(false);
   468→  }, [subject, day, period, teacherId, classId, room, startTime, endTime, color, notes, isEdit, block, teachers, classes, addScheduleBlock, updateScheduleBlock, onOpenChange]);
   469→
   470→  return (
   471→    <Dialog open={open} onOpenChange={onOpenChange}>
   472→      <DialogContent
   473→        className="sm:max-w-lg max-h-[85vh] overflow-y-auto"
   474→        dir="rtl"
   475→        style={{
   476→          backgroundColor: '#1A1A24',
   477→          border: '1px solid rgba(255,255,255,0.06)',
   478→        }}
   479→      >
   480→        <DialogHeader>
   481→          <DialogTitle className="text-white text-xl">
   482→            {isEdit ? 'تعديل الحصة' : 'إضافة حصة جديدة'}

   913→function ListView({
   914→  blocks,
   915→  onBlockClick,
   916→  filterDay,
   917→  setFilterDay,
   918→  filterSubject,
   919→  setFilterSubject,
   920→}: {
   921→  blocks: ScheduleBlock[];
   922→  onBlockClick: (block: ScheduleBlock) => void;
   923→  filterDay: string;
   924→  setFilterDay: (d: string) => void;
   925→  filterSubject: string;
   926→  setFilterSubject: (s: string) => void;
   927→}) {
   928→  const uniqueSubjects = useMemo(() => {
   929→    return Array.from(new Set(blocks.map((b) => b.subject))).sort();
   930→  }, [blocks]);
   931→
   932→  const filteredBlocks = useMemo(() => {
   933→    let result = [...blocks];
   934→    if (filterDay !== 'all') {
   935→      result = result.filter((b) => b.dayOfWeek === filterDay);
   936→    }
   937→    if (filterSubject !== 'all') {
   938→      result = result.filter((b) => b.subject === filterSubject);
   939→    }
   940→    // Sort by day then period
   941→    const dayOrder: Record<string, number> = { sat: 0, sun: 1, mon: 2, tue: 3, wed: 4, thu: 5 };
   942→    result.sort((a, b) => {
   943→      const dayDiff = (dayOrder[a.dayOfWeek] ?? 0) - (dayOrder[b.dayOfWeek] ?? 0);
   944→      if (dayDiff !== 0) return dayDiff;
   945→      return a.period - b.period;
   946→    });
   947→    return result;
   948→  }, [blocks, filterDay, filterSubject]);
   949→
   950→  // Group by day
   951→  const groupedByDay = useMemo(() => {
   952→    const groups: Record<string, ScheduleBlock[]> = {};
   953→    for (const b of filteredBlocks) {
   954→      const dayLabel = getDayLabel(b.dayOfWeek);
   955→      if (!groups[dayLabel]) groups[dayLabel] = [];
   956→      groups[dayLabel].push(b);
   957→    }
   958→    return groups;
   959→  }, [filteredBlocks]);
   960→
   961→  return (
   962→    <div className="flex flex-col gap-4">
   963→      {/* Filters */}
   964→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   965→        <div className="flex flex-col gap-2">
   966→          <Label className="text-slate-300 text-sm">تصفية حسب اليوم</Label>
   967→          <Select value={filterDay} onValueChange={setFilterDay}>
   968→            <SelectTrigger className="bg-white/[0.03] border-white/[0.08] text-slate-200">
   969→              <SelectValue />
   970→            </SelectTrigger>
   971→            <SelectContent>
   972→              <SelectItem value="all">جميع الأيام</SelectItem>
   973→              {DAYS.map((d) => (
   974→                <SelectItem key={d.value} value={d.value}>
   975→                  {d.label}
   976→                </SelectItem>
   977→              ))}
   978→            </SelectContent>
   979→          </Select>
   980→        </div>
   981→        <div className="flex flex-col gap-2">
   982→          <Label className="text-slate-300 text-sm">تصفية حسب المادة</Label>
   983→          <Select value={filterSubject} onValueChange={setFilterSubject}>
   984→            <SelectTrigger className="bg-white/[0.03] border-white/[0.08] text-slate-200">
   985→              <SelectValue />
   986→            </SelectTrigger>
   987→            <SelectContent>
   988→              <SelectItem value="all">جميع المواد</SelectItem>
   989→              {uniqueSubjects.map((s) => (
   990→                <SelectItem key={s} value={s}>
   991→                  {s}
   992→                </SelectItem>
   993→              ))}
   994→            </SelectContent>
   995→          </Select>
   996→        </div>
   997→      </div>
   998→
   999→      {/* Grouped list */}
  1000→      <div className="flex flex-col gap-4 max-h-[600px] overflow-y-auto custom-scrollbar">
  1001→        {Object.entries(groupedByDay).map(([dayLabel, dayBlocks]) => (
  1002→          <div key={dayLabel}>
