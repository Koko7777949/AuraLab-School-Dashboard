   410→function StudentFormContent({
   411→  editingStudent,
   412→  onClose,
   413→}: {
   414→  editingStudent: Student | null;
   415→  onClose: () => void;
   416→}) {
   417→  const { addStudent, updateStudent } = useAppStore();
   418→  const isEditing = !!editingStudent;
   419→
   420→  // Form state – initialised from props on mount
   421→  const [name, setName] = useState(editingStudent?.name ?? '');
   422→  const [studentId, setStudentId] = useState(editingStudent?.studentId ?? '');
   423→  const [classId, setClassId] = useState<string>(editingStudent?.classId ?? 'class-1');
   424→  const [status, setStatus] = useState<Student['status']>(editingStudent?.status ?? 'active');
   425→  const [notes, setNotes] = useState(editingStudent?.notes ?? '');
   426→  const [psychBackground, setPsychBackground] = useState(editingStudent?.psychBackground ?? '');
   427→
   428→  const handleSubmit = (e: React.FormEvent) => {
   429→    e.preventDefault();
   430→
   431→    if (!name.trim() || !studentId.trim()) return;
   432→
   433→    const now = new Date().toISOString();
   434→
   435→    if (isEditing && editingStudent) {
   436→      updateStudent(editingStudent.id, {
   437→        name: name.trim(),
   438→        studentId: studentId.trim(),
   439→        classId,
   440→        status,
   441→        notes: notes.trim(),
   442→        psychBackground: psychBackground.trim(),
   443→        updatedAt: now,
   444→      });
   445→      toast.success('تم تحديث بيانات الطالب');
   446→    } else {
   447→      const newStudent: Student = {
   448→        id: `std-${Date.now()}`,
   449→        name: name.trim(),
   450→        studentId: studentId.trim(),
   451→        classId,
   452→        status,
   453→        notes: notes.trim(),
   454→        psychBackground: psychBackground.trim(),
   455→        avatar: '',
   456→        createdAt: now,
   457→        updatedAt: now,
   458→      };
   459→      addStudent(newStudent);
   460→      toast.success('تم إضافة الطالب بنجاح');
   461→    }
   462→
   463→    onClose();
   464→  };
   465→
   466→  return (
   467→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   468→      {/* Name */}
   469→      <div className="flex flex-col gap-2">
   470→        <Label className="text-slate-300 text-sm">اسم الطالب الكامل</Label>
   471→        <Input
   472→          value={name}
   473→          onChange={(e) => setName(e.target.value)}
   474→          placeholder="مثال: أحمد محمد علي"
   475→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   476→          required
   477→        />
   478→      </div>
   479→
   480→      {/* Student ID */}
   481→      <div className="flex flex-col gap-2">
   482→        <Label className="text-slate-300 text-sm">الرقم التعريفي</Label>
   483→        <Input
   484→          value={studentId}
   485→          onChange={(e) => setStudentId(e.target.value)}
   486→          placeholder="مثال: STU-2413"
   487→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   488→          dir="ltr"
   489→          required
   490→        />
   491→      </div>
   492→
   493→      {/* Class & Status row */}
   494→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   495→        {/* Class dropdown */}
   496→        <div className="flex flex-col gap-2">
   497→          <Label className="text-slate-300 text-sm">الفصل</Label>
   498→          <Select value={classId} onValueChange={setClassId}>
   499→            <SelectTrigger className="w-full bg-white/[0.03] border-white/[0.08] text-slate-200 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20">
   500→              <SelectValue placeholder="اختر الفصل" />
   501→            </SelectTrigger>
   502→            <SelectContent className="bg-[#1A1A24] border-white/[0.08]">
   503→              {CLASS_OPTIONS.map((cls) => (
   504→                <SelectItem key={cls.value} value={cls.value} className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   505→                  {cls.label}
   506→                </SelectItem>
   507→              ))}
   508→            </SelectContent>
   509→          </Select>
   510→        </div>
   511→
   512→        {/* Status dropdown */}
   513→        <div className="flex flex-col gap-2">
   514→          <Label className="text-slate-300 text-sm">الحالة</Label>
   515→          <Select
   516→            value={status}
   517→            onValueChange={(val) => setStatus(val as Student['status'])}
