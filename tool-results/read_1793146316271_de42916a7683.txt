   109→function getScoreColor(score: number | null, maxScore: number): { bar: string; text: string; bg: string } {
   110→  if (score === null) return { bar: 'bg-slate-500', text: 'text-slate-400', bg: '' };
   111→  const pct = (score / maxScore) * 100;
   112→  if (pct >= 70) return { bar: 'bg-emerald-500', text: 'text-emerald-400', bg: 'bg-emerald-500/10' };
   113→  if (pct >= 50) return { bar: 'bg-amber-500', text: 'text-amber-400', bg: 'bg-amber-500/10' };
   114→  return { bar: 'bg-rose-500', text: 'text-rose-400', bg: 'bg-rose-500/10' };
   115→}
   116→
   117→function getScoreLabel(score: number | null, maxScore: number): string {
   118→  if (score === null) return 'لم يُحدد';
   119→  const pct = (score / maxScore) * 100;
   120→  if (pct >= 90) return 'ممتاز';
   121→  if (pct >= 70) return 'جيد جداً';
   122→  if (pct >= 50) return 'مقبول';
   123→  return 'يحتاج تحسين';
   124→}

   130→interface SubjectAvg {
   131→  subject: string;
   132→  avg: number;

   137→interface AnalysisStats {
   138→  highest: { pct: number; subject: string };
   139→  lowest: { pct: number; subject: string };
   140→  overallAvg: number;
   141→  passRate: number;
   142→}
   143→
   144→function getBarColorHex(avg: number): string {
   145→  if (avg >= 85) return '#10b981';
   146→  if (avg >= 70) return '#f59e0b';
   147→  if (avg >= 50) return '#0ea5e9';
   148→  return '#f43f5e';
   149→}
   150→
   151→function computeSubjectAverages(exams: Exam[]): SubjectAvg[] {
   152→  const groups: Record<string, { total: number; count: number }> = {};
   153→
   154→  for (const exam of exams) {
   155→    if (exam.score === null) continue;
   156→    const pct = (exam.score / Math.max(0.001, exam.maxScore)) * 100;
   157→    if (!groups[exam.subject]) {
   158→      groups[exam.subject] = { total: 0, count: 0 };
   159→    }
   160→    groups[exam.subject] = {
   161→      total: groups[exam.subject].total + pct,
   162→      count: groups[exam.subject].count + 1,
   163→    };
   164→  }
   165→
   166→  return Object.entries(groups)
   167→    .map(([subject, data]) => ({
   168→      subject,
   169→      avg: Math.round((data.total / Math.max(0.001, data.count)) * 10) / 10,
   170→      count: data.count,
   171→      color: getBarColorHex(data.total / Math.max(0.001, data.count)),
   172→    }))
   173→    .sort((a, b) => b.avg - a.avg);
   174→}
   175→
   176→function computeAnalysisStats(exams: Exam[]): AnalysisStats {
   177→  const scoredExams = exams.filter((e) => e.score !== null);
   178→
   179→  if (scoredExams.length === 0) {
   180→    return {
   181→      highest: { pct: 0, subject: '—' },
   182→      lowest: { pct: 0, subject: '—' },
   183→      overallAvg: 0,
   184→      passRate: 0,
   185→    };
   186→  }
   187→
   188→  const result = scoredExams.reduce(
   189→    (acc, exam) => {
   190→      const pct = (exam.score! / Math.max(0.001, exam.maxScore)) * 100;
   191→      return {
   192→        totalPct: acc.totalPct + pct,
   193→        passCount: acc.passCount + (pct >= 50 ? 1 : 0),
   194→        highestPct: Math.max(acc.highestPct, pct),
   195→        highestSubject: pct > acc.highestPct ? exam.subject : acc.highestSubject,
   196→        lowestPct: Math.min(acc.lowestPct, pct),
   197→        lowestSubject: pct < acc.lowestPct ? exam.subject : acc.lowestSubject,
   198→      };
   199→    },
   200→    { totalPct: 0, passCount: 0, highestPct: -1, highestSubject: '', lowestPct: 101, lowestSubject: '' },
   201→  );
   202→
   203→  const count = Math.max(0.001, scoredExams.length);
   204→  return {
   205→    highest: { pct: Math.round(result.highestPct * 10) / 10, subject: result.highestSubject },
   206→    lowest: { pct: Math.round(result.lowestPct * 10) / 10, subject: result.lowestSubject },
   207→    overallAvg: Math.round((result.totalPct / count) * 10) / 10,
   208→    passRate: Math.round(((result.passCount / count) * 100) * 10) / 10,
   209→  };
   210→}
