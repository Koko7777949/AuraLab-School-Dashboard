   154→function getScoreColor(pct: number): { bar: string; text: string; bg: string; badge: string } {
   155→  if (pct >= 90) return {
   156→    bar: 'bg-emerald-500',
   157→    text: 'text-emerald-400',
   158→    bg: 'bg-emerald-500/10',
   159→    badge: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/20',
   160→  };
   161→  if (pct >= 70) return {
   162→    bar: 'bg-amber-500',
   163→    text: 'text-amber-400',
   164→    bg: 'bg-amber-500/10',
   165→    badge: 'bg-amber-500/15 text-amber-400 border-amber-500/20',
   166→  };
   167→  return {
   168→    bar: 'bg-rose-500',
   169→    text: 'text-rose-400',
   170→    bg: 'bg-rose-500/10',
   171→    badge: 'bg-rose-500/15 text-rose-400 border-rose-500/20',
   172→  };
   173→}
   174→
   175→function getGpaColor(gpa: number): string {
   176→  if (gpa >= 3.5) return 'text-emerald-400';
   177→  if (gpa >= 2.5) return 'text-amber-400';
   178→  return 'text-rose-400';
   179→}
   180→
   181→function getGpaLabel(gpa: number): string {
   182→  if (gpa >= 3.5) return 'ممتاز';
   183→  if (gpa >= 3.0) return 'جيد جداً';
   184→  if (gpa >= 2.5) return 'جيد';
   185→  if (gpa >= 2.0) return 'مقبول';
   186→  return 'يحتاج تحسين';
   187→}
   188→
   189→function calculateGpa(score: number, maxScore: number): number {
   190→  const pct = (score / maxScore) * 100;
   191→  if (pct >= 90) return 4.0;
   192→  if (pct >= 85) return 3.75;
   193→  if (pct >= 80) return 3.5;
   194→  if (pct >= 75) return 3.25;
   195→  if (pct >= 70) return 3.0;
   196→  if (pct >= 65) return 2.75;
   197→  if (pct >= 60) return 2.5;
   198→  if (pct >= 55) return 2.25;
   199→  if (pct >= 50) return 2.0;
   200→  return 1.0;
   201→}

   738→
   739→function computeClassAverages(grades: GradeRecord[]): { className: string; average: number; count: number }[] {
   740→  const classMap = new Map<string, { totalWeighted: number; totalWeight: number; count: number }>();
   741→  for (const grade of grades) {
   742→    const className = grade.className ?? grade.classId ?? 'غير محدد';
   743→    const existing = classMap.get(className) ?? { totalWeighted: 0, totalWeight: 0, count: 0 };
   744→    const eps = 0.001;
   745→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   746→    classMap.set(className, {
   747→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
   748→      totalWeight: existing.totalWeight + grade.weight,
   749→      count: existing.count + 1,
   750→    });
   751→  }
   752→  const result: { className: string; average: number; count: number }[] = [];
   753→  classMap.forEach((val, key) => {
   754→    result.push({
   755→      className: key,
   756→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
   757→      count: val.count,
   758→    });
   759→  });
   760→  return result.sort((a, b) => b.average - a.average);
   761→}

   764→// Pure helper: compute grade distribution counts
   765→// ---------------------------------------------------------------------------

   767→function computeGradeDistribution(grades: GradeRecord[]): { excellent: number; veryGood: number; good: number; weak: number } {
   768→  const eps = 0.001;
   769→  let excellent = 0;
   770→  let veryGood = 0;
   771→  let good = 0;
   772→  let weak = 0;
   773→  for (const grade of grades) {
   774→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   775→    if (pct >= 90) excellent++;
   776→    else if (pct >= 75) veryGood++;
   777→    else if (pct >= 60) good++;
   778→    else weak++;
   779→  }
   780→  return { excellent, veryGood, good, weak };
   781→}

   794→function ClassPerformanceChart({ classData }: { classData: { className: string; average: number; count: number }[] }) {
   795→  const [animated, setAnimated] = useState(false);
   796→  const [hoveredIndex, setHoveredIndex] = useState<number | null>(null);
   797→
   798→  useEffect(() => {
   799→    const timer = requestAnimationFrame(() => setAnimated(true));
   800→    return () => cancelAnimationFrame(timer);
   801→  }, []);
   802→
   803→  if (classData.length === 0) {
   804→    return (
   805→      <div className="flex flex-col items-center justify-center py-10">

   808→      </div>
   809→    );
   810→  }
   811→
   812→  const maxAvg = Math.max(...classData.map((c) => c.average), 1);
   813→  const eps = 0.001;
   814→
   815→  return (
   816→    <div className="flex flex-col gap-4">
   817→      {/* Chart */}
   818→      <div className="flex flex-col gap-3">
   819→        {classData.map((item, idx) => {
   820→          const barWidth = Math.max(eps, (item.average / maxAvg) * 100);
   821→          const color = getBarColor(item.average);
   822→          const isHovered = hoveredIndex === idx;
   823→
   824→          return (
   825→            <div
   826→              key={item.className}
   827→              className="group relative flex items-center gap-3 cursor-default"
   828→              onMouseEnter={() => setHoveredIndex(idx)}
   829→              onMouseLeave={() => setHoveredIndex(null)}
   830→            >
   831→              {/* Class name label */}
   832→              <div className="w-36 sm:w-44 shrink-0 text-left">
   833→                <span className={cn(
   834→                  'text-sm font-medium transition-colors duration-200',
   835→                  isHovered ? 'text-[#E8E8ED]' : 'text-[#C4C4D4]'
   836→                )}>

   961→// Pure helper: compute subject averages for report card
   962→// ---------------------------------------------------------------------------
   963→
   964→function computeSubjectAverages(grades: GradeRecord[]): { subject: string; average: number }[] {
   965→  const subjectMap = new Map<string, { totalWeighted: number; totalWeight: number }>();
   966→  for (const grade of grades) {
   967→    const existing = subjectMap.get(grade.subject) ?? { totalWeighted: 0, totalWeight: 0 };
   968→    const pct = (grade.score / grade.maxScore) * 100;
   969→    subjectMap.set(grade.subject, {
   970→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
   971→      totalWeight: existing.totalWeight + grade.weight,
   972→    });
   973→  }
   974→  const result: { subject: string; average: number }[] = [];
   975→  subjectMap.forEach((val, key) => {
   976→    result.push({
   977→      subject: key,
   978→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
   979→    });
   980→  });
   981→  return result.sort((a, b) => b.average - a.average);
   982→}
