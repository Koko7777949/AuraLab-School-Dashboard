   923→function ListView({
   924→  blocks,
   925→  onBlockClick,
   926→  filterDay,
   927→  setFilterDay,
   928→  filterSubject,
   929→  setFilterSubject,
   930→}: {
   931→  blocks: ScheduleBlock[];
   932→  onBlockClick: (block: ScheduleBlock) => void;
   933→  filterDay: string;
   934→  setFilterDay: (d: string) => void;
   935→  filterSubject: string;
   936→  setFilterSubject: (s: string) => void;
   937→}) {
   938→  const uniqueSubjects = useMemo(() => {
   939→    return Array.from(new Set(blocks.map((b) => b.subject))).sort();
   940→  }, [blocks]);
   941→
   942→  const filteredBlocks = useMemo(() => {
   943→    let result = [...blocks];
   944→    if (filterDay !== 'all') {
   945→      result = result.filter((b) => b.dayOfWeek === filterDay);
   946→    }
   947→    if (filterSubject !== 'all') {
   948→      result = result.filter((b) => b.subject === filterSubject);
   949→    }
   950→    // Sort by day then period
   951→    const dayOrder: Record<string, number> = { sat: 0, sun: 1, mon: 2, tue: 3, wed: 4, thu: 5 };
   952→    result.sort((a, b) => {
   953→      const dayDiff = (dayOrder[a.dayOfWeek] ?? 0) - (dayOrder[b.dayOfWeek] ?? 0);
   954→      if (dayDiff !== 0) return dayDiff;
   955→      return a.period - b.period;
   956→    });
   957→    return result;
   958→  }, [blocks, filterDay, filterSubject]);
   959→
   960→  // Group by day
   961→  const groupedByDay = useMemo(() => {
   962→    const groups: Record<string, ScheduleBlock[]> = {};
   963→    for (const b of filteredBlocks) {
   964→      const dayLabel = getDayLabel(b.dayOfWeek);
   965→      if (!groups[dayLabel]) groups[dayLabel] = [];
   966→      groups[dayLabel].push(b);
   967→    }
   968→    return groups;
   969→  }, [filteredBlocks]);
   970→
   971→  return (
   972→    <div className="flex flex-col gap-4">
   973→      {/* Filters */}
   974→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   975→        <div className="flex flex-col gap-2">
   976→          <Label className="text-slate-300 text-sm">تصفية حسب اليوم</Label>
   977→          <Select value={filterDay} onValueChange={setFilterDay}>
   978→            <SelectTrigger className="bg-white/[0.03] border-white/[0.08] text-slate-200">
   979→              <SelectValue />
   980→            </SelectTrigger>
   981→            <SelectContent>
   982→              <SelectItem value="all">جميع الأيام</SelectItem>
   983→              {DAYS.map((d) => (
   984→                <SelectItem key={d.value} value={d.value}>
   985→                  {d.label}
   986→                </SelectItem>
   987→              ))}
   988→            </SelectContent>
   989→          </Select>
   990→        </div>
   991→        <div className="flex flex-col gap-2">
   992→          <Label className="text-slate-300 text-sm">تصفية حسب المادة</Label>
   993→          <Select value={filterSubject} onValueChange={setFilterSubject}>
   994→            <SelectTrigger className="bg-white/[0.03] border-white/[0.08] text-slate-200">
   995→              <SelectValue />
   996→            </SelectTrigger>
   997→            <SelectContent>
   998→              <SelectItem value="all">جميع المواد</SelectItem>
   999→              {uniqueSubjects.map((s) => (
  1000→                <SelectItem key={s} value={s}>
  1001→                  {s}
  1002→                </SelectItem>
  1003→              ))}
  1004→            </SelectContent>
  1005→          </Select>
  1006→        </div>
  1007→      </div>
  1008→
  1009→      {/* Grouped list */}
  1010→      <div className="flex flex-col gap-4 max-h-[600px] overflow-y-auto custom-scrollbar">
  1011→        {Object.entries(groupedByDay).map(([dayLabel, dayBlocks]) => (
  1012→          <div key={dayLabel}>
  1013→            <div className="flex items-center gap-2 mb-2">
  1014→              <CalendarRange className="h-4 w-4 text-[#D4A574]" />
  1015→              <h3 className="text-sm font-semibold text-[#D4A574]">{dayLabel}</h3>
  1016→              <Badge variant="outline" className="text-[10px] border-[#D4A574]/20 text-[#D4A574]">
  1017→                {dayBlocks.length} حصة
  1018→              </Badge>
  1019→            </div>
  1020→            <div className="flex flex-col gap-2">
  1021→              {dayBlocks.map((block) => (
