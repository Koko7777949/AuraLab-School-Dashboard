   860→/**
   861→ * Translate a key to the current locale string.
   862→ * Falls back to Arabic if key not found in the target locale.
   863→ * Falls back to the key itself if not found anywhere.
   864→ */
   865→export function t(key: string, locale: Locale = 'ar'): string {
   866→  return translations[locale]?.[key] ?? translations.ar[key] ?? key;
   867→}
   868→
   869→// ---------------------------------------------------------------------------
   870→// Convenience hook-friendly getter
   871→// ---------------------------------------------------------------------------
   872→
   873→/** Get all translations for a locale (for bulk use) */
   874→export function getTranslations(locale: Locale): Record<string, string> {
   875→  return translations[locale];
   876→}
