   193→function formatDate(dateStr: string): string {
   194→  return new Date(dateStr).toLocaleDateString('ar-SA', {
   195→    year: 'numeric',
   196→    month: 'short',
   197→    day: 'numeric',
   198→  });
   199→}
   200→
   201→function formatTime(dateStr: string): string {
   202→  return new Date(dateStr).toLocaleTimeString('ar-SA', {
   203→    hour: '2-digit',
   204→    minute: '2-digit',
   205→  });
   206→}
   207→
   208→function getDateGroup(dateStr: string): string {
   209→  const date = new Date(dateStr);
   210→  const today = new Date();
   211→  const todayStart = new Date(today.getFullYear(), today.getMonth(), today.getDate());
   212→  const yesterdayStart = new Date(todayStart);
   213→  yesterdayStart.setDate(yesterdayStart.getDate() - 1);
   214→  const weekStart = new Date(todayStart);
   215→  weekStart.setDate(weekStart.getDate() - 7);
   216→
   217→  if (date >= todayStart) return 'اليوم';
   218→  if (date >= yesterdayStart) return 'أمس';
   219→  if (date >= weekStart) return 'هذا الأسبوع';
   220→  return 'أقدم';
   221→}

   231→function exportToCSV(records: BehaviorRecord[]) {
   232→  const headers = ['اسم الطالب', 'الفصل', 'التاريخ', 'النوع', 'الفئة', 'الشدة', 'الوصف', 'الإجراء', 'النقاط', 'بواسطة'];
   233→  const rows = records.map((r) => [
   234→    r.studentName || '',
   235→    r.className || '',
   236→    formatDate(r.date),
   237→    r.type === 'positive' ? 'إيجابي' : 'سلبي',
   238→    r.category,
   239→    SEVERITY_LABELS[r.severity],
   240→    r.description,
   241→    r.action,
   242→    r.points.toString(),
   243→    r.recordedBy,
   244→  ]);
   245→
   246→  const BOM = '\uFEFF';
   247→  const csv = BOM + [headers, ...rows].map((row) => row.map((cell) => `"${cell}"`).join(',')).join('\n');
   248→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   249→  const url = URL.createObjectURL(blob);
   250→  const link = document.createElement('a');
   251→  link.href = url;
   252→  link.download = `behavior_records_${new Date().toISOString().split('T')[0]}.csv`;
   253→  link.click();
   254→  URL.revokeObjectURL(url);
   255→  toast.success('تم تصدير البيانات بنجاح');
   256→}

   886→function StudentBehaviorSummary({ records }: { records: BehaviorRecord[] }) {
   887→  const positiveByStudent = useMemo(() => {
   888→    const map = new Map<string, { name: string; total: number; count: number }>();
   889→    for (const r of records.filter((r) => r.type === 'positive')) {
   890→      const key = r.studentId;
   891→      if (!map.has(key)) {
   892→        map.set(key, { name: r.studentName || 'غير معروف', total: 0, count: 0 });
   893→      }
   894→      const entry = map.get(key)!;
   895→      entry.total += r.points;
   896→      entry.count += 1;
   897→    }
   898→    return Array.from(map.values())
   899→      .sort((a, b) => b.total - a.total)
   900→      .slice(0, 5);
   901→  }, [records]);
   902→
   903→  const negativeByStudent = useMemo(() => {
   904→    const map = new Map<string, { name: string; total: number; count: number }>();
   905→    for (const r of records.filter((r) => r.type === 'negative')) {
   906→      const key = r.studentId;
   907→      if (!map.has(key)) {
   908→        map.set(key, { name: r.studentName || 'غير معروف', total: 0, count: 0 });
   909→      }
   910→      const entry = map.get(key)!;
   911→      entry.total += Math.abs(r.points);
   912→      entry.count += 1;
   913→    }
   914→    return Array.from(map.values())
   915→      .sort((a, b) => b.total - a.total)
   916→      .slice(0, 5);
   917→  }, [records]);
   918→
   919→  const maxPositive = positiveByStudent.length > 0 ? positiveByStudent[0].total : 1;
   920→  const maxNegative = negativeByStudent.length > 0 ? negativeByStudent[0].total : 1;
   921→
   922→  if (positiveByStudent.length === 0 && negativeByStudent.length === 0) return null;
   923→
   924→  return (
   925→    <div
   926→      className="rounded-xl p-5"
   927→      style={{
   928→        backgroundColor: 'rgba(255,255,255,0.02)',
   929→        border: '1px solid rgba(255,255,255,0.06)',
   930→      }}
   931→    >
   932→      <h3 className="text-sm font-semibold text-slate-300 mb-4 flex items-center gap-2">
   933→        <Star className="h-4 w-4 text-[#D4A574]" />
   934→        ملخص سلوك الطلاب
   935→      </h3>
   936→
   937→      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
   938→        {/* Top Positive */}
   939→        {positiveByStudent.length > 0 && (
   940→          <div>
   941→            <h4 className="text-xs font-medium text-emerald-400 mb-3 flex items-center gap-1.5">
   942→              <TrendingUp className="h-3.5 w-3.5" />
   943→              الأكثر إيجابية
   944→            </h4>
   945→            <div className="flex flex-col gap-2">
   946→              {positiveByStudent.map((s, i) => (
   947→                <div key={s.name} className="flex items-center gap-3">
   948→                  <span className="text-xs text-muted-foreground w-4 tabular-nums">{i + 1}.</span>
   949→                  <div className="flex-1 min-w-0">
   950→                    <div className="flex items-center justify-between mb-1">
   951→                      <span className="text-xs text-slate-300 truncate">{s.name}</span>
   952→                      <span className="text-xs font-bold text-emerald-400 tabular-nums">+{s.total}</span>
   953→                    </div>
   954→                    <div className="h-1.5 w-full rounded-full bg-white/[0.04] overflow-hidden">
   955→                      <div
   956→                        className="h-full rounded-full bg-emerald-500/50 transition-all duration-500"
   957→                        style={{ width: `${Math.max(8, (s.total / maxPositive) * 100)}%` }}
   958→                      />
   959→                    </div>
   960→                  </div>
   961→                </div>
   962→              ))}
   963→            </div>
   964→          </div>
   965→        )}
   966→
   967→        {/* Top Negative */}
   968→        {negativeByStudent.length > 0 && (
   969→          <div>
   970→            <h4 className="text-xs font-medium text-rose-400 mb-3 flex items-center gap-1.5">
   971→              <TrendingDown className="h-3.5 w-3.5" />
   972→              الأكثر سلبية
   973→            </h4>
   974→            <div className="flex flex-col gap-2">
   975→              {negativeByStudent.map((s, i) => (
   976→                <div key={s.name} className="flex items-center gap-3">
   977→                  <span className="text-xs text-muted-foreground w-4 tabular-nums">{i + 1}.</span>
   978→                  <div className="flex-1 min-w-0">
   979→                    <div className="flex items-center justify-between mb-1">

  1171→function Pagination({
  1172→  currentPage,
  1173→  totalPages,
  1174→  onPageChange,
  1175→}: {
  1176→  currentPage: number;
  1177→  totalPages: number;
  1178→  onPageChange: (page: number) => void;
  1179→}) {
  1180→  if (totalPages <= 1) return null;
  1181→
  1182→  const getPages = () => {
  1183→    const pages: (number | '...')[] = [];
  1184→    if (totalPages <= 7) {
  1185→      for (let i = 1; i <= totalPages; i++) pages.push(i);
  1186→    } else {
  1187→      pages.push(1);
  1188→      if (currentPage > 3) pages.push('...');
  1189→      for (let i = Math.max(2, currentPage - 1); i <= Math.min(totalPages - 1, currentPage + 1); i++) {
  1190→        pages.push(i);
  1191→      }
  1192→      if (currentPage < totalPages - 2) pages.push('...');
  1193→      pages.push(totalPages);
  1194→    }
  1195→    return pages;
  1196→  };
  1197→
  1198→  return (
  1199→    <div className="flex items-center justify-center gap-1.5 mt-6">
  1200→      <button
  1201→        onClick={() => onPageChange(currentPage - 1)}
  1202→        disabled={currentPage === 1}
  1203→        className="flex items-center justify-center h-9 w-9 rounded-lg border border-white/[0.06] text-slate-400 hover:text-white hover:bg-white/[0.05] disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
  1204→      >
  1205→        <ChevronRight className="h-4 w-4" />
  1206→      </button>
  1207→      {getPages().map((page, i) =>
  1208→        page === '...' ? (
  1209→          <span key={`dots-${i}`} className="h-9 w-9 flex items-center justify-center text-sm text-muted-foreground">
  1210→            ...
  1211→          </span>
  1212→        ) : (
  1213→          <button
  1214→            key={page}
  1215→            onClick={() => onPageChange(page as number)}
  1216→            className={cn(
  1217→              'flex items-center justify-center h-9 w-9 rounded-lg text-sm font-medium transition-all duration-200',
  1218→              currentPage === page
  1219→                ? 'bg-[#D4A574] text-slate-900 font-bold'
  1220→                : 'border border-white/[0.06] text-slate-400 hover:text-white hover:bg-white/[0.05]'
  1221→            )}
  1222→          >
  1223→            {page}
  1224→          </button>
  1225→        )
  1226→      )}
  1227→      <button
  1228→        onClick={() => onPageChange(currentPage + 1)}
  1229→        disabled={currentPage === totalPages}
  1230→        className="flex items-center justify-center h-9 w-9 rounded-lg border border-white/[0.06] text-slate-400 hover:text-white hover:bg-white/[0.05] disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
  1231→      >
  1232→        <ChevronLeft className="h-4 w-4" />
  1233→      </button>
  1234→    </div>
  1235→  );
  1236→}
