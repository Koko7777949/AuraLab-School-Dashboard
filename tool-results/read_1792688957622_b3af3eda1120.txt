   274→
   275→function getClassName(classId: string | null, classes: { id: string; name: string }[]): string {
   276→  if (!classId) return '—';
   277→  const found = classes.find((c) => c.id === classId);
   278→  if (found) return found.name;
   279→  const legacy = CLASS_OPTIONS.find((c) => c.value === classId);
   280→  return legacy ? legacy.label : '—';
   281→}

   429→function StudentFormContent({
   430→  editingStudent,
   431→  onClose,
   432→}: {
   433→  editingStudent: Student | null;
   434→  onClose: () => void;
   435→}) {
   436→  const { addStudent, updateStudent } = useAppStore();
   437→  const isEditing = !!editingStudent;
   438→
   439→  // Form state – initialised from props on mount
   440→  const [name, setName] = useState(editingStudent?.name ?? '');
   441→  const [studentId, setStudentId] = useState(editingStudent?.studentId ?? '');
   442→  const [classId, setClassId] = useState<string>(editingStudent?.classId ?? 'cls-1');
   443→  const [status, setStatus] = useState<Student['status']>(editingStudent?.status ?? 'active');
   444→  const [notes, setNotes] = useState(editingStudent?.notes ?? '');
   445→  const [psychBackground, setPsychBackground] = useState(editingStudent?.psychBackground ?? '');
   446→
   447→  const handleSubmit = (e: React.FormEvent) => {
   448→    e.preventDefault();
   449→
   450→    if (!name.trim() || !studentId.trim()) return;
   451→
   452→    const now = new Date().toISOString();
   453→
   454→    if (isEditing && editingStudent) {
   455→      updateStudent(editingStudent.id, {
   456→        name: name.trim(),
   457→        studentId: studentId.trim(),
   458→        classId,
   459→        status,
   460→        notes: notes.trim(),
   461→        psychBackground: psychBackground.trim(),
   462→        updatedAt: now,
   463→      });
   464→      toast.success('تم تحديث بيانات الطالب');
   465→    } else {
   466→      const newStudent: Student = {
   467→        id: `std-${Date.now()}`,
   468→        name: name.trim(),
   469→        studentId: studentId.trim(),
   470→        classId,
   471→        status,
   472→        notes: notes.trim(),
   473→        psychBackground: psychBackground.trim(),
   474→        avatar: '',
   475→        createdAt: now,
   476→        updatedAt: now,
   477→      };
   478→      addStudent(newStudent);
   479→      toast.success('تم إضافة الطالب بنجاح');
   480→    }
   481→
   482→    onClose();
   483→  };
   484→
   485→  return (
   486→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   487→      {/* Name */}
   488→      <div className="flex flex-col gap-2">
   489→        <Label className="text-slate-300 text-sm">اسم الطالب الكامل</Label>
   490→        <Input
   491→          value={name}
   492→          onChange={(e) => setName(e.target.value)}
   493→          placeholder="مثال: أحمد محمد علي"
   494→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   495→          required
   496→        />
   497→      </div>
   498→
   499→      {/* Student ID */}
   500→      <div className="flex flex-col gap-2">
   501→        <Label className="text-slate-300 text-sm">الرقم التعريفي</Label>
   502→        <Input
   503→          value={studentId}
   504→          onChange={(e) => setStudentId(e.target.value)}
   505→          placeholder="مثال: STU-2413"
   506→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   507→          dir="ltr"
   508→          required
   509→        />
   510→      </div>
   511→
   512→      {/* Class & Status row */}
   513→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   514→        {/* Class dropdown */}
   515→        <div className="flex flex-col gap-2">
   516→          <Label className="text-slate-300 text-sm">الفصل</Label>
   517→          <Select value={classId} onValueChange={setClassId}>
   518→            <SelectTrigger className="w-full bg-white/[0.03] border-white/[0.08] text-slate-200 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20">
   519→              <SelectValue placeholder="اختر الفصل" />
   520→            </SelectTrigger>
