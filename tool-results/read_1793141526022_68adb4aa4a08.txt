   144→function getCapacityColor(percentage: number): { bar: string; text: string } {
   145→  if (percentage >= 90) {
   146→    return { bar: 'bg-rose-500', text: 'text-rose-400' };
   147→  }
   148→  if (percentage >= 70) {
   149→    return { bar: 'bg-amber-500', text: 'text-amber-400' };
   150→  }
   151→  return { bar: 'bg-emerald-500', text: 'text-emerald-400' };
   152→}

   312→function ClassFormContent({
   313→  editingClass,
   314→  onClose,
   315→}: {
   316→  editingClass: ClassRoom | null;
   317→  onClose: () => void;
   318→}) {
   319→  const { addClass, updateClass } = useAppStore();
   320→  const isEditing = !!editingClass;
   321→
   322→  // Form state
   323→  const [name, setName] = useState(editingClass?.name ?? '');
   324→  const [grade, setGrade] = useState(editingClass?.grade ?? 'الأول');
   325→  const [room, setRoom] = useState(editingClass?.room ?? '');
   326→  const [notes, setNotes] = useState(editingClass?.notes ?? '');
   327→
   328→  const handleSubmit = (e: React.FormEvent) => {
   329→    e.preventDefault();
   330→    if (!name.trim() || !room.trim()) return;
   331→
   332→    const now = new Date().toISOString();
   333→
   334→    if (isEditing && editingClass) {
   335→      updateClass(editingClass.id, {
   336→        name: name.trim(),
   337→        grade,
   338→        room: room.trim(),
   339→        notes: notes.trim(),
   340→        updatedAt: now,
   341→      });
   342→      toast.success('تم تحديث بيانات الصف');
   343→    } else {
   344→      const newClass: ClassRoom = {
   345→        id: `cls-${Date.now()}`,
   346→        name: name.trim(),
   347→        grade,
   348→        room: room.trim(),
   349→        notes: notes.trim(),
   350→        createdAt: now,
   351→        updatedAt: now,
   352→        studentCount: 0,
   353→        teacherNames: [],
   354→      };
   355→      addClass(newClass);
   356→      toast.success('تم إضافة الصف بنجاح');
   357→    }
   358→
   359→    onClose();
   360→  };
   361→
   362→  return (
   363→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   364→      {/* Class name */}
   365→      <div className="flex flex-col gap-2">
   366→        <Label className="text-slate-300 text-sm">اسم الصف</Label>
   367→        <Input
   368→          value={name}
   369→          onChange={(e) => setName(e.target.value)}
   370→          placeholder="مثال: الصف الأول أ"
   371→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   372→          required
   373→        />
   374→      </div>
   375→
   376→      {/* Grade & Room row */}
   377→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   378→        {/* Grade dropdown */}
   379→        <div className="flex flex-col gap-2">
   380→          <Label className="text-slate-300 text-sm">المرحلة الدراسية</Label>
   381→          <Select value={grade} onValueChange={setGrade}>
   382→            <SelectTrigger className="w-full bg-white/[0.03] border-white/[0.08] text-slate-200 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20">
   383→              <SelectValue placeholder="اختر المرحلة" />
   384→            </SelectTrigger>
   385→            <SelectContent className="bg-[#1A1A24] border-white/[0.08]">
   386→              <SelectItem value="الأول" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   387→                الصف الأول
   388→              </SelectItem>
   389→              <SelectItem value="الثاني" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   390→                الصف الثاني
   391→              </SelectItem>
   392→              <SelectItem value="الثالث" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   393→                الصف الثالث
   394→              </SelectItem>
   395→            </SelectContent>
   396→          </Select>
   397→        </div>
   398→
   399→        {/* Room */}
   400→        <div className="flex flex-col gap-2">
   401→          <Label className="text-slate-300 text-sm">رقم القاعة</Label>
   402→          <Input
   403→            value={room}
