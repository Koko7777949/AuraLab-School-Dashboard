   279→   279→function formatTime(time: string | null): string {
   280→   280→  if (!time) return '—';

   283→   283→
   284→   284→function calcDuration(start: string | null, end: string | null): string {
   285→   285→  if (!start || !end) return '—';
   286→   286→  const [sh, sm] = start.split(':').map(Number);
   287→   287→  const [eh, em] = end.split(':').map(Number);
   288→   288→  const diff = (eh * 60 + em) - (sh * 60 + sm);
   289→   289→  if (diff <= 0) return '—';
   290→   290→  const h = Math.floor(diff / 60);
   291→   291→  const m = diff % 60;
   292→   292→  if (h === 0) return `${m} دقيقة`;
   293→   293→  if (m === 0) return `${h} ساعة`;
   294→   294→  return `${h} ساعة و ${m} دقيقة`;
   295→   295→}
   296→   296→
   297→   297→function getTodayLabel(dateStr: string): string {
   298→   298→  const today = new Date().toISOString().split('T')[0];
   299→   299→  const tomorrow = new Date(Date.now() + 86400000).toISOString().split('T')[0];
   300→   300→  const yesterday = new Date(Date.now() - 86400000).toISOString().split('T')[0];
   301→   301→  if (dateStr === today) return 'اليوم';
   302→   302→  if (dateStr === tomorrow) return 'غداً';
   303→   303→  if (dateStr === yesterday) return 'أمس';
   304→   304→  return formatDateArabic(dateStr);
   305→   305→}

   311→   311→/** Circular engagement indicator using SVG */
   312→   312→function EngagementRing({
   313→   313→  score,
   314→   314→  size = 52,
   315→   315→  strokeWidth = 4,
   316→   316→}: {
   317→   317→  score: number;
   318→   318→  size?: number;
   319→   319→  strokeWidth?: number;
   320→   320→}) {
   321→   321→  const radius = (size - strokeWidth) / 2;
   322→   322→  const circumference = radius * 2 * Math.PI;
   323→   323→  const color = score > 0 ? getEngagementColor(score) : 'rgba(255,255,255,0.1)';
   324→   324→
   325→   325→  return (
   326→   326→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   327→   327→      <svg width={size} height={size} className="-rotate-90">
   328→   328→        <circle
   329→   329→          cx={size / 2}
   330→   330→          cy={size / 2}
   331→   331→          r={radius}
   332→   332→          fill="none"
   333→   333→          stroke="rgba(255,255,255,0.06)"
   334→   334→          strokeWidth={strokeWidth}
   335→   335→        />
   336→   336→        {score > 0 && (
   337→   337→          <circle
   338→   338→            cx={size / 2}
   339→   339→            cy={size / 2}
   340→   340→            r={radius}
   341→   341→            fill="none"
   342→   342→            stroke={color}
   343→   343→            strokeWidth={strokeWidth}
   344→   344→            strokeLinecap="round"
   345→   345→            strokeDasharray={`${circumference} ${circumference}`}
   346→   346→            style={{
   347→   347→              strokeDashoffset: circumference - (score / 100) * circumference,
   348→   348→              transition: 'stroke-dashoffset 1s ease-out',
   349→   349→            }}
   350→   350→          />
   351→   351→        )}
   352→   352→      </svg>
   353→   353→      <span className="absolute text-xs font-bold text-white tabular-nums select-none">
   354→   354→        {score > 0 ? score : '—'}
   355→   355→      </span>
   356→   356→    </div>
   357→   357→  );
   358→   358→}
