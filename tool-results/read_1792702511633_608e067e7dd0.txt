   277→function getClassName(classId: string | null, classes: { id: string; name: string }[]): string {
   278→  if (!classId) return '—';
   279→  const found = classes.find((c) => c.id === classId);
   280→  if (found) return found.name;
   281→  const legacy = CLASS_OPTIONS.find((c) => c.value === classId);
   282→  return legacy ? legacy.label : '—';
   283→}
   284→
   285→/** Highlight matching text in amber color */
   286→function HighlightText({ text, query }: { text: string; query: string }) {
   287→  if (!query.trim()) return <>{text}</>;
   288→  const lowerText = text.toLowerCase();
   289→  const lowerQuery = query.trim().toLowerCase();
   290→  const index = lowerText.indexOf(lowerQuery);
   291→  if (index === -1) return <>{text}</>;
   292→  return (
   293→    <>
   294→      {text.slice(0, index)}
   295→      <span className="text-[#D4A574] font-bold">{text.slice(index, index + query.trim().length)}</span>
   296→      {text.slice(index + query.trim().length)}
   297→    </>
   298→  );
   299→}

   447→function StudentFormContent({
   448→  editingStudent,
   449→  onClose,
   450→}: {
   451→  editingStudent: Student | null;
   452→  onClose: () => void;
   453→}) {
   454→  const { addStudent, updateStudent } = useAppStore();
   455→  const isEditing = !!editingStudent;
   456→
   457→  // Form state – initialised from props on mount
   458→  const [name, setName] = useState(editingStudent?.name ?? '');
   459→  const [studentId, setStudentId] = useState(editingStudent?.studentId ?? '');
   460→  const [classId, setClassId] = useState<string>(editingStudent?.classId ?? 'cls-1');
   461→  const [status, setStatus] = useState<Student['status']>(editingStudent?.status ?? 'active');
   462→  const [notes, setNotes] = useState(editingStudent?.notes ?? '');
   463→  const [psychBackground, setPsychBackground] = useState(editingStudent?.psychBackground ?? '');
   464→
   465→  const handleSubmit = (e: React.FormEvent) => {
   466→    e.preventDefault();
   467→
   468→    if (!name.trim() || !studentId.trim()) return;
   469→
   470→    const now = new Date().toISOString();
   471→
   472→    if (isEditing && editingStudent) {
   473→      updateStudent(editingStudent.id, {
   474→        name: name.trim(),
   475→        studentId: studentId.trim(),
   476→        classId,
   477→        status,
   478→        notes: notes.trim(),
   479→        psychBackground: psychBackground.trim(),
   480→        updatedAt: now,
   481→      });
   482→      toast.success('تم تحديث بيانات الطالب');
   483→    } else {
   484→      const newStudent: Student = {
   485→        id: `std-${Date.now()}`,
   486→        name: name.trim(),
   487→        studentId: studentId.trim(),
   488→        classId,
   489→        status,
   490→        notes: notes.trim(),
   491→        psychBackground: psychBackground.trim(),
   492→        avatar: '',
   493→        createdAt: now,
   494→        updatedAt: now,
   495→      };
   496→      addStudent(newStudent);
   497→      toast.success('تم إضافة الطالب بنجاح');
   498→    }
   499→
   500→    onClose();
   501→  };
   502→
   503→  return (
   504→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   505→      {/* Name */}
   506→      <div className="flex flex-col gap-2">
   507→        <Label className="text-slate-300 text-sm">اسم الطالب الكامل</Label>
   508→        <Input
   509→          value={name}
   510→          onChange={(e) => setName(e.target.value)}
   511→          placeholder="مثال: أحمد محمد علي"
   512→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   513→          required
   514→        />
   515→      </div>
   516→
   517→      {/* Student ID */}
   518→      <div className="flex flex-col gap-2">
   519→        <Label className="text-slate-300 text-sm">الرقم التعريفي</Label>
   520→        <Input
   521→          value={studentId}
