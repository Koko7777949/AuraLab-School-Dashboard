   217→function getInitials(name: string): string {
   218→  const parts = name.trim().split(/\s+/);
   219→  if (parts.length >= 2) {
   220→    return parts[0].charAt(0) + parts[1].charAt(0);
   221→  }
   222→  return name.charAt(0);
   223→}

   421→function TeacherFormContent({
   422→  editingTeacher,
   423→  onClose,
   424→}: {
   425→  editingTeacher: Teacher | null;
   426→  onClose: () => void;
   427→}) {
   428→  const { addTeacher, updateTeacher } = useAppStore();
   429→  const isEditing = !!editingTeacher;
   430→
   431→  // Form state
   432→  const [name, setName] = useState(editingTeacher?.name ?? '');
   433→  const [subject, setSubject] = useState(editingTeacher?.subject ?? '');
   434→  const [email, setEmail] = useState(editingTeacher?.email ?? '');

   436→  const [permissions, setPermissions] = useState<Teacher['permissions']>(
   437→    editingTeacher?.permissions ?? 'teacher'
   438→  );
   439→  const [status, setStatus] = useState<Teacher['status']>(
   440→    editingTeacher?.status ?? 'active'
   441→  );
   442→  const [classNamesInput, setClassNamesInput] = useState(
   443→    editingTeacher?.classNames?.join('، ') ?? ''
   444→  );

   447→    e.preventDefault();
   448→    if (!name.trim() || !subject.trim()) return;
   449→
   450→    const now = new Date().toISOString();

   456→        email: email.trim(),
   457→        phone: phone.trim(),

   470→        permissions,
   471→        avatar: '',
   472→        email: email.trim(),
   473→        phone: phone.trim(),
   474→        createdAt: now,
   475→        updatedAt: now,
   476→        classNames: classNamesInput.trim() ? classNamesInput.split(/[,،]/).map((s) => s.trim()).filter(Boolean) : [],

   482→    onClose();

   492→          onChange={(e) => setName(e.target.value)}
   493→          placeholder="مثال: محمد بن عبدالله السالم"
   494→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   495→          required
   496→        />
   497→      </div>
   498→
   499→      {/* Subject */}

   501→        <Label className="text-slate-300 text-sm">المادة الدراسية</Label>

   503→          value={subject}
   504→          onChange={(e) => setSubject(e.target.value)}
   505→          placeholder="مثال: الرياضيات"
   506→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"

   508→        />
   509→      </div>

   513→        <div className="flex flex-col gap-2">
   514→          <Label className="text-slate-300 text-sm">البريد الإلكتروني</Label>

   521→            className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"

   658→/** Delete confirmation dialog */
   659→function DeleteTeacherDialog({
   660→  teacher,
   661→  open,
   662→  onOpenChange,
   663→}: {
   664→  teacher: Teacher | null;
   665→  open: boolean;
   666→  onOpenChange: (open: boolean) => void;
   667→}) {
   668→  const { deleteTeacher, addToTrash } = useAppStore();
   669→
   670→  const handleDelete = () => {
   671→    if (teacher) {
   672→      addToTrash({
   673→        id: `trash-${Date.now()}-${Math.random().toString(36).slice(2, 8)}`,
   674→        itemType: 'teacher',

   687→

   689→    <AlertDialog open={open} onOpenChange={onOpenChange}>

   700→          </AlertDialogTitle>
   701→          <AlertDialogDescription className="text-muted-foreground leading-relaxed">
   702→            هل أنت متأكد من حذف المعلم{' '}
   703→            <span className="font-semibold text-slate-200">{teacher?.name}</span>؟
   704→            <br />
   705→            لا يمكن التراجع عن هذا الإجراء. سيتم حذف جميع بيانات المعلم بشكل نهائي.

   707→        </AlertDialogHeader>
   708→        <AlertDialogFooter className="flex-row gap-3 sm:justify-start mt-2">

   710→            className="border-white/[0.08] text-slate-300 hover:bg-white/[0.05] hover:text-white rounded-lg px-6 mt-0"
