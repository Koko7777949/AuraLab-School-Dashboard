   425→function StudentFormContent({
   426→  editingStudent,
   427→  onClose,
   428→}: {
   429→  editingStudent: Student | null;
   430→  onClose: () => void;
   431→}) {
   432→  const { addStudent, updateStudent } = useAppStore();
   433→  const isEditing = !!editingStudent;
   434→
   435→  // Form state – initialised from props on mount
   436→  const [name, setName] = useState(editingStudent?.name ?? '');
   437→  const [studentId, setStudentId] = useState(editingStudent?.studentId ?? '');
   438→  const [classId, setClassId] = useState<string>(editingStudent?.classId ?? 'class-1');
   439→  const [status, setStatus] = useState<Student['status']>(editingStudent?.status ?? 'active');
   440→  const [notes, setNotes] = useState(editingStudent?.notes ?? '');
   441→  const [psychBackground, setPsychBackground] = useState(editingStudent?.psychBackground ?? '');
   442→
   443→  const handleSubmit = (e: React.FormEvent) => {
   444→    e.preventDefault();
   445→
   446→    if (!name.trim() || !studentId.trim()) return;
   447→
   448→    const now = new Date().toISOString();
   449→
   450→    if (isEditing && editingStudent) {
   451→      updateStudent(editingStudent.id, {
   452→        name: name.trim(),
   453→        studentId: studentId.trim(),
   454→        classId,
   455→        status,
   456→        notes: notes.trim(),
   457→        psychBackground: psychBackground.trim(),
   458→        updatedAt: now,
   459→      });
   460→      toast.success('تم تحديث بيانات الطالب');
   461→    } else {
   462→      const newStudent: Student = {
   463→        id: `std-${Date.now()}`,
   464→        name: name.trim(),
   465→        studentId: studentId.trim(),
   466→        classId,
   467→        status,
   468→        notes: notes.trim(),
   469→        psychBackground: psychBackground.trim(),
   470→        avatar: '',
   471→        createdAt: now,
   472→        updatedAt: now,
   473→      };
   474→      addStudent(newStudent);
   475→      toast.success('تم إضافة الطالب بنجاح');
   476→    }
   477→
   478→    onClose();
   479→  };
   480→
   481→  return (
   482→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   483→      {/* Name */}
   484→      <div className="flex flex-col gap-2">
   485→        <Label className="text-slate-300 text-sm">اسم الطالب الكامل</Label>
   486→        <Input
   487→          value={name}
   488→          onChange={(e) => setName(e.target.value)}
   489→          placeholder="مثال: أحمد محمد علي"
   490→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   491→          required
   492→        />
   493→      </div>
   494→
   495→      {/* Student ID */}
   496→      <div className="flex flex-col gap-2">
   497→        <Label className="text-slate-300 text-sm">الرقم التعريفي</Label>
   498→        <Input
   499→          value={studentId}
   500→          onChange={(e) => setStudentId(e.target.value)}
   501→          placeholder="مثال: STU-2413"
   502→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   503→          dir="ltr"
   504→          required
   505→        />
   506→      </div>
   507→
   508→      {/* Class & Status row */}
   509→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   510→        {/* Class dropdown */}
   511→        <div className="flex flex-col gap-2">
   512→          <Label className="text-slate-300 text-sm">الفصل</Label>
   513→          <Select value={classId} onValueChange={setClassId}>
   514→            <SelectTrigger className="w-full bg-white/[0.03] border-white/[0.08] text-slate-200 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20">
   515→              <SelectValue placeholder="اختر الفصل" />
   516→            </SelectTrigger>
   517→            <SelectContent className="bg-[#1A1A24] border-white/[0.08]">
   518→              {CLASS_OPTIONS.map((cls) => (
   519→                <SelectItem key={cls.value} value={cls.value} className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   520→                  {cls.label}
