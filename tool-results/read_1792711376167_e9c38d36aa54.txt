   155→function getScoreColor(pct: number): { bar: string; text: string; bg: string; badge: string } {
   156→  if (pct >= 90) return {

   160→    badge: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/20',
   161→  };
   162→  if (pct >= 70) return {
   163→    bar: 'bg-amber-500',
   164→    text: 'text-amber-400',
   165→    bg: 'bg-amber-500/10',
   166→    badge: 'bg-amber-500/15 text-amber-400 border-amber-500/20',
   167→  };
   168→  return {
   169→    bar: 'bg-rose-500',
   170→    text: 'text-rose-400',

   176→function getGpaColor(gpa: number): string {
   177→  if (gpa >= 3.5) return 'text-emerald-400';
   178→  if (gpa >= 2.5) return 'text-amber-400';
   179→  return 'text-rose-400';

   182→function getGpaLabel(gpa: number): string {
   183→  if (gpa >= 3.5) return 'ممتاز';
   184→  if (gpa >= 3.0) return 'جيد جداً';
   185→  if (gpa >= 2.5) return 'جيد';
   186→  if (gpa >= 2.0) return 'مقبول';
   187→  return 'يحتاج تحسين';
   188→}
   189→
   190→function calculateGpa(score: number, maxScore: number): number {
   191→  const pct = (score / maxScore) * 100;
   192→  if (pct >= 90) return 4.0;
   193→  if (pct >= 85) return 3.75;
   194→  if (pct >= 80) return 3.5;
   195→  if (pct >= 75) return 3.25;
   196→  if (pct >= 70) return 3.0;
   197→  if (pct >= 65) return 2.75;
   198→  if (pct >= 60) return 2.5;
   199→  if (pct >= 55) return 2.25;
   200→  if (pct >= 50) return 2.0;
   201→  return 1.0;
   202→}

   327→/** Inline editable score cell */
   328→function InlineScoreCell({ grade }: { grade: GradeRecord }) {
   329→  const { updateGrade } = useAppStore();
   330→  const [editing, setEditing] = useState(false);
   331→  const [value, setValue] = useState(String(grade.score));
   332→
   333→  const handleSave = () => {
   334→    const num = Number(value);
   335→    if (!isNaN(num) && num >= 0 && num <= grade.maxScore && num !== grade.score) {
   336→      updateGrade(grade.id, { score: num });
   337→      toast.success('تم تحديث الدرجة');
   338→    }
   339→    setEditing(false);
   340→  };
   341→
   342→  if (editing) {
   343→    return (
   344→      <div className="flex items-center gap-1">
   345→        <input
   346→          autoFocus
   347→          type="number"
   348→          min={0}
   349→          max={grade.maxScore}
   350→          value={value}
   351→          onChange={(e) => setValue(e.target.value)}
   352→          onBlur={handleSave}
   353→          onKeyDown={(e) => {
   354→            if (e.key === 'Enter') handleSave();
   355→            if (e.key === 'Escape') { setValue(String(grade.score)); setEditing(false); }
   356→          }}
   357→          className="h-7 w-16 text-xs bg-white/[0.06] border border-[#D4A574]/40 text-[#C4C4D4] rounded px-2 outline-none tabular-nums"
   358→          dir="ltr"
   359→        />
   360→        <span className="text-[10px] text-[#8B8B9E]">/ {grade.maxScore}</span>
   361→      </div>
   362→    );
   363→  }
   364→
   365→  return (
   366→    <button
   367→      onClick={() => { setValue(String(grade.score)); setEditing(true); }}
   368→      className="cursor-pointer hover:opacity-80 transition-opacity"
   369→      title="انقر لتعديل الدرجة"
   370→    >
   371→      <ScoreBadge score={grade.score} maxScore={grade.maxScore} />
   372→    </button>
   373→  );
   374→}

   796→// Pure helper: compute class averages for performance comparison chart
   797→// ---------------------------------------------------------------------------
   798→
   799→function computeClassAverages(grades: GradeRecord[]): { className: string; average: number; count: number }[] {
   800→  const classMap = new Map<string, { totalWeighted: number; totalWeight: number; count: number }>();
   801→  for (const grade of grades) {
   802→    const className = grade.className ?? grade.classId ?? 'غير محدد';
   803→    const existing = classMap.get(className) ?? { totalWeighted: 0, totalWeight: 0, count: 0 };
   804→    const eps = 0.001;
   805→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   806→    classMap.set(className, {
   807→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
   808→      totalWeight: existing.totalWeight + grade.weight,
   809→      count: existing.count + 1,
   810→    });
   811→  }
   812→  const result: { className: string; average: number; count: number }[] = [];
   813→  classMap.forEach((val, key) => {
   814→    result.push({
   815→      className: key,
   816→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
   817→      count: val.count,
   818→    });
   819→  });
   820→  return result.sort((a, b) => b.average - a.average);
   821→}

   823→// ---------------------------------------------------------------------------
   824→// Pure helper: compute grade distribution counts
   825→// ---------------------------------------------------------------------------
   826→
   827→function computeGradeDistribution(grades: GradeRecord[]): { excellent: number; veryGood: number; good: number; weak: number } {
   828→  const eps = 0.001;
   829→  let excellent = 0;
   830→  let veryGood = 0;
   831→  let good = 0;
   832→  let weak = 0;
   833→  for (const grade of grades) {
   834→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   835→    if (pct >= 90) excellent++;
   836→    else if (pct >= 75) veryGood++;
   837→    else if (pct >= 60) good++;
   838→    else weak++;

  1024→function computeSubjectAverages(grades: GradeRecord[]): { subject: string; average: number }[] {
  1025→  const subjectMap = new Map<string, { totalWeighted: number; totalWeight: number }>();
  1026→  for (const grade of grades) {
  1027→    const existing = subjectMap.get(grade.subject) ?? { totalWeighted: 0, totalWeight: 0 };
  1028→    const pct = (grade.score / grade.maxScore) * 100;
  1029→    subjectMap.set(grade.subject, {
  1030→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
  1031→      totalWeight: existing.totalWeight + grade.weight,
  1032→    });
  1033→  }
  1034→  const result: { subject: string; average: number }[] = [];
  1035→  subjectMap.forEach((val, key) => {
  1036→    result.push({
  1037→      subject: key,
  1038→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
  1039→    });
  1040→  });
  1041→  return result.sort((a, b) => b.average - a.average);
  1042→}
