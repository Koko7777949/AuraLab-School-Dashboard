   298→function getRelativeTime(dateString: string): string {
   299→  const now = Date.now();
   300→  const date = new Date(dateString).getTime();
   301→  const diffMs = now - date;
   302→  const diffSeconds = Math.floor(diffMs / 1000);
   303→  const diffMinutes = Math.floor(diffSeconds / 60);
   304→  const diffHours = Math.floor(diffMinutes / 60);
   305→  const diffDays = Math.floor(diffHours / 24);
   306→
   307→  if (diffSeconds < 60) return 'منذ لحظات';
   308→  if (diffMinutes < 60) return `منذ ${diffMinutes} دقيقة`;
   309→  if (diffHours < 24) return `منذ ${diffHours} ساعة`;
   310→  if (diffDays < 7) return `منذ ${diffDays} يوم`;
   311→  return new Date(dateString).toLocaleDateString('ar-SA');
   312→}
   313→
   314→function isExpired(expiresAt: string | null): boolean {
   315→  if (!expiresAt) return false;
   316→  return new Date(expiresAt) < new Date();
   317→}

   966→        }
   967→      }
   968→      return true;
   969→    });
   970→
   971→    return {
   972→      pinned: allFiltered.filter((a) => a.pinned),
   973→      unpinned: allFiltered.filter((a) => !a.pinned),
   974→    };

   987→  useEffect(() => {
   988→    const prev = prevFiltersRef.current;
   989→    if (prev.categoryFilter !== categoryFilter || prev.localSearch !== localSearch) {
