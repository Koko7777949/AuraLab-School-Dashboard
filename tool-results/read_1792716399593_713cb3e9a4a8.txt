  1182→/**
  1183→ * Translate a key to the current locale string.
  1184→ * Falls back to Arabic if key not found in the target locale.
  1185→ * Falls back to the key itself if not found anywhere.
  1186→ */
  1187→export function t(key: string, locale: Locale = 'ar'): string {
  1188→  return translations[locale]?.[key] ?? translations.ar[key] ?? key;
  1189→}
