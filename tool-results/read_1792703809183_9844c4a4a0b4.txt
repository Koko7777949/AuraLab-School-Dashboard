   248→interface QuickStatItem {
   249→  label: string;
   250→  value: number;
   251→  suffix?: string;
   252→  icon: React.ElementType;
   253→  color: string;
   254→  gradientFrom: string;
   255→  gradientTo: string;
   256→  trend?: string;
   257→}
   258→
   259→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   260→  const [count, setCount] = useState(0);
   261→  const ref = useRef<HTMLSpanElement>(null);
   262→  const started = useRef(false);
   263→
   264→  useEffect(() => {
   265→    if (started.current) return;
   266→    started.current = true;
   267→    const start = performance.now();
   268→    function step(now: number) {
   269→      const progress = Math.min((now - start) / duration, 1);
   270→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
   271→      setCount(Math.round(eased * target));
   272→      if (progress < 1) requestAnimationFrame(step);
   273→    }
   274→    requestAnimationFrame(step);
   275→  }, [target, duration]);
   276→
   277→  return <span ref={ref} className="tabular-nums">{count}</span>;
   278→}
