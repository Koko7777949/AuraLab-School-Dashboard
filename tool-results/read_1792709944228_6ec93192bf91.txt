   155→function getScoreColor(pct: number): { bar: string; text: string; bg: string; badge: string } {
   156→  if (pct >= 90) return {
   157→    bar: 'bg-emerald-500',
   158→    text: 'text-emerald-400',
   159→    bg: 'bg-emerald-500/10',
   160→    badge: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/20',
   161→  };
   162→  if (pct >= 70) return {
   163→    bar: 'bg-amber-500',
   164→    text: 'text-amber-400',
   165→    bg: 'bg-amber-500/10',
   166→    badge: 'bg-amber-500/15 text-amber-400 border-amber-500/20',
   167→  };
   168→  return {
   169→    bar: 'bg-rose-500',
   170→    text: 'text-rose-400',
   171→    bg: 'bg-rose-500/10',
   172→    badge: 'bg-rose-500/15 text-rose-400 border-rose-500/20',
   173→  };
   174→}
   175→
   176→function getGpaColor(gpa: number): string {
   177→  if (gpa >= 3.5) return 'text-emerald-400';
   178→  if (gpa >= 2.5) return 'text-amber-400';
   179→  return 'text-rose-400';
   180→}
   181→
   182→function getGpaLabel(gpa: number): string {
   183→  if (gpa >= 3.5) return 'ممتاز';
   184→  if (gpa >= 3.0) return 'جيد جداً';
   185→  if (gpa >= 2.5) return 'جيد';
   186→  if (gpa >= 2.0) return 'مقبول';
   187→  return 'يحتاج تحسين';
   188→}
   189→
   190→function calculateGpa(score: number, maxScore: number): number {
   191→  const pct = (score / maxScore) * 100;
   192→  if (pct >= 90) return 4.0;
   193→  if (pct >= 85) return 3.75;
   194→  if (pct >= 80) return 3.5;
   195→  if (pct >= 75) return 3.25;
   196→  if (pct >= 70) return 3.0;
   197→  if (pct >= 65) return 2.75;
   198→  if (pct >= 60) return 2.5;
   199→  if (pct >= 55) return 2.25;
   200→  if (pct >= 50) return 2.0;
   201→  return 1.0;
   202→}

   747→// Pure helper: compute class averages for performance comparison chart
   748→// ---------------------------------------------------------------------------
   749→
   750→function computeClassAverages(grades: GradeRecord[]): { className: string; average: number; count: number }[] {
   751→  const classMap = new Map<string, { totalWeighted: number; totalWeight: number; count: number }>();
   752→  for (const grade of grades) {
   753→    const className = grade.className ?? grade.classId ?? 'غير محدد';
   754→    const existing = classMap.get(className) ?? { totalWeighted: 0, totalWeight: 0, count: 0 };
   755→    const eps = 0.001;
   756→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   757→    classMap.set(className, {
   758→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
   759→      totalWeight: existing.totalWeight + grade.weight,
   760→      count: existing.count + 1,
   761→    });
   762→  }
   763→  const result: { className: string; average: number; count: number }[] = [];
   764→  classMap.forEach((val, key) => {
   765→    result.push({
   766→      className: key,
   767→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
   768→      count: val.count,
   769→    });
   770→  });
   771→  return result.sort((a, b) => b.average - a.average);
   772→}

   774→// ---------------------------------------------------------------------------
   775→// Pure helper: compute grade distribution counts
   776→// ---------------------------------------------------------------------------
   777→
   778→function computeGradeDistribution(grades: GradeRecord[]): { excellent: number; veryGood: number; good: number; weak: number } {
   779→  const eps = 0.001;
   780→  let excellent = 0;
   781→  let veryGood = 0;
   782→  let good = 0;
   783→  let weak = 0;
   784→  for (const grade of grades) {
   785→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   786→    if (pct >= 90) excellent++;
   787→    else if (pct >= 75) veryGood++;
   788→    else if (pct >= 60) good++;
   789→    else weak++;
   790→  }
   791→  return { excellent, veryGood, good, weak };
   792→}

   971→// ---------------------------------------------------------------------------
   972→// Pure helper: compute subject averages for report card
   973→// ---------------------------------------------------------------------------
   974→
   975→function computeSubjectAverages(grades: GradeRecord[]): { subject: string; average: number }[] {
   976→  const subjectMap = new Map<string, { totalWeighted: number; totalWeight: number }>();
   977→  for (const grade of grades) {
   978→    const existing = subjectMap.get(grade.subject) ?? { totalWeighted: 0, totalWeight: 0 };
   979→    const pct = (grade.score / grade.maxScore) * 100;
   980→    subjectMap.set(grade.subject, {
   981→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
   982→      totalWeight: existing.totalWeight + grade.weight,
   983→    });
   984→  }
   985→  const result: { subject: string; average: number }[] = [];
   986→  subjectMap.forEach((val, key) => {
   987→    result.push({
   988→      subject: key,
   989→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
   990→    });
   991→  });
   992→  return result.sort((a, b) => b.average - a.average);
   993→}
