   266→interface QuickStatItem {
   267→  label: string;
   268→  value: number;
   269→  suffix?: string;
   270→  icon: React.ElementType;
   271→  color: string;
   272→  gradientFrom: string;
   273→  gradientTo: string;
   274→  trend?: string;
   275→}
   276→
   277→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   278→  const [count, setCount] = useState(0);
   279→  const ref = useRef<HTMLSpanElement>(null);
   280→  const started = useRef(false);
   281→
   282→  useEffect(() => {
   283→    if (started.current) return;
   284→    started.current = true;
   285→    const start = performance.now();
   286→    function step(now: number) {
   287→      const progress = Math.min((now - start) / duration, 1);
   288→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
   289→      setCount(Math.round(eased * target));
   290→      if (progress < 1) requestAnimationFrame(step);
   291→    }
   292→    requestAnimationFrame(step);
   293→  }, [target, duration]);
   294→
   295→  return <span ref={ref} className="tabular-nums">{count}</span>;
   296→}
