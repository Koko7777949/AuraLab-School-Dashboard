   284→function ReportBarChart({
   285→  data,
   286→  height = 180,
   287→}: {
   288→  data: { label: string; value: number; color: string }[];
   289→  height?: number;
   290→}) {
   291→  const maxValue = Math.max(...data.map((d) => d.value), 1);
   292→  const barWidth = Math.min(40, Math.max(16, 200 / data.length));
   293→  const gap = Math.max(4, (200 - barWidth * data.length) / (data.length + 1));
   294→
   295→  return (
   296→    <svg
   297→      viewBox={`0 0 200 ${height + 30}`}
   298→      width="100%"
   299→      height={height + 30}
   300→      preserveAspectRatio="xMidYMid meet"
   301→    >
   302→      {/* Grid lines */}
   303→      {[0, 25, 50, 75, 100].map((pct) => {
   304→        const y = height - (pct / 100) * height + 5;
   305→        return (
   306→          <g key={pct}>
   307→            <line
   308→              x1="25"
   309→              y1={y}
   310→              x2="195"
   311→              y2={y}
   312→              stroke="rgba(255,255,255,0.05)"
   313→              strokeDasharray="4 3"
   314→            />
   315→            <text
   316→              x="20"
   317→              y={y + 3}
   318→              textAnchor="end"
   319→              fill="rgba(255,255,255,0.25)"
   320→              fontSize="7"
   321→            >
   322→              {pct}%
   323→            </text>
   324→          </g>
   325→        );
   326→      })}
   327→
   328→      {/* Bars */}
   329→      {data.map((item, i) => {
   330→        const barHeight = (item.value / maxValue) * height;
   331→        const x = 30 + i * (barWidth + gap);
   332→        const y = height - barHeight + 5;
   333→
   334→        return (
   335→          <g key={item.label}>
   336→            {/* Bar gradient */}
   337→            <defs>
   338→              <linearGradient id={`bar-grad-${i}`} x1="0" y1="0" x2="0" y2="1">
   339→                <stop offset="0%" stopColor={item.color} stopOpacity="1" />
   340→                <stop offset="100%" stopColor={item.color} stopOpacity="0.4" />
   341→              </linearGradient>
   342→            </defs>
   343→            {/* Bar */}
   344→            <rect
   345→              x={x}
   346→              y={y}
   347→              width={barWidth}
   348→              height={barHeight}
   349→              rx="3"
   350→              fill={`url(#bar-grad-${i})`}
   351→              className="transition-all duration-700"
   352→              style={{ animationDelay: `${i * 80}ms` }}
   353→            />
   354→            {/* Value on top */}
   355→            <text
   356→              x={x + barWidth / 2}
   357→              y={y - 4}
   358→              textAnchor="middle"
   359→              fill="white"
   360→              fontSize="8"
   361→              fontWeight="600"
   362→            >
   363→              {item.value}
   364→            </text>
   365→            {/* Label below */}
   366→            <text
   367→              x={x + barWidth / 2}
   368→              y={height + 22}
   369→              textAnchor="middle"
   370→              fill="rgba(255,255,255,0.4)"
   371→              fontSize="7"
   372→            >
   373→              {item.label}
   374→            </text>
   375→          </g>
   376→        );
   377→      })}
   378→    </svg>
   379→  );
   380→}

   382→function ReportDonutChart({
   383→  data,
   384→  size = 160,
   385→}: {
   386→  data: { label: string; value: number; color: string }[];
   387→  size?: number;
   388→}) {
   389→  const total = data.reduce((sum, d) => sum + d.value, 0);
   390→  const strokeWidth = 24;
   391→  const radius = (size - strokeWidth) / 2;
   392→  const circumference = 2 * Math.PI * radius;
   393→  const center = size / 2;
   394→
   395→  // Compute segment offsets immutably
   396→  const segments = data.reduce(
   397→    (acc, item) => {
   398→      const pct = item.value / total;
   399→      const dashLength = pct * circumference;
   400→      const gapSize = 2; // small gap between segments
   401→      const offset = acc.accumulated;
   402→      return {
   403→        segments: [
   404→          ...acc.segments,
   405→          { ...item, dashLength, gapSize, offset },
   406→        ],
   407→        accumulated: offset + dashLength + gapSize,
   408→      };
   409→    },
   410→    { segments: [] as { label: string; value: number; color: string; dashLength: number; gapSize: number; offset: number }[], accumulated: 0 },
   411→  ).segments;
   412→
   413→  return (
   414→    <div className="flex flex-col items-center">
   415→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
   416→        {segments.map((seg, i) => (
   417→          <circle
   418→            key={seg.label}
   419→            cx={center}
   420→            cy={center}
   421→            r={radius}
   422→            fill="none"
   423→            stroke={seg.color}
   424→            strokeWidth={strokeWidth}
   425→            strokeDasharray={`${seg.dashLength} ${circumference - seg.dashLength}`}
   426→            strokeDashoffset={-seg.offset}
   427→            strokeLinecap="round"
   428→            className="transition-all duration-700"
   429→            style={{ animationDelay: `${i * 100}ms` }}
   430→          />
   431→        ))}
   432→        {/* Center text */}
   433→        <text
   434→          x={center}
   435→          y={center - 6}
   436→          textAnchor="middle"
   437→          fill="white"
   438→          fontSize="20"
   439→          fontWeight="700"
   440→        >
   441→          {total}
   442→        </text>
   443→        <text
   444→          x={center}
   445→          y={center + 12}
   446→          textAnchor="middle"
   447→          fill="rgba(255,255,255,0.4)"
   448→          fontSize="10"
   449→        >
   450→          إجمالي
   451→        </text>
   452→      </svg>
   453→      {/* Legend */}
   454→      <div className="flex flex-wrap items-center justify-center gap-3 mt-3">
   455→        {data.map((item) => (
   456→          <div key={item.label} className="flex items-center gap-1.5">
   457→            <span
   458→              className="h-2 w-2 rounded-full"
   459→              style={{ backgroundColor: item.color }}
   460→            />
   461→            <span className="text-[10px] text-slate-400">
   462→              {item.label} ({item.value})
   463→            </span>
   464→          </div>
   465→        ))}
   466→      </div>
   467→    </div>
   468→  );
   469→}

   471→function ReportTrendChart({
   472→  data,
   473→  height = 80,
   474→  width = 300,
   475→}: {
   476→  data: { label: string; value: number }[];
   477→  height?: number;
   478→  width?: number;
   479→}) {
   480→  const maxValue = Math.max(...data.map((d) => d.value), 1);
   481→  const minValue = Math.min(...data.map((d) => d.value), 0);
   482→  const range = maxValue - minValue || 1;
   483→  const padding = { top: 10, bottom: 20, left: 30, right: 10 };
   484→  const chartW = width - padding.left - padding.right;
   485→  const chartH = height - padding.top - padding.bottom;
   486→
   487→  const points = data.map((d, i) => ({
   488→    x: padding.left + (i / Math.max(data.length - 1, 1)) * chartW,
   489→    y: padding.top + chartH - ((d.value - minValue) / range) * chartH,
   490→    label: d.label,
   491→    value: d.value,
   492→  }));
   493→
   494→  const linePath =
   495→    points.length > 1
   496→      ? points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ')
   497→      : '';
   498→
   499→  const areaPath =
   500→    points.length > 1
   501→      ? `${linePath} L ${points[points.length - 1].x} ${padding.top + chartH} L ${points[0].x} ${padding.top + chartH} Z`
   502→      : '';
   503→
   504→  return (
   505→    <svg
   506→      viewBox={`0 0 ${width} ${height}`}
   507→      width="100%"
   508→      height={height}
   509→      preserveAspectRatio="xMidYMid meet"
   510→    >
   511→      <defs>
   512→        <linearGradient id="trend-fill" x1="0" y1="0" x2="0" y2="1">
   513→          <stop offset="0%" stopColor="#D4A574" stopOpacity="0.3" />
   514→          <stop offset="100%" stopColor="#D4A574" stopOpacity="0" />
   515→        </linearGradient>
   516→      </defs>
   517→
   518→      {/* Grid */}
   519→      {[0, 1, 2, 3].map((i) => {
   520→        const y = padding.top + (i / 3) * chartH;
   521→        return (
   522→          <line
   523→            key={i}
   524→            x1={padding.left}
   525→            y1={y}
   526→            x2={width - padding.right}
   527→            y2={y}
   528→            stroke="rgba(255,255,255,0.04)"
   529→            strokeDasharray="3 3"
   530→          />
   531→        );
   532→      })}
   533→
   534→      {/* Area */}
   535→      {areaPath && <path d={areaPath} fill="url(#trend-fill)" />}
   536→
   537→      {/* Line */}
   538→      {linePath && (
   539→        <path
   540→          d={linePath}
   541→          fill="none"
   542→          stroke="#D4A574"
   543→          strokeWidth="2"
   544→          strokeLinecap="round"
   545→          strokeLinejoin="round"
   546→        />
   547→      )}
   548→
   549→      {/* Dots & labels */}
   550→      {points.map((p) => (
   551→        <g key={p.label}>
   552→          <circle cx={p.x} cy={p.y} r="3" fill="#D4A574" stroke="#1A1A24" strokeWidth="1.5" />
   553→          <text
   554→            x={p.x}
   555→            y={p.y - 8}
   556→            textAnchor="middle"
   557→            fill="rgba(255,255,255,0.7)"
   558→            fontSize="7"
   559→            fontWeight="600"
   560→          >
   561→            {p.value}%
   562→          </text>
   563→          <text
   564→            x={p.x}
   565→            y={height - 4}

   578→// ---------------------------------------------------------------------------
   579→// Animated Counter
   580→// ---------------------------------------------------------------------------
   581→
   582→function AnimatedCounter({ target, duration = 800 }: { target: number; duration?: number }) {
   583→  const [count, setCount] = useState(0);
   584→  const rafRef = useRef(0);
   585→
   586→  useEffect(() => {
   587→    const start = performance.now();
   588→    const animate = (now: number) => {
   589→      const elapsed = now - start;
   590→      const progress = Math.min(elapsed / duration, 1);
   591→      // Ease out cubic
   592→      const eased = 1 - Math.pow(1 - progress, 3);
   593→      setCount(Math.round(eased * target));
   594→      if (progress < 1) {
   595→        rafRef.current = requestAnimationFrame(animate);
   596→      }
   597→    };
   598→    rafRef.current = requestAnimationFrame(animate);
   599→    return () => cancelAnimationFrame(rafRef.current);
   600→  }, [target, duration]);
   601→
   602→  return <span className="tabular-nums">{count}</span>;
   603→}

  1363→function generateReport(
  1364→  type: ReportType,
  1365→  title: string,
  1366→  dateRange: { from: string; to: string },
  1367→  filters: ReportFilter,
  1368→  storeData: {
  1369→    students: ReturnType<typeof useAppStore.getState>['students'];
  1370→    teachers: ReturnType<typeof useAppStore.getState>['teachers'];
  1371→    classes: ReturnType<typeof useAppStore.getState>['classes'];
  1372→    exams: ReturnType<typeof useAppStore.getState>['exams'];
  1373→    grades: ReturnType<typeof useAppStore.getState>['grades'];
  1374→  },
  1375→): GeneratedReport {
  1376→  const { students, teachers, classes, exams, grades } = storeData;
  1377→  const id = `r-${Date.now()}`;
  1378→  const now = new Date().toISOString();
  1379→
  1380→  // Get unique class names
  1381→  const classNames = classes.length > 0
  1382→    ? classes.map((c) => c.name)
  1383→    : ['الصف الأول - أ', 'الصف الأول - ب', 'الصف الثاني - أ', 'الصف الثاني - ب', 'الصف الثالث - أ', 'الصف الثالث - ب'];
  1384→
  1385→  switch (type) {
  1386→    case 'students':
  1387→      return generateStudentReport(id, title, now, dateRange, filters, students, classNames);
  1388→    case 'teachers':
  1389→      return generateTeacherReport(id, title, now, dateRange, filters, teachers);
  1390→    case 'attendance':
  1391→      return generateAttendanceReport(id, title, now, dateRange, filters, classNames);
  1392→    case 'grades':
  1393→      return generateGradesReport(id, title, now, dateRange, filters, grades);
  1394→    case 'announcements':
  1395→      return generateAnnouncementsReport(id, title, now, dateRange, filters);
  1396→    case 'comprehensive':
  1397→      return generateComprehensiveReport(id, title, now, dateRange, filters, students, teachers, classNames, grades);
  1398→    default:
  1399→      return generateStudentReport(id, title, now, dateRange, filters, students, classNames);
  1400→  }
  1401→}
