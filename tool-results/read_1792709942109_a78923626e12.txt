   265→interface QuickStatItem {
   266→  label: string;
   267→  value: number;
   268→  suffix?: string;
   269→  icon: React.ElementType;
   270→  color: string;
   271→  gradientFrom: string;
   272→  gradientTo: string;
   273→  trend?: string;
   274→}
   275→
   276→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   277→  const [count, setCount] = useState(0);
   278→  const ref = useRef<HTMLSpanElement>(null);
   279→  const started = useRef(false);
   280→
   281→  useEffect(() => {
   282→    if (started.current) return;
   283→    started.current = true;
   284→    const start = performance.now();
   285→    function step(now: number) {
   286→      const progress = Math.min((now - start) / duration, 1);
   287→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
   288→      setCount(Math.round(eased * target));
   289→      if (progress < 1) requestAnimationFrame(step);
   290→    }
   291→    requestAnimationFrame(step);
   292→  }, [target, duration]);
   293→
   294→  return <span ref={ref} className="tabular-nums">{count}</span>;
   295→}
