   244→interface QuickStatItem {
   245→  label: string;
   246→  value: number;
   247→  suffix?: string;
   248→  icon: React.ElementType;
   249→  color: string;
   250→  gradientFrom: string;
   251→  gradientTo: string;
   252→  trend?: string;
   253→}
   254→
   255→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   256→  const [count, setCount] = useState(0);
   257→  const ref = useRef<HTMLSpanElement>(null);
   258→  const started = useRef(false);
   259→
   260→  useEffect(() => {
   261→    if (started.current) return;
   262→    started.current = true;
   263→    const start = performance.now();
   264→    function step(now: number) {
   265→      const progress = Math.min((now - start) / duration, 1);
   266→      const eased = 1 - Math.pow(1 - progress, 3); // ease-out cubic
   267→      setCount(Math.round(eased * target));
   268→      if (progress < 1) requestAnimationFrame(step);
   269→    }
   270→    requestAnimationFrame(step);
   271→  }, [target, duration]);
   272→
   273→  return <span ref={ref} className="tabular-nums">{count}</span>;
   274→}
