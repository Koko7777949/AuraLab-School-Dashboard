   278→function getClassName(classId: string | null, classes: { id: string; name: string }[]): string {
   279→  if (!classId) return '—';
   280→  const found = classes.find((c) => c.id === classId);
   281→  if (found) return found.name;
   282→  const legacy = CLASS_OPTIONS.find((c) => c.value === classId);
   283→  return legacy ? legacy.label : '—';
   284→}

   286→/** Highlight matching text in amber color */
   287→function HighlightText({ text, query }: { text: string; query: string }) {
   288→  if (!query.trim()) return <>{text}</>;
   289→  const lowerText = text.toLowerCase();
   290→  const lowerQuery = query.trim().toLowerCase();
   291→  const index = lowerText.indexOf(lowerQuery);
   292→  if (index === -1) return <>{text}</>;
   293→  return (
   294→    <>
   295→      {text.slice(0, index)}
   296→      <span className="text-[#D4A574] font-bold">{text.slice(index, index + query.trim().length)}</span>
   297→      {text.slice(index + query.trim().length)}
   298→    </>
   299→  );
   300→}

   448→function StudentFormContent({
   449→  editingStudent,
   450→  onClose,
   451→}: {
   452→  editingStudent: Student | null;
   453→  onClose: () => void;
   454→}) {
   455→  const { addStudent, updateStudent, classes: storeClasses } = useAppStore();
   456→  const isEditing = !!editingStudent;
   457→
   458→  // Dynamic class options from store, with fallback to static for legacy data
   459→  const classOptions = useMemo(() => {
   460→    if (storeClasses.length > 0) {
   461→      return storeClasses.map((c) => ({ value: c.id, label: c.name }));
   462→    }
   463→    return CLASS_OPTIONS;
   464→  }, [storeClasses]);
   465→
   466→  // Form state – initialised from props on mount
   467→  const [name, setName] = useState(editingStudent?.name ?? '');
   468→  const [studentId, setStudentId] = useState(editingStudent?.studentId ?? '');
   469→  const [classId, setClassId] = useState<string>(editingStudent?.classId ?? '');
   470→  const [status, setStatus] = useState<Student['status']>(editingStudent?.status ?? 'active');
   471→  const [notes, setNotes] = useState(editingStudent?.notes ?? '');
   472→  const [psychBackground, setPsychBackground] = useState(editingStudent?.psychBackground ?? '');
   473→
   474→  const handleSubmit = (e: React.FormEvent) => {
   475→    e.preventDefault();
   476→
   477→    if (!name.trim() || !studentId.trim()) return;
   478→
   479→    const now = new Date().toISOString();
   480→    // Convert empty string to null for classId (Student type expects string | null)
   481→    const resolvedClassId = classId || null;
   482→
   483→    if (isEditing && editingStudent) {
   484→      updateStudent(editingStudent.id, {
   485→        name: name.trim(),
   486→        studentId: studentId.trim(),
   487→        classId: resolvedClassId,
   488→        status,
   489→        notes: notes.trim(),
   490→        psychBackground: psychBackground.trim(),
   491→        updatedAt: now,
   492→      });
   493→      toast.success('تم تحديث بيانات الطالب');
   494→    } else {
   495→      const newStudent: Student = {
   496→        id: `std-${Date.now()}`,
   497→        name: name.trim(),
   498→        studentId: studentId.trim(),
   499→        classId: resolvedClassId,
   500→        status,
   501→        notes: notes.trim(),
   502→        psychBackground: psychBackground.trim(),
   503→        avatar: '',
   504→        createdAt: now,
   505→        updatedAt: now,
   506→      };
   507→      addStudent(newStudent);
   508→      toast.success('تم إضافة الطالب بنجاح');
   509→    }
   510→
   511→    onClose();
   512→  };
   513→
   514→  return (
   515→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   516→      {/* Name */}
   517→      <div className="flex flex-col gap-2">
   518→        <Label className="text-slate-300 text-sm">اسم الطالب الكامل</Label>
   519→        <Input
   520→          value={name}
   521→          onChange={(e) => setName(e.target.value)}
   522→          placeholder="مثال: أحمد محمد علي"
   523→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   524→          required
