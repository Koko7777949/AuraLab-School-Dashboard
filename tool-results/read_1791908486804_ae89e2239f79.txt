   211→
   212→function getInitials(name: string): string {
   213→  const parts = name.trim().split(/\s+/);
   214→  if (parts.length >= 2) {
   215→    return parts[0].charAt(0) + parts[1].charAt(0);
   216→  }
   217→  return name.charAt(0);
   218→}

   392→function TeacherFormContent({
   393→  editingTeacher,
   394→  onClose,
   395→}: {
   396→  editingTeacher: Teacher | null;
   397→  onClose: () => void;
   398→}) {
   399→  const { addTeacher, updateTeacher } = useAppStore();
   400→  const isEditing = !!editingTeacher;
   401→
   402→  // Form state
   403→  const [name, setName] = useState(editingTeacher?.name ?? '');
   404→  const [subject, setSubject] = useState(editingTeacher?.subject ?? '');
   405→  const [email, setEmail] = useState(editingTeacher?.email ?? '');
   406→  const [phone, setPhone] = useState(editingTeacher?.phone ?? '');
   407→  const [permissions, setPermissions] = useState<Teacher['permissions']>(
   408→    editingTeacher?.permissions ?? 'teacher'
   409→  );
   410→  const [status, setStatus] = useState<Teacher['status']>(
   411→    editingTeacher?.status ?? 'active'
   412→  );
   413→
   414→  const handleSubmit = (e: React.FormEvent) => {
   415→    e.preventDefault();
   416→    if (!name.trim() || !subject.trim()) return;
   417→
   418→    const now = new Date().toISOString();
   419→
   420→    if (isEditing && editingTeacher) {
   421→      updateTeacher(editingTeacher.id, {
   422→        name: name.trim(),
   423→        subject: subject.trim(),
   424→        email: email.trim(),
   425→        phone: phone.trim(),
   426→        permissions,
   427→        status,
   428→        updatedAt: now,
   429→      });
   430→      toast.success('تم تحديث بيانات المعلم');
   431→    } else {
   432→      const newTeacher: Teacher = {
   433→        id: `tch-${Date.now()}`,
   434→        name: name.trim(),
   435→        subject: subject.trim(),
   436→        status,
   437→        permissions,
   438→        avatar: '',
   439→        email: email.trim(),
   440→        phone: phone.trim(),
   441→        createdAt: now,
   442→        updatedAt: now,
   443→        classNames: [],
   444→      };
   445→      addTeacher(newTeacher);
   446→      toast.success('تم إضافة المعلم بنجاح');
   447→    }
   448→
   449→    onClose();
   450→  };

   452→  return (

   459→          onChange={(e) => setName(e.target.value)}
   460→          placeholder="مثال: محمد بن عبدالله السالم"
   461→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   462→          required
   463→        />
   464→      </div>
   465→
   466→      {/* Subject */}

   468→        <Label className="text-slate-300 text-sm">المادة الدراسية</Label>

   470→          value={subject}
   471→          onChange={(e) => setSubject(e.target.value)}
   472→          placeholder="مثال: الرياضيات"
   473→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"

   479→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
