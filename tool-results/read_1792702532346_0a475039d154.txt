   118→/** Simple relative-time formatter for Arabic */
   119→function relativeTime(dateStr: string): string {
   120→  const now = Date.now();
   121→  const then = new Date(dateStr).getTime();
   122→  const diffMs = now - then;
   123→  const seconds = Math.floor(diffMs / 1000);
   124→  const minutes = Math.floor(seconds / 60);
   125→  const hours = Math.floor(minutes / 60);
   126→  const days = Math.floor(hours / 24);
   127→
   128→  if (seconds < 60) return 'الآن';
   129→  if (minutes < 60) return `منذ ${minutes} دقيقة`;
   130→  if (hours < 24) return `منذ ${hours} ساعة`;
   131→  if (days < 7) return `منذ ${days} يوم`;
   132→  return new Date(dateStr).toLocaleDateString('ar-EG');
   133→}

   135→function getInitials(name: string): string {
   136→  const parts = name.trim().split(/\s+/);
   137→  if (parts.length >= 2) return parts[0][0] + parts[1][0];
   138→  return parts[0].slice(0, 2);
   139→}
