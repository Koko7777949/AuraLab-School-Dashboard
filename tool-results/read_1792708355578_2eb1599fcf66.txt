   860→   860→/**
   861→   861→ * Translate a key to the current locale string.
   862→   862→ * Falls back to Arabic if key not found in the target locale.
   863→   863→ * Falls back to the key itself if not found anywhere.
   864→   864→ */
   865→   865→export function t(key: string, locale: Locale = 'ar'): string {
   866→   866→  return translations[locale]?.[key] ?? translations.ar[key] ?? key;
   867→   867→}
