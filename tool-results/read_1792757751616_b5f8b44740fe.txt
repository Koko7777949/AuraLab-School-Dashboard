   154→function getScoreColor(pct: number): { bar: string; text: string; bg: string; badge: string } {
   155→  if (pct >= 90) return {
   156→    bar: 'bg-emerald-500',
   157→    text: 'text-emerald-400',
   158→    bg: 'bg-emerald-500/10',
   159→    badge: 'bg-emerald-500/15 text-emerald-400 border-emerald-500/20',
   160→  };
   161→  if (pct >= 70) return {
   162→    bar: 'bg-amber-500',
   163→    text: 'text-amber-400',
   164→    bg: 'bg-amber-500/10',
   165→    badge: 'bg-amber-500/15 text-amber-400 border-amber-500/20',
   166→  };
   167→  return {
   168→    bar: 'bg-rose-500',
   169→    text: 'text-rose-400',
   170→    bg: 'bg-rose-500/10',
   171→    badge: 'bg-rose-500/15 text-rose-400 border-rose-500/20',
   172→  };
   173→}
   174→
   175→function getGpaColor(gpa: number): string {
   176→  if (gpa >= 3.5) return 'text-emerald-400';
   177→  if (gpa >= 2.5) return 'text-amber-400';
   178→  return 'text-rose-400';
   179→}
   180→
   181→function getGpaLabel(gpa: number): string {
   182→  if (gpa >= 3.5) return 'ممتاز';
   183→  if (gpa >= 3.0) return 'جيد جداً';
   184→  if (gpa >= 2.5) return 'جيد';
   185→  if (gpa >= 2.0) return 'مقبول';
   186→  return 'يحتاج تحسين';
   187→}
   188→
   189→function calculateGpa(score: number, maxScore: number): number {
   190→  const pct = (score / maxScore) * 100;
   191→  if (pct >= 90) return 4.0;
   192→  if (pct >= 85) return 3.75;
   193→  if (pct >= 80) return 3.5;
   194→  if (pct >= 75) return 3.25;
   195→  if (pct >= 70) return 3.0;
   196→  if (pct >= 65) return 2.75;
   197→  if (pct >= 60) return 2.5;
   198→  if (pct >= 55) return 2.25;
   199→  if (pct >= 50) return 2.0;
   200→  return 1.0;
   201→}

   749→function computeClassAverages(grades: GradeRecord[]): { className: string; average: number; count: number }[] {
   750→  const classMap = new Map<string, { totalWeighted: number; totalWeight: number; count: number }>();
   751→  for (const grade of grades) {
   752→    const className = grade.className ?? grade.classId ?? 'غير محدد';
   753→    const existing = classMap.get(className) ?? { totalWeighted: 0, totalWeight: 0, count: 0 };
   754→    const eps = 0.001;
   755→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   756→    classMap.set(className, {
   757→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
   758→      totalWeight: existing.totalWeight + grade.weight,
   759→      count: existing.count + 1,
   760→    });
   761→  }
   762→  const result: { className: string; average: number; count: number }[] = [];
   763→  classMap.forEach((val, key) => {
   764→    result.push({
   765→      className: key,
   766→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
   767→      count: val.count,
   768→    });
   769→  });
   770→  return result.sort((a, b) => b.average - a.average);
   771→}

   774→// Pure helper: compute grade distribution counts
   775→// ---------------------------------------------------------------------------

   777→function computeGradeDistribution(grades: GradeRecord[]): { excellent: number; veryGood: number; good: number; weak: number } {
   778→  const eps = 0.001;
   779→  let excellent = 0;
   780→  let veryGood = 0;
   781→  let good = 0;
   782→  let weak = 0;
   783→  for (const grade of grades) {
   784→    const pct = (grade.score / Math.max(eps, grade.maxScore)) * 100;
   785→    if (pct >= 90) excellent++;
   786→    else if (pct >= 75) veryGood++;
   787→    else if (pct >= 60) good++;
   788→    else weak++;
   789→  }
   790→  return { excellent, veryGood, good, weak };
   791→}

   804→function ClassPerformanceChart({ classData }: { classData: { className: string; average: number; count: number }[] }) {

   819→    );
   820→  }
   821→
   822→  const maxAvg = Math.max(...classData.map((c) => c.average), 1);
   823→  const eps = 0.001;
   824→
   825→  return (
   826→    <div className="flex flex-col gap-4">
   827→      {/* Chart */}
   828→      <div className="flex flex-col gap-3">
   829→        {classData.map((item, idx) => {
   830→          const barWidth = Math.max(eps, (item.average / maxAvg) * 100);

   833→
   834→          return (
   835→            <div
   836→              key={item.className}

   971→// Pure helper: compute subject averages for report card
   972→// ---------------------------------------------------------------------------
   973→
   974→function computeSubjectAverages(grades: GradeRecord[]): { subject: string; average: number }[] {
   975→  const subjectMap = new Map<string, { totalWeighted: number; totalWeight: number }>();
   976→  for (const grade of grades) {
   977→    const existing = subjectMap.get(grade.subject) ?? { totalWeighted: 0, totalWeight: 0 };
   978→    const pct = (grade.score / grade.maxScore) * 100;
   979→    subjectMap.set(grade.subject, {
   980→      totalWeighted: existing.totalWeighted + (pct * grade.weight),
   981→      totalWeight: existing.totalWeight + grade.weight,
   982→    });
   983→  }
   984→  const result: { subject: string; average: number }[] = [];
   985→  subjectMap.forEach((val, key) => {
   986→    result.push({
   987→      subject: key,
   988→      average: val.totalWeight > 0 ? Math.round(val.totalWeighted / val.totalWeight) : 0,
   989→    });
   990→  });
   991→  return result.sort((a, b) => b.average - a.average);
   992→}
