   277→function getClassName(classId: string | null, classes: { id: string; name: string }[]): string {
   278→  if (!classId) return '—';
   279→  const found = classes.find((c) => c.id === classId);
   280→  if (found) return found.name;
   281→  const legacy = CLASS_OPTIONS.find((c) => c.value === classId);
   282→  return legacy ? legacy.label : '—';
   283→}

   285→/** Highlight matching text in amber color */
   286→function HighlightText({ text, query }: { text: string; query: string }) {
   287→  if (!query.trim()) return <>{text}</>;
   288→  const lowerText = text.toLowerCase();
   289→  const lowerQuery = query.trim().toLowerCase();
   290→  const index = lowerText.indexOf(lowerQuery);
   291→  if (index === -1) return <>{text}</>;
   292→  return (
   293→    <>
   294→      {text.slice(0, index)}
   295→      <span className="text-[#D4A574] font-bold">{text.slice(index, index + query.trim().length)}</span>
   296→      {text.slice(index + query.trim().length)}
   297→    </>
   298→  );
   299→}

   447→function StudentFormContent({
   448→  editingStudent,
   449→  onClose,
   450→}: {
   451→  editingStudent: Student | null;
   452→  onClose: () => void;
   453→}) {
   454→  const { addStudent, updateStudent, classes: storeClasses } = useAppStore();
   455→  const isEditing = !!editingStudent;
   456→
   457→  // Dynamic class options from store, with fallback to static for legacy data
   458→  const classOptions = useMemo(() => {
   459→    if (storeClasses.length > 0) {
   460→      return storeClasses.map((c) => ({ value: c.id, label: c.name }));
   461→    }
   462→    return CLASS_OPTIONS;
   463→  }, [storeClasses]);
   464→
   465→  // Form state – initialised from props on mount
   466→  const [name, setName] = useState(editingStudent?.name ?? '');
   467→  const [studentId, setStudentId] = useState(editingStudent?.studentId ?? '');
   468→  const [classId, setClassId] = useState<string>(editingStudent?.classId ?? '');
   469→  const [status, setStatus] = useState<Student['status']>(editingStudent?.status ?? 'active');
   470→  const [notes, setNotes] = useState(editingStudent?.notes ?? '');
   471→  const [psychBackground, setPsychBackground] = useState(editingStudent?.psychBackground ?? '');
   472→
   473→  const handleSubmit = (e: React.FormEvent) => {
   474→    e.preventDefault();
   475→
   476→    if (!name.trim() || !studentId.trim()) return;
   477→
   478→    const now = new Date().toISOString();
   479→    // Convert empty string to null for classId (Student type expects string | null)
   480→    const resolvedClassId = classId || null;
   481→
   482→    if (isEditing && editingStudent) {
   483→      updateStudent(editingStudent.id, {
   484→        name: name.trim(),
   485→        studentId: studentId.trim(),
   486→        classId: resolvedClassId,
   487→        status,
   488→        notes: notes.trim(),
   489→        psychBackground: psychBackground.trim(),
   490→        updatedAt: now,
   491→      });
   492→      toast.success('تم تحديث بيانات الطالب');
   493→    } else {
   494→      const newStudent: Student = {
   495→        id: `std-${Date.now()}`,
   496→        name: name.trim(),
   497→        studentId: studentId.trim(),
   498→        classId: resolvedClassId,
   499→        status,
   500→        notes: notes.trim(),
   501→        psychBackground: psychBackground.trim(),
   502→        avatar: '',
   503→        createdAt: now,
   504→        updatedAt: now,
   505→      };
   506→      addStudent(newStudent);
   507→      toast.success('تم إضافة الطالب بنجاح');
   508→    }
   509→
   510→    onClose();
   511→  };
   512→
   513→  return (
   514→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   515→      {/* Name */}
   516→      <div className="flex flex-col gap-2">
   517→        <Label className="text-slate-300 text-sm">اسم الطالب الكامل</Label>
   518→        <Input
   519→          value={name}
   520→          onChange={(e) => setName(e.target.value)}
   521→          placeholder="مثال: أحمد محمد علي"
   522→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   523→          required
   524→        />
   525→      </div>
