    49→function formatTime(date: Date): string {
    50→  return date.toLocaleTimeString('ar-SA', {
    51→    hour: '2-digit',
    52→    minute: '2-digit',
    53→    second: '2-digit',
    54→    hour12: false,
    55→  });
    56→}
    57→
    58→function formatDate(date: Date): string {
    59→  return date.toLocaleDateString('ar-SA', {
    60→    weekday: 'long',
    61→    year: 'numeric',
    62→    month: 'long',
    63→    day: 'numeric',
    64→  });
    65→}

    71→function CameraFeedArea({ className, patternIndex }: { className?: string; patternIndex: number }) {
    72→  const patterns = [
    73→    // Static noise pattern
    74→    'repeating-linear-gradient(0deg, transparent, transparent 2px, rgba(255,255,255,0.015) 2px, rgba(255,255,255,0.015) 4px), repeating-linear-gradient(90deg, transparent, transparent 2px, rgba(255,255,255,0.01) 2px, rgba(255,255,255,0.01) 4px)',
    75→    // Grid lines pattern
    76→    'repeating-linear-gradient(0deg, transparent, transparent 24px, rgba(212,165,116,0.04) 24px, rgba(212,165,116,0.04) 25px), repeating-linear-gradient(90deg, transparent, transparent 24px, rgba(212,165,116,0.04) 24px, rgba(212,165,116,0.04) 25px)',
    77→    // Diagonal scan lines
    78→    'repeating-linear-gradient(45deg, transparent, transparent 3px, rgba(255,255,255,0.012) 3px, rgba(255,255,255,0.012) 6px)',
    79→    // Horizontal scan lines
    80→    'repeating-linear-gradient(0deg, transparent, transparent 1px, rgba(212,165,116,0.02) 1px, rgba(212,165,116,0.02) 2px)',
    81→    // Cross-hatch pattern
    82→    'repeating-linear-gradient(45deg, transparent, transparent 8px, rgba(255,255,255,0.01) 8px, rgba(255,255,255,0.01) 9px), repeating-linear-gradient(-45deg, transparent, transparent 8px, rgba(255,255,255,0.01) 8px, rgba(255,255,255,0.01) 9px)',
    83→    // Fine grid
    84→    'repeating-linear-gradient(0deg, transparent, transparent 16px, rgba(212,165,116,0.03) 16px, rgba(212,165,116,0.03) 17px), repeating-linear-gradient(90deg, transparent, transparent 16px, rgba(212,165,116,0.03) 16px, rgba(212,165,116,0.03) 17px)',
    85→    // Dot matrix
    86→    'radial-gradient(circle, rgba(255,255,255,0.02) 1px, transparent 1px)',
    87→    // Vignette with scan
    88→    'repeating-linear-gradient(0deg, transparent, transparent 2px, rgba(255,255,255,0.008) 2px, rgba(255,255,255,0.008) 3px)',
    89→  ];
    90→
    91→  const baseColors = [
    92→    'rgba(10, 10, 18, 1)',
    93→    'rgba(12, 12, 22, 1)',
    94→    'rgba(8, 10, 16, 1)',
    95→    'rgba(14, 12, 20, 1)',
    96→    'rgba(10, 14, 18, 1)',
    97→    'rgba(12, 8, 16, 1)',
    98→    'rgba(16, 12, 10, 1)',
    99→    'rgba(10, 16, 14, 1)',
   100→    'rgba(14, 10, 12, 1)',
   101→  ];
   102→
   103→  return (
   104→    <div
   105→      className={className}
   106→      style={{
   107→        background: `${baseColors[patternIndex % baseColors.length]}, ${patterns[patternIndex % patterns.length]}`,
   108→        backgroundSize: patternIndex === 6 ? '8px 8px' : undefined,
   109→      }}
   110→    />
   111→  );
   112→}

   290→export default function ClassroomCamera() {
   291→  const { classes, teachers } = useAppStore();
   292→
   293→  // ── State ──
   294→  const [viewMode, setViewMode] = useState<ViewMode>('grid-2x2');
   295→  const [selectedCameraId, setSelectedCameraId] = useState<string | null>(null);
   296→  const [currentTimestamp, setCurrentTimestamp] = useState<Date>(new Date());
   297→  const [isRefreshing, setIsRefreshing] = useState(false);
   298→
   299→  // ── Live timestamp update ──
   300→  useEffect(() => {
   301→    const interval = setInterval(() => {
   302→      setCurrentTimestamp(new Date());
   303→    }, 1000);
   304→    return () => clearInterval(interval);
   305→  }, []);
   306→
   307→  // ── Build camera feeds from store data ──
   308→  const cameraFeeds: CameraFeed[] = useMemo(() => {
   309→    // If store has classes, use them; otherwise use defaults
   310→    const classList = classes.length > 0 ? classes : defaultClasses;
   311→
   312→    return classList.map((cls, index) => {
   313→      // Find teacher for this class
   314→      const classTeacher = teachers.find((t) =>
   315→        t.status === 'active' && cls.teacherNames?.includes(t.name)
   316→      );
   317→      // Fallback teacher
   318→      const teacherName = classTeacher?.name || cls.teacherNames?.[0] || defaultTeachers[index % defaultTeachers.length].name;
   319→
   320→      return {
   321→        id: `cam-${cls.id}`,
   322→        classId: cls.id,
   323→        className: cls.name,
   324→        grade: cls.grade,
   325→        room: cls.room || `ف-${index + 1}`,
   326→        teacherName,
   327→        studentCount: cls.studentCount || Math.floor(Math.random() * 15) + 20,
   328→        status: (index % 5 === 4 ? 'offline' : index % 7 === 0 ? 'recording' : 'online') as CameraStatus,
   329→        sessionActive: index % 3 !== 2,
   330→        lastMotion: `منذ ${Math.floor(Math.random() * 30) + 1} ثانية`,
   331→      };
   332→    });
   333→  }, [classes, teachers]);
   334→
   335→  // ── Handlers ──
   336→  const handleCameraClick = useCallback((feedId: string) => {
   337→    if (selectedCameraId === feedId) {
   338→      setSelectedCameraId(null);
   339→    } else {
   340→      setSelectedCameraId(feedId);
   341→      setViewMode('single');
   342→    }
   343→  }, [selectedCameraId]);
   344→
   345→  const handleBackToGrid = useCallback(() => {
   346→    setSelectedCameraId(null);
   347→    setViewMode('grid-2x2');
   348→  }, []);
   349→
   350→  const handleRefresh = useCallback(() => {
   351→    setIsRefreshing(true);
   352→    setTimeout(() => setIsRefreshing(false), 1000);
   353→  }, []);
   354→
   355→  // ── Grid layout classes ──
   356→  const gridClasses = useMemo(() => {
   357→    switch (viewMode) {
   358→      case 'grid-3x3':
   359→        return 'grid-cols-1 sm:grid-cols-2 lg:grid-cols-3';
   360→      case 'grid-2x2':
   361→        return 'grid-cols-1 sm:grid-cols-2';
   362→      case 'single':
   363→        return 'grid-cols-1';
   364→      default:
   365→        return 'grid-cols-1 sm:grid-cols-2';
   366→    }
   367→  }, [viewMode]);
   368→
   369→  // ── Visible feeds ──
   370→  const visibleFeeds = useMemo(() => {
   371→    if (viewMode === 'single' && selectedCameraId) {
   372→      return cameraFeeds.filter((f) => f.id === selectedCameraId);
   373→    }
   374→    if (viewMode === 'grid-3x3') {
   375→      return cameraFeeds.slice(0, 9);
   376→    }
   377→    return cameraFeeds.slice(0, 4);
   378→  }, [viewMode, selectedCameraId, cameraFeeds]);
   379→
   380→  // ── Stats ──
   381→  const onlineCount = cameraFeeds.filter((f) => f.status !== 'offline').length;
   382→  const activeSessionCount = cameraFeeds.filter((f) => f.sessionActive).length;
   383→  const totalStudents = cameraFeeds.reduce((sum, f) => sum + f.studentCount, 0);
   384→
   385→  return (
   386→    <div dir="rtl" className="space-y-6">
   387→      {/* ── Header ── */}
   388→      <div className="glass-card-elevated rounded-2xl p-6">
   389→        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-4">
   390→          {/* Title section */}
   391→          <div className="flex items-center gap-4">
   392→            <div
   393→              className="flex items-center justify-center h-12 w-12 rounded-xl shadow-lg"
   394→              style={{
   395→                background: 'linear-gradient(135deg, #D4A574 0%, #B07D4F 100%)',
   396→                boxShadow: '0 4px 20px rgba(212,165,116,0.25)',
   397→              }}
   398→            >
   399→              <Video size={24} className="text-white" />
   400→            </div>
   401→            <div>
   402→              <h2 className="text-xl font-bold text-slate-100">مراقبة الفصول</h2>

   404→                {formatDate(currentTimestamp)} — {formatTime(currentTimestamp)}

   406→            </div>
   407→          </div>
   408→
   409→          {/* View Mode Toggle + Actions */}
   410→          <div className="flex items-center gap-2 flex-wrap">
   411→            {/* View mode buttons */}
   412→            <div className="flex items-center gap-1 bg-white/[0.04] rounded-lg p-1 border border-white/[0.06]">
   413→              <button

   415→                className={`
   416→                  flex items-center gap-1.5 px-3 py-1.5 rounded-md text-xs font-medium
   417→                  transition-all duration-200
   418→                  ${viewMode === 'grid-2x2'
   419→                    ? 'bg-[#D4A574]/15 text-[#D4A574] shadow-sm'

   503→              <p className="text-[10px] text-slate-500">جلسة نشطة</p>
   504→            </div>
   505→          </div>
   506→
   507→          <div className="flex items-center gap-2.5 bg-white/[0.03] rounded-lg px-3 py-2.5 border border-white/[0.04]">
   508→            <div className="flex items-center justify-center h-8 w-8 rounded-lg bg-amber-500/10">
   509→              <Users size={16} className="text-amber-400" />
   510→            </div>
   511→            <div>
   512→              <p className="text-lg font-bold text-amber-400 tabular-nums">{totalStudents}</p>
   513→              <p className="text-[10px] text-slate-500">طالب مراقب</p>
   514→            </div>
   515→          </div>
   516→        </div>
   517→      </div>
   518→
   519→      {/* ── Single View: Back button + Classroom selector ── */}
   520→      {viewMode === 'single' && (
   521→        <div className="flex items-center gap-3 flex-wrap">
   522→          <button
   523→            onClick={handleBackToGrid}
   524→            className="flex items-center gap-2 px-4 py-2 rounded-lg bg-white/[0.04] border border-white/[0.06]
   525→              text-sm text-slate-300 transition-all duration-200
   526→              hover:bg-[#D4A574]/10 hover:text-[#D4A574] hover:border-[#D4A574]/20"
   527→          >
   528→            <ChevronLeft size={16} />
   529→            <span>العودة للشبكة</span>
   530→          </button>
   531→
   532→          {/* Classroom quick-select dropdown */}
   533→          <div className="flex items-center gap-2 overflow-x-auto pb-1">
   534→            {cameraFeeds.map((feed) => (
   535→              <button
   536→                key={feed.id}
   537→                onClick={() => setSelectedCameraId(feed.id)}
   538→                className={`
   539→                  shrink-0 px-3 py-1.5 rounded-lg text-xs font-medium
   540→                  transition-all duration-200 border
   541→                  ${selectedCameraId === feed.id
   542→                    ? 'bg-[#D4A574]/15 text-[#D4A574] border-[#D4A574]/30'
   543→                    : 'bg-white/[0.03] text-slate-400 border-white/[0.06] hover:text-slate-300 hover:bg-white/[0.06]'
   544→                  }
   545→                `}
   546→              >
   547→                {feed.className}
   548→              </button>
   549→            ))}
   550→          </div>
   551→        </div>
   552→      )}
   553→
   554→      {/* ── Camera Grid ── */}
   555→      <div className={`grid gap-4 ${gridClasses}`}>
   556→        {visibleFeeds.map((feed) => (
   557→          <CameraView
   558→            key={feed.id}
   559→            feed={feed}
   560→            isExpanded={viewMode === 'single' && selectedCameraId === feed.id}
   561→            onClick={() => handleCameraClick(feed.id)}
   562→            currentTimestamp={currentTimestamp}
   563→          />
   564→        ))}
   565→      </div>
   566→
   567→      {/* ── Expanded Camera Details (shown in single view) ── */}
   568→      {viewMode === 'single' && selectedCameraId && (
   569→        <div className="glass-card rounded-2xl p-6 space-y-4">
   570→          {(() => {
   571→            const feed = cameraFeeds.find((f) => f.id === selectedCameraId);
   572→            if (!feed) return null;
   573→            return (
   574→              <>
   575→                <h3 className="text-base font-bold text-slate-100 flex items-center gap-2">
   576→                  <Camera size={18} className="text-[#D4A574]" />
   577→                  تفاصيل الكاميرا — {feed.className}
   578→                </h3>
   579→                <div className="grid grid-cols-2 sm:grid-cols-4 gap-4">
   580→                  <DetailItem label="الفصل" value={feed.className} />
   581→                  <DetailItem label="المرحلة" value={feed.grade} />
   582→                  <DetailItem label="القاعة" value={feed.room} />
   583→                  <DetailItem label="المعلم" value={feed.teacherName} />
   584→                  <DetailItem label="عدد الطلاب" value={`${feed.studentCount} طالب`} />
   585→                  <DetailItem label="الحالة" value={feed.status === 'online' ? 'متصل' : feed.status === 'recording' ? 'يسجّل' : 'غير متصل'} statusColor={feed.status === 'online' ? 'emerald' : feed.status === 'recording' ? 'amber' : 'slate'} />
   586→                  <DetailItem label="الجلسة" value={feed.sessionActive ? 'نشطة' : 'غير نشطة'} statusColor={feed.sessionActive ? 'emerald' : 'slate'} />
   587→                  <DetailItem label="آخر حركة" value={feed.lastMotion} />
   588→                </div>
   589→              </>
   590→            );
   591→          })()}
   592→        </div>
   593→      )}
   594→
   595→      {/* ── Offline Cameras Warning ── */}
   596→      {cameraFeeds.some((f) => f.status === 'offline') && (
   597→        <div className="flex items-center gap-3 bg-amber-500/10 border border-amber-500/20 rounded-xl p-4">
   598→          <WifiOff size={18} className="text-amber-400 shrink-0" />
   599→          <div>
   600→            <p className="text-sm font-medium text-amber-300">
   601→              {cameraFeeds.filter((f) => f.status === 'offline').length} كاميرا غير متصلة
   602→            </p>
   603→            <p className="text-xs text-amber-400/70 mt-0.5">
   604→              يرجى التحقق من اتصال الكاميرات في الفصول المتأثرة
   605→            </p>
   606→          </div>
   607→        </div>
   608→      )}
   609→    </div>
   610→  );
   611→}

   636→        {value}
   637→      </p>
   638→    </div>
   639→  );
