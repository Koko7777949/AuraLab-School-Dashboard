   193→function formatDate(dateStr: string): string {
   194→  return new Date(dateStr).toLocaleDateString('ar-SA', {
   195→    year: 'numeric',
   196→    month: 'short',
   197→    day: 'numeric',
   198→  });
   199→}
   200→
   201→function formatTime(dateStr: string): string {
   202→  return new Date(dateStr).toLocaleTimeString('ar-SA', {
   203→    hour: '2-digit',
   204→    minute: '2-digit',
   205→  });
   206→}
   207→
   208→function getDateGroup(dateStr: string): string {
   209→  const date = new Date(dateStr);
   210→  const today = new Date();
   211→  const todayStart = new Date(today.getFullYear(), today.getMonth(), today.getDate());
   212→  const yesterdayStart = new Date(todayStart);
   213→  yesterdayStart.setDate(yesterdayStart.getDate() - 1);
   214→  const weekStart = new Date(todayStart);
   215→  weekStart.setDate(weekStart.getDate() - 7);
   216→
   217→  if (date >= todayStart) return 'اليوم';
   218→  if (date >= yesterdayStart) return 'أمس';
   219→  if (date >= weekStart) return 'هذا الأسبوع';
   220→  return 'أقدم';
   221→}

   231→function exportToCSV(records: BehaviorRecord[]) {
   232→  const headers = ['اسم الطالب', 'الفصل', 'التاريخ', 'النوع', 'الفئة', 'الشدة', 'الوصف', 'الإجراء', 'النقاط', 'بواسطة'];
   233→  const rows = records.map((r) => [
   234→    r.studentName || '',
   235→    r.className || '',
   236→    formatDate(r.date),
   237→    r.type === 'positive' ? 'إيجابي' : 'سلبي',
   238→    r.category,
   239→    SEVERITY_LABELS[r.severity],
   240→    r.description,
   241→    r.action,
   242→    r.points.toString(),
   243→    r.recordedBy,
   244→  ]);
   245→
   246→  const BOM = '\uFEFF';
   247→  const csv = BOM + [headers, ...rows].map((row) => row.map((cell) => `"${cell}"`).join(',')).join('\n');
   248→  const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
   249→  const url = URL.createObjectURL(blob);
   250→  const link = document.createElement('a');
   251→  link.href = url;
   252→  link.download = `behavior_records_${new Date().toISOString().split('T')[0]}.csv`;
   253→  link.click();
   254→  URL.revokeObjectURL(url);
   255→  toast.success('تم تصدير البيانات بنجاح');
   256→}

   876→function StudentBehaviorSummary({ records }: { records: BehaviorRecord[] }) {
   877→  const positiveByStudent = useMemo(() => {
   878→    const map = new Map<string, { name: string; total: number; count: number }>();
   879→    for (const r of records.filter((r) => r.type === 'positive')) {
   880→      const key = r.studentId;
   881→      if (!map.has(key)) {
   882→        map.set(key, { name: r.studentName || 'غير معروف', total: 0, count: 0 });
   883→      }
   884→      const entry = map.get(key)!;
   885→      entry.total += r.points;
   886→      entry.count += 1;
   887→    }
   888→    return Array.from(map.values())
   889→      .sort((a, b) => b.total - a.total)
   890→      .slice(0, 5);
   891→  }, [records]);
   892→
   893→  const negativeByStudent = useMemo(() => {
   894→    const map = new Map<string, { name: string; total: number; count: number }>();
   895→    for (const r of records.filter((r) => r.type === 'negative')) {
   896→      const key = r.studentId;
   897→      if (!map.has(key)) {
   898→        map.set(key, { name: r.studentName || 'غير معروف', total: 0, count: 0 });
   899→      }
   900→      const entry = map.get(key)!;
   901→      entry.total += Math.abs(r.points);
   902→      entry.count += 1;
   903→    }
   904→    return Array.from(map.values())
   905→      .sort((a, b) => b.total - a.total)
   906→      .slice(0, 5);
   907→  }, [records]);
   908→
   909→  const maxPositive = positiveByStudent.length > 0 ? positiveByStudent[0].total : 1;
   910→  const maxNegative = negativeByStudent.length > 0 ? negativeByStudent[0].total : 1;
   911→
   912→  if (positiveByStudent.length === 0 && negativeByStudent.length === 0) return null;
   913→
   914→  return (
   915→    <div
   916→      className="rounded-xl p-5"
   917→      style={{
   918→        backgroundColor: 'rgba(255,255,255,0.02)',
   919→        border: '1px solid rgba(255,255,255,0.06)',
   920→      }}
   921→    >
   922→      <h3 className="text-sm font-semibold text-slate-300 mb-4 flex items-center gap-2">
   923→        <Star className="h-4 w-4 text-[#D4A574]" />
   924→        ملخص سلوك الطلاب
   925→      </h3>
   926→
   927→      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
   928→        {/* Top Positive */}
   929→        {positiveByStudent.length > 0 && (
   930→          <div>
   931→            <h4 className="text-xs font-medium text-emerald-400 mb-3 flex items-center gap-1.5">
   932→              <TrendingUp className="h-3.5 w-3.5" />
   933→              الأكثر إيجابية
   934→            </h4>
   935→            <div className="flex flex-col gap-2">
   936→              {positiveByStudent.map((s, i) => (
   937→                <div key={s.name} className="flex items-center gap-3">
   938→                  <span className="text-xs text-muted-foreground w-4 tabular-nums">{i + 1}.</span>
   939→                  <div className="flex-1 min-w-0">
   940→                    <div className="flex items-center justify-between mb-1">
   941→                      <span className="text-xs text-slate-300 truncate">{s.name}</span>
   942→                      <span className="text-xs font-bold text-emerald-400 tabular-nums">+{s.total}</span>
   943→                    </div>
   944→                    <div className="h-1.5 w-full rounded-full bg-white/[0.04] overflow-hidden">
   945→                      <div
   946→                        className="h-full rounded-full bg-emerald-500/50 transition-all duration-500"
   947→                        style={{ width: `${Math.max(8, (s.total / maxPositive) * 100)}%` }}
   948→                      />
   949→                    </div>
   950→                  </div>
   951→                </div>
   952→              ))}
   953→            </div>
   954→          </div>
   955→        )}
   956→
   957→        {/* Top Negative */}
   958→        {negativeByStudent.length > 0 && (
   959→          <div>
   960→            <h4 className="text-xs font-medium text-rose-400 mb-3 flex items-center gap-1.5">
   961→              <TrendingDown className="h-3.5 w-3.5" />
   962→              الأكثر سلبية
   963→            </h4>
   964→            <div className="flex flex-col gap-2">
   965→              {negativeByStudent.map((s, i) => (
   966→                <div key={s.name} className="flex items-center gap-3">
   967→                  <span className="text-xs text-muted-foreground w-4 tabular-nums">{i + 1}.</span>
   968→                  <div className="flex-1 min-w-0">
   969→                    <div className="flex items-center justify-between mb-1">
   970→                      <span className="text-xs text-slate-300 truncate">{s.name}</span>

   974→                      <div
   975→                        className="h-full rounded-full bg-rose-500/50 transition-all duration-500"

  1161→function Pagination({
  1162→  currentPage,
  1163→  totalPages,
  1164→  onPageChange,
  1165→}: {
  1166→  currentPage: number;
  1167→  totalPages: number;
  1168→  onPageChange: (page: number) => void;
  1169→}) {
  1170→  if (totalPages <= 1) return null;
  1171→
  1172→  const getPages = () => {
  1173→    const pages: (number | '...')[] = [];
  1174→    if (totalPages <= 7) {
  1175→      for (let i = 1; i <= totalPages; i++) pages.push(i);
  1176→    } else {
  1177→      pages.push(1);
  1178→      if (currentPage > 3) pages.push('...');
  1179→      for (let i = Math.max(2, currentPage - 1); i <= Math.min(totalPages - 1, currentPage + 1); i++) {
  1180→        pages.push(i);
  1181→      }
  1182→      if (currentPage < totalPages - 2) pages.push('...');
  1183→      pages.push(totalPages);
  1184→    }
  1185→    return pages;
  1186→  };
  1187→
  1188→  return (
  1189→    <div className="flex items-center justify-center gap-1.5 mt-6">
  1190→      <button
  1191→        onClick={() => onPageChange(currentPage - 1)}
  1192→        disabled={currentPage === 1}
  1193→        className="flex items-center justify-center h-9 w-9 rounded-lg border border-white/[0.06] text-slate-400 hover:text-white hover:bg-white/[0.05] disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
  1194→      >
  1195→        <ChevronRight className="h-4 w-4" />
  1196→      </button>
  1197→      {getPages().map((page, i) =>
  1198→        page === '...' ? (
  1199→          <span key={`dots-${i}`} className="h-9 w-9 flex items-center justify-center text-sm text-muted-foreground">
  1200→            ...
  1201→          </span>
  1202→        ) : (
  1203→          <button
  1204→            key={page}
  1205→            onClick={() => onPageChange(page as number)}
  1206→            className={cn(
  1207→              'flex items-center justify-center h-9 w-9 rounded-lg text-sm font-medium transition-all duration-200',
  1208→              currentPage === page
  1209→                ? 'bg-[#D4A574] text-slate-900 font-bold'
  1210→                : 'border border-white/[0.06] text-slate-400 hover:text-white hover:bg-white/[0.05]'
  1211→            )}
  1212→          >
  1213→            {page}
  1214→          </button>
  1215→        )
  1216→      )}
  1217→      <button
  1218→        onClick={() => onPageChange(currentPage + 1)}
  1219→        disabled={currentPage === totalPages}
  1220→        className="flex items-center justify-center h-9 w-9 rounded-lg border border-white/[0.06] text-slate-400 hover:text-white hover:bg-white/[0.05] disabled:opacity-30 disabled:cursor-not-allowed transition-colors"
  1221→      >
  1222→        <ChevronLeft className="h-4 w-4" />
  1223→      </button>
  1224→    </div>
  1225→  );
  1226→}
