   147→function getCapacityColor(percentage: number): { bar: string; text: string } {
   148→  if (percentage >= 90) {
   149→    return { bar: 'bg-rose-500', text: 'text-rose-400' };
   150→  }
   151→  if (percentage >= 70) {
   152→    return { bar: 'bg-amber-500', text: 'text-amber-400' };
   153→  }
   154→  return { bar: 'bg-emerald-500', text: 'text-emerald-400' };
   155→}

   189→function ClassCard({
   190→  cls,
   191→  onEdit,
   192→  onDelete,
   193→  onQuickRoomEdit,
   194→}: {
   195→  cls: ClassRoom;
   196→  onEdit: (c: ClassRoom) => void;
   197→  onDelete: (c: ClassRoom) => void;
   198→  onQuickRoomEdit: (c: ClassRoom, newRoom: string) => void;
   199→}) {
   200→  const [editingRoom, setEditingRoom] = useState(false);
   201→  const [roomValue, setRoomValue] = useState(cls.room);
   202→  const studentCount = cls.studentCount ?? 0;
   203→  const percentage = Math.min(100, Math.round((studentCount / MAX_STUDENTS_PER_CLASS) * 100));
   204→  const capacityColor = getCapacityColor(percentage);
   205→
   206→  return (
   207→    <div
   208→      className="group card-interactive rounded-xl p-5 transition-all duration-200 hover:bg-white/[0.04]"
   209→      style={{
   210→        backgroundColor: 'rgba(255,255,255,0.02)',
   211→        border: '1px solid rgba(255,255,255,0.06)',
   212→      }}
   213→    >
   214→      {/* Top row: icon + name + actions */}
   215→      <div className="flex items-start justify-between gap-3 mb-4">
   216→        <div className="flex items-center gap-3 min-w-0">
   217→          <div className="flex items-center justify-center h-11 w-11 rounded-xl bg-gradient-to-br from-[#D4A574]/20 to-[#D4A574]/5 shrink-0 border border-[#D4A574]/10">
   218→            <School className="h-5 w-5 text-[#D4A574]" />
   219→          </div>
   220→          <div className="min-w-0">
   221→            <h3 className="text-sm font-semibold text-slate-100 truncate">
   222→              {cls.name}
   223→            </h3>
   224→            <div className="flex items-center gap-1 mt-0.5">
   225→              <MapPin className="h-3 w-3 text-muted-foreground" />
   226→              {editingRoom ? (
   227→                <input
   228→                  autoFocus
   229→                  value={roomValue}
   230→                  onChange={(e) => setRoomValue(e.target.value)}
   231→                  onBlur={() => {
   232→                    if (roomValue.trim() && roomValue.trim() !== cls.room) {
   233→                      onQuickRoomEdit(cls, roomValue.trim());
   234→                    }
   235→                    setEditingRoom(false);
   236→                  }}
   237→                  onKeyDown={(e) => {
   238→                    if (e.key === 'Enter') {
   239→                      if (roomValue.trim() && roomValue.trim() !== cls.room) {
   240→                        onQuickRoomEdit(cls, roomValue.trim());
   241→                      }
   242→                      setEditingRoom(false);
   243→                    }
   244→                    if (e.key === 'Escape') {
   245→                      setRoomValue(cls.room);
   246→                      setEditingRoom(false);
   247→                    }
   248→                  }}
   249→                  className="h-5 w-24 text-xs bg-white/[0.06] border border-[#D4A574]/40 text-slate-200 rounded px-1.5 outline-none"
   250→                  dir="rtl"
   251→                />
   252→              ) : (
   253→                <button
   254→                  onClick={() => { setRoomValue(cls.room); setEditingRoom(true); }}
   255→                  className="text-xs text-muted-foreground hover:text-[#D4A574] transition-colors flex items-center gap-1"
   256→                  title="انقر لتعديل القاعة"
   257→                >
   258→                  {cls.room}
   259→                  <Pencil className="h-2.5 w-2.5 opacity-0 group-hover:opacity-60 transition-opacity" />
   260→                </button>
   261→              )}
   262→            </div>
   263→          </div>
   264→        </div>
   265→
   266→        {/* Action buttons */}
   267→        <div className="flex items-center gap-1 shrink-0 opacity-0 group-hover:opacity-100 transition-opacity duration-200">
   268→          <Button
   269→            variant="ghost"
   270→            size="icon"
   271→            onClick={() => onEdit(cls)}
   272→            className="h-8 w-8 text-slate-400 hover:text-amber-400 hover:bg-amber-400/10 rounded-lg"
   273→            title="تعديل"
   274→          >
   275→            <Pencil className="h-4 w-4" />
   276→          </Button>
   277→          <Button
   278→            variant="ghost"
   279→            size="icon"
   280→            onClick={() => onDelete(cls)}
   281→            className="h-8 w-8 text-slate-400 hover:text-rose-400 hover:bg-rose-400/10 rounded-lg"
   282→            title="حذف"
   283→          >
   284→            <Trash2 className="h-4 w-4" />
   285→          </Button>
   286→        </div>
   287→      </div>
   288→
   289→      {/* Grade badge */}
   290→      <div className="mb-4">
   291→        <Badge
   292→          variant="outline"
   293→          className="text-xs font-medium px-2.5 py-0.5 rounded-full bg-[#D4A574]/10 text-[#D4A574] border-[#D4A574]/20"
   294→        >
   295→          {cls.grade}
   296→        </Badge>
   297→      </div>
   298→
   299→      {/* Student count + capacity bar */}
   300→      <div className="mb-4">
   301→        <div className="flex items-center justify-between mb-2">
   302→          <span className="text-xs text-muted-foreground flex items-center gap-1.5">
   303→            <Users className="h-3.5 w-3.5 text-slate-500" />
   304→            عدد الطلاب
   305→          </span>
   306→          <span className={cn('text-xs font-semibold tabular-nums', capacityColor.text)}>
   307→            {studentCount} / {MAX_STUDENTS_PER_CLASS}
   308→          </span>
   309→        </div>
   310→        <div className="relative h-2 w-full overflow-hidden rounded-full bg-white/[0.06]">
   311→          <div
   312→            className={cn('h-full rounded-full transition-all duration-500', capacityColor.bar)}
   313→            style={{ width: `${percentage}%` }}
   314→          />
   315→        </div>
   316→      </div>
   317→
   318→      {/* Assigned teachers */}
   319→      {cls.teacherNames && cls.teacherNames.length > 0 && (
   320→        <div className="pt-3 border-t border-white/[0.05]">
   321→          <p className="text-xs text-muted-foreground mb-2 flex items-center gap-1.5">
   322→            <UserCheck className="h-3.5 w-3.5 text-slate-500" />
   323→            المعلمون المسندون
   324→          </p>
   325→          <div className="flex flex-wrap gap-1.5">
   326→            {cls.teacherNames.slice(0, 3).map((teacher, idx) => (
   327→              <Badge
   328→                key={idx}
   329→                variant="outline"
   330→                className="text-[10px] font-medium px-2 py-0 rounded-full bg-sky-500/8 border-sky-500/15 text-sky-300"
   331→              >
   332→                {teacher}
   333→              </Badge>
   334→            ))}
   335→            {cls.teacherNames.length > 3 && (
   336→              <Badge
   337→                variant="outline"
   338→                className="text-[10px] font-medium px-2 py-0 rounded-full bg-white/[0.03] border-white/[0.08] text-slate-500"
   339→              >
   340→                +{cls.teacherNames.length - 3}
   341→              </Badge>
   342→            )}
   343→          </div>
   344→        </div>
   345→      )}
   346→    </div>
   347→  );
   348→}

   354→function ClassFormContent({
   355→  editingClass,
   356→  onClose,
   357→}: {
   358→  editingClass: ClassRoom | null;
   359→  onClose: () => void;
   360→}) {
   361→  const { addClass, updateClass } = useAppStore();
   362→  const isEditing = !!editingClass;
   363→
   364→  // Form state
   365→  const [name, setName] = useState(editingClass?.name ?? '');
   366→  const [grade, setGrade] = useState(editingClass?.grade ?? 'الأول');
   367→  const [room, setRoom] = useState(editingClass?.room ?? '');
   368→  const [notes, setNotes] = useState(editingClass?.notes ?? '');
   369→
   370→  const handleSubmit = (e: React.FormEvent) => {
   371→    e.preventDefault();
   372→    if (!name.trim() || !room.trim()) return;
   373→
   374→    const now = new Date().toISOString();
   375→
   376→    if (isEditing && editingClass) {
   377→      updateClass(editingClass.id, {
   378→        name: name.trim(),
   379→        grade,
   380→        room: room.trim(),
   381→        notes: notes.trim(),
   382→        updatedAt: now,
   383→      });
   384→      toast.success('تم تحديث بيانات الصف');
   385→    } else {
   386→      const newClass: ClassRoom = {
   387→        id: `cls-${Date.now()}`,
   388→        name: name.trim(),
   389→        grade,
   390→        room: room.trim(),
   391→        notes: notes.trim(),
   392→        createdAt: now,
   393→        updatedAt: now,
   394→        studentCount: 0,
   395→        teacherNames: [],
   396→      };
   397→      addClass(newClass);
   398→      toast.success('تم إضافة الصف بنجاح');
   399→    }
   400→
   401→    onClose();
   402→  };
   403→
   404→  return (
   405→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   406→      {/* Class name */}
   407→      <div className="flex flex-col gap-2">
   408→        <Label className="text-slate-300 text-sm">اسم الصف</Label>
   409→        <Input
   410→          value={name}
   411→          onChange={(e) => setName(e.target.value)}
   412→          placeholder="مثال: الصف الأول أ"
   413→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   414→          required
   415→        />
   416→      </div>
   417→
   418→      {/* Grade & Room row */}
   419→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   420→        {/* Grade dropdown */}
   421→        <div className="flex flex-col gap-2">
   422→          <Label className="text-slate-300 text-sm">المرحلة الدراسية</Label>
   423→          <Select value={grade} onValueChange={setGrade}>
   424→            <SelectTrigger className="w-full bg-white/[0.03] border-white/[0.08] text-slate-200 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20">
   425→              <SelectValue placeholder="اختر المرحلة" />
   426→            </SelectTrigger>
   427→            <SelectContent className="bg-[#1A1A24] border-white/[0.08]">
   428→              <SelectItem value="الأول" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   429→                الصف الأول
   430→              </SelectItem>
   431→              <SelectItem value="الثاني" className="text-slate-200 focus:bg-white/[0.05] focus:text-white">
   432→                الصف الثاني
   433→              </SelectItem>

   435→                الصف الثالث

   437→            </SelectContent>
   438→          </Select>

   440→
   441→        {/* Room */}

   557→        dataJson: JSON.stringify(cls),

   578→            تأكيد حذف الصف
   579→          </AlertDialogTitle>
   580→          <AlertDialogDescription className="text-muted-foreground leading-relaxed">
   581→            هل أنت متأكد من حذف الصف{' '}

   583→            <br />
   584→            لا يمكن التراجع عن هذا الإجراء. سيتم إلغاء ارتباط جميع الطلاب والمعلمين بهذا الصف.
   585→          </AlertDialogDescription>
   586→        </AlertDialogHeader>
   587→        <AlertDialogFooter className="flex-row gap-3 sm:justify-start mt-2">

   599→          </AlertDialogAction>
