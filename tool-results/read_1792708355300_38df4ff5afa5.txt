   284→   284→function ReportBarChart({
   285→   285→  data,
   286→   286→  height = 180,
   287→   287→}: {
   288→   288→  data: { label: string; value: number; color: string }[];
   289→   289→  height?: number;
   290→   290→}) {
   291→   291→  const maxValue = Math.max(...data.map((d) => d.value), 1);
   292→   292→  const barWidth = Math.min(40, Math.max(16, 200 / data.length));
   293→   293→  const gap = Math.max(4, (200 - barWidth * data.length) / (data.length + 1));
   294→   294→
   295→   295→  return (
   296→   296→    <svg
   297→   297→      viewBox={`0 0 200 ${height + 30}`}
   298→   298→      width="100%"
   299→   299→      height={height + 30}
   300→   300→      preserveAspectRatio="xMidYMid meet"
   301→   301→    >
   302→   302→      {/* Grid lines */}
   303→   303→      {[0, 25, 50, 75, 100].map((pct) => {
   304→   304→        const y = height - (pct / 100) * height + 5;
   305→   305→        return (
   306→   306→          <g key={pct}>
   307→   307→            <line
   308→   308→              x1="25"
   309→   309→              y1={y}
   310→   310→              x2="195"
   311→   311→              y2={y}
   312→   312→              stroke="rgba(255,255,255,0.05)"
   313→   313→              strokeDasharray="4 3"
   314→   314→            />
   315→   315→            <text
   316→   316→              x="20"
   317→   317→              y={y + 3}
   318→   318→              textAnchor="end"
   319→   319→              fill="rgba(255,255,255,0.25)"
   320→   320→              fontSize="7"
   321→   321→            >
   322→   322→              {pct}%
   323→   323→            </text>
   324→   324→          </g>
   325→   325→        );
   326→   326→      })}
   327→   327→
   328→   328→      {/* Bars */}
   329→   329→      {data.map((item, i) => {
   330→   330→        const barHeight = (item.value / maxValue) * height;
   331→   331→        const x = 30 + i * (barWidth + gap);
   332→   332→        const y = height - barHeight + 5;
   333→   333→
   334→   334→        return (
   335→   335→          <g key={item.label}>
   336→   336→            {/* Bar gradient */}
   337→   337→            <defs>
   338→   338→              <linearGradient id={`bar-grad-${i}`} x1="0" y1="0" x2="0" y2="1">
   339→   339→                <stop offset="0%" stopColor={item.color} stopOpacity="1" />
   340→   340→                <stop offset="100%" stopColor={item.color} stopOpacity="0.4" />
   341→   341→              </linearGradient>
   342→   342→            </defs>
   343→   343→            {/* Bar */}
   344→   344→            <rect
   345→   345→              x={x}
   346→   346→              y={y}
   347→   347→              width={barWidth}
   348→   348→              height={barHeight}
   349→   349→              rx="3"
   350→   350→              fill={`url(#bar-grad-${i})`}
   351→   351→              className="transition-all duration-700"
   352→   352→              style={{ animationDelay: `${i * 80}ms` }}
   353→   353→            />
   354→   354→            {/* Value on top */}
   355→   355→            <text
   356→   356→              x={x + barWidth / 2}
   357→   357→              y={y - 4}
   358→   358→              textAnchor="middle"
   359→   359→              fill="white"
   360→   360→              fontSize="8"
   361→   361→              fontWeight="600"
   362→   362→            >
   363→   363→              {item.value}
   364→   364→            </text>
   365→   365→            {/* Label below */}
   366→   366→            <text
   367→   367→              x={x + barWidth / 2}
   368→   368→              y={height + 22}
   369→   369→              textAnchor="middle"
   370→   370→              fill="rgba(255,255,255,0.4)"
   371→   371→              fontSize="7"
   372→   372→            >
   373→   373→              {item.label}
   374→   374→            </text>
   375→   375→          </g>
   376→   376→        );
   377→   377→      })}
   378→   378→    </svg>
   379→   379→  );
   380→   380→}

   382→   382→function ReportDonutChart({
   383→   383→  data,
   384→   384→  size = 160,
   385→   385→}: {
   386→   386→  data: { label: string; value: number; color: string }[];
   387→   387→  size?: number;
   388→   388→}) {
   389→   389→  const total = data.reduce((sum, d) => sum + d.value, 0);
   390→   390→  const strokeWidth = 24;
   391→   391→  const radius = (size - strokeWidth) / 2;
   392→   392→  const circumference = 2 * Math.PI * radius;
   393→   393→  const center = size / 2;
   394→   394→
   395→   395→  // Compute segment offsets immutably
   396→   396→  const segments = data.reduce(
   397→   397→    (acc, item) => {
   398→   398→      const pct = item.value / total;
   399→   399→      const dashLength = pct * circumference;
   400→   400→      const gapSize = 2; // small gap between segments
   401→   401→      const offset = acc.accumulated;
   402→   402→      return {
   403→   403→        segments: [
   404→   404→          ...acc.segments,
   405→   405→          { ...item, dashLength, gapSize, offset },
   406→   406→        ],
   407→   407→        accumulated: offset + dashLength + gapSize,
   408→   408→      };
   409→   409→    },
   410→   410→    { segments: [] as { label: string; value: number; color: string; dashLength: number; gapSize: number; offset: number }[], accumulated: 0 },
   411→   411→  ).segments;
   412→   412→
   413→   413→  return (
   414→   414→    <div className="flex flex-col items-center">
   415→   415→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
   416→   416→        {segments.map((seg, i) => (
   417→   417→          <circle
   418→   418→            key={seg.label}
   419→   419→            cx={center}
   420→   420→            cy={center}
   421→   421→            r={radius}
   422→   422→            fill="none"
   423→   423→            stroke={seg.color}
   424→   424→            strokeWidth={strokeWidth}
   425→   425→            strokeDasharray={`${seg.dashLength} ${circumference - seg.dashLength}`}
   426→   426→            strokeDashoffset={-seg.offset}
   427→   427→            strokeLinecap="round"
   428→   428→            className="transition-all duration-700"
   429→   429→            style={{ animationDelay: `${i * 100}ms` }}
   430→   430→          />
   431→   431→        ))}
   432→   432→        {/* Center text */}
   433→   433→        <text
   434→   434→          x={center}
   435→   435→          y={center - 6}
   436→   436→          textAnchor="middle"
   437→   437→          fill="white"
   438→   438→          fontSize="20"
   439→   439→          fontWeight="700"
   440→   440→        >
   441→   441→          {total}
   442→   442→        </text>
   443→   443→        <text
   444→   444→          x={center}
   445→   445→          y={center + 12}
   446→   446→          textAnchor="middle"
   447→   447→          fill="rgba(255,255,255,0.4)"
   448→   448→          fontSize="10"
   449→   449→        >
   450→   450→          إجمالي
   451→   451→        </text>
   452→   452→      </svg>
   453→   453→      {/* Legend */}
   454→   454→      <div className="flex flex-wrap items-center justify-center gap-3 mt-3">
   455→   455→        {data.map((item) => (
   456→   456→          <div key={item.label} className="flex items-center gap-1.5">
   457→   457→            <span
   458→   458→              className="h-2 w-2 rounded-full"
   459→   459→              style={{ backgroundColor: item.color }}
   460→   460→            />
   461→   461→            <span className="text-[10px] text-slate-400">

   471→   471→function ReportTrendChart({
   472→   472→  data,
   473→   473→  height = 80,
   474→   474→  width = 300,
   475→   475→}: {
   476→   476→  data: { label: string; value: number }[];
   477→   477→  height?: number;
   478→   478→  width?: number;
   479→   479→}) {
   480→   480→  const maxValue = Math.max(...data.map((d) => d.value), 1);
   481→   481→  const minValue = Math.min(...data.map((d) => d.value), 0);
   482→   482→  const range = maxValue - minValue || 1;
   483→   483→  const padding = { top: 10, bottom: 20, left: 30, right: 10 };
   484→   484→  const chartW = width - padding.left - padding.right;
   485→   485→  const chartH = height - padding.top - padding.bottom;
   486→   486→
   487→   487→  const points = data.map((d, i) => ({
   488→   488→    x: padding.left + (i / Math.max(data.length - 1, 1)) * chartW,
   489→   489→    y: padding.top + chartH - ((d.value - minValue) / range) * chartH,
   490→   490→    label: d.label,
   491→   491→    value: d.value,
   492→   492→  }));
   493→   493→
   494→   494→  const linePath =
   495→   495→    points.length > 1
   496→   496→      ? points.map((p, i) => `${i === 0 ? 'M' : 'L'} ${p.x} ${p.y}`).join(' ')
   497→   497→      : '';
   498→   498→
   499→   499→  const areaPath =
   500→   500→    points.length > 1
   501→   501→      ? `${linePath} L ${points[points.length - 1].x} ${padding.top + chartH} L ${points[0].x} ${padding.top + chartH} Z`
   502→   502→      : '';
   503→   503→
   504→   504→  return (
   505→   505→    <svg
   506→   506→      viewBox={`0 0 ${width} ${height}`}
   507→   507→      width="100%"
   508→   508→      height={height}
   509→   509→      preserveAspectRatio="xMidYMid meet"
   510→   510→    >
   511→   511→      <defs>
   512→   512→        <linearGradient id="trend-fill" x1="0" y1="0" x2="0" y2="1">
   513→   513→          <stop offset="0%" stopColor="#D4A574" stopOpacity="0.3" />
   514→   514→          <stop offset="100%" stopColor="#D4A574" stopOpacity="0" />
   515→   515→        </linearGradient>
   516→   516→      </defs>
   517→   517→
   518→   518→      {/* Grid */}
   519→   519→      {[0, 1, 2, 3].map((i) => {
   520→   520→        const y = padding.top + (i / 3) * chartH;
   521→   521→        return (
   522→   522→          <line
   523→   523→            key={i}
   524→   524→            x1={padding.left}
   525→   525→            y1={y}
   526→   526→            x2={width - padding.right}
   527→   527→            y2={y}
   528→   528→            stroke="rgba(255,255,255,0.04)"
   529→   529→            strokeDasharray="3 3"
   530→   530→          />
   531→   531→        );
   532→   532→      })}
   533→   533→
   534→   534→      {/* Area */}
   535→   535→      {areaPath && <path d={areaPath} fill="url(#trend-fill)" />}
   536→   536→
   537→   537→      {/* Line */}
   538→   538→      {linePath && (
   539→   539→        <path
   540→   540→          d={linePath}
   541→   541→          fill="none"
   542→   542→          stroke="#D4A574"
   543→   543→          strokeWidth="2"
   544→   544→          strokeLinecap="round"
   545→   545→          strokeLinejoin="round"
   546→   546→        />
   547→   547→      )}
   548→   548→
   549→   549→      {/* Dots & labels */}
   550→   550→      {points.map((p) => (
   551→   551→        <g key={p.label}>
   552→   552→          <circle cx={p.x} cy={p.y} r="3" fill="#D4A574" stroke="#1A1A24" strokeWidth="1.5" />
   553→   553→          <text
   554→   554→            x={p.x}
   555→   555→            y={p.y - 8}
   556→   556→            textAnchor="middle"
   557→   557→            fill="rgba(255,255,255,0.7)"
   558→   558→            fontSize="7"
   559→   559→            fontWeight="600"
   560→   560→          >
   561→   561→            {p.value}%
   562→   562→          </text>
   563→   563→          <text
   564→   564→            x={p.x}
   565→   565→            y={height - 4}
   566→   566→            textAnchor="middle"
   567→   567→            fill="rgba(255,255,255,0.3)"
   568→   568→            fontSize="7"
   569→   569→          >
   570→   570→            {p.label}
   571→   571→          </text>
   572→   572→        </g>
   573→   573→      ))}
   574→   574→    </svg>
   575→   575→  );
   576→   576→}

   582→   582→function AnimatedCounter({ target, duration = 800 }: { target: number; duration?: number }) {
   583→   583→  const [count, setCount] = useState(0);
   584→   584→  const rafRef = useRef(0);
   585→   585→
   586→   586→  useEffect(() => {
   587→   587→    const start = performance.now();
   588→   588→    const animate = (now: number) => {
   589→   589→      const elapsed = now - start;
   590→   590→      const progress = Math.min(elapsed / duration, 1);
   591→   591→      // Ease out cubic
   592→   592→      const eased = 1 - Math.pow(1 - progress, 3);
   593→   593→      setCount(Math.round(eased * target));
   594→   594→      if (progress < 1) {
   595→   595→        rafRef.current = requestAnimationFrame(animate);
   596→   596→      }
   597→   597→    };
   598→   598→    rafRef.current = requestAnimationFrame(animate);
   599→   599→    return () => cancelAnimationFrame(rafRef.current);
   600→   600→  }, [target, duration]);
   601→   601→
   602→   602→  return <span className="tabular-nums">{count}</span>;
   603→   603→}
