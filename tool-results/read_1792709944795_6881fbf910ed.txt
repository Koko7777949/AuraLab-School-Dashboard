   217→function getInitials(name: string): string {
   218→  const parts = name.trim().split(/\s+/);
   219→  if (parts.length >= 2) {
   220→    return parts[0].charAt(0) + parts[1].charAt(0);
   221→  }
   222→  return name.charAt(0);
   223→}

   401→function TeacherFormContent({
   402→  editingTeacher,
   403→  onClose,
   404→}: {
   405→  editingTeacher: Teacher | null;
   406→  onClose: () => void;
   407→}) {
   408→  const { addTeacher, updateTeacher } = useAppStore();
   409→  const isEditing = !!editingTeacher;
   410→
   411→  // Form state
   412→  const [name, setName] = useState(editingTeacher?.name ?? '');
   413→  const [subject, setSubject] = useState(editingTeacher?.subject ?? '');
   414→  const [email, setEmail] = useState(editingTeacher?.email ?? '');
   415→  const [phone, setPhone] = useState(editingTeacher?.phone ?? '');
   416→  const [permissions, setPermissions] = useState<Teacher['permissions']>(
   417→    editingTeacher?.permissions ?? 'teacher'
   418→  );
   419→  const [status, setStatus] = useState<Teacher['status']>(
   420→    editingTeacher?.status ?? 'active'
   421→  );
   422→
   423→  const handleSubmit = (e: React.FormEvent) => {
   424→    e.preventDefault();
   425→    if (!name.trim() || !subject.trim()) return;
   426→
   427→    const now = new Date().toISOString();
   428→
   429→    if (isEditing && editingTeacher) {
   430→      updateTeacher(editingTeacher.id, {
   431→        name: name.trim(),
   432→        subject: subject.trim(),
   433→        email: email.trim(),
   434→        phone: phone.trim(),
   435→        permissions,
   436→        status,
   437→        updatedAt: now,
   438→      });
   439→      toast.success('تم تحديث بيانات المعلم');
   440→    } else {
   441→      const newTeacher: Teacher = {
   442→        id: `tch-${Date.now()}`,
   443→        name: name.trim(),
   444→        subject: subject.trim(),
   445→        status,
   446→        permissions,
   447→        avatar: '',
   448→        email: email.trim(),
   449→        phone: phone.trim(),
   450→        createdAt: now,
   451→        updatedAt: now,
   452→        classNames: [],
   453→      };
   454→      addTeacher(newTeacher);
   455→      toast.success('تم إضافة المعلم بنجاح');
   456→    }
   457→
   458→    onClose();
   459→  };
   460→
   461→  return (
   462→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   463→      {/* Name */}
   464→      <div className="flex flex-col gap-2">
   465→        <Label className="text-slate-300 text-sm">اسم المعلم الكامل</Label>

   467→          value={name}
   468→          onChange={(e) => setName(e.target.value)}
   469→          placeholder="مثال: محمد بن عبدالله السالم"
   470→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   471→          required
   472→        />
   473→      </div>
   474→
   475→      {/* Subject */}
   476→      <div className="flex flex-col gap-2">
   477→        <Label className="text-slate-300 text-sm">المادة الدراسية</Label>
   478→        <Input
   479→          value={subject}
   480→          onChange={(e) => setSubject(e.target.value)}
   481→          placeholder="مثال: الرياضيات"
   482→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20"
   483→          required
   484→        />
   485→      </div>
   486→
   487→      {/* Email & Phone row */}
   488→      <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   489→        <div className="flex flex-col gap-2">
   490→          <Label className="text-slate-300 text-sm">البريد الإلكتروني</Label>

   622→/** Delete confirmation dialog */
   623→function DeleteTeacherDialog({
   624→  teacher,
   625→  open,
   626→  onOpenChange,
   627→}: {
   628→  teacher: Teacher | null;
   629→  open: boolean;
   630→  onOpenChange: (open: boolean) => void;
   631→}) {

   667→            <span className="font-semibold text-slate-200">{teacher?.name}</span>؟

   671→        </AlertDialogHeader>
   672→        <AlertDialogFooter className="flex-row gap-3 sm:justify-start mt-2">

   674→            className="border-white/[0.08] text-slate-300 hover:bg-white/[0.05] hover:text-white rounded-lg px-6 mt-0"

   684→          </AlertDialogAction>
   685→        </AlertDialogFooter>

   687→    </AlertDialog>
