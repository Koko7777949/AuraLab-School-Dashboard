   420→function AttendanceDistribution({ apiAttendance }: { apiAttendance: { rate: number; present: number; absent: number; late: number; excused: number; total: number } | null }) {
   421→  const total = apiAttendance?.total || 32;
   422→  const presentPct = apiAttendance && total > 0 ? Math.round((apiAttendance.present / total) * 100) : 85;
   423→  const latePct = apiAttendance && total > 0 ? Math.round((apiAttendance.late / total) * 100) : 8;
   424→  const absentPct = apiAttendance && total > 0 ? Math.round((apiAttendance.absent / total) * 100) : 7;
   425→
   426→  const localAttendanceData = [
   427→    { label: 'حاضر', percentage: presentPct, color: 'bg-emerald-500', textColor: 'text-emerald-400' },
   428→    { label: 'متأخر', percentage: latePct, color: 'bg-amber-500', textColor: 'text-amber-400' },
   429→    { label: 'غائب', percentage: absentPct, color: 'bg-rose-500', textColor: 'text-rose-400' },
   430→  ];
   431→
   432→  return (
   433→    <div className="flex flex-col gap-5">
   434→      {localAttendanceData.map((item) => (
   435→        <div key={item.label} className="flex flex-col gap-2">
   436→          <div className="flex items-center justify-between">
   437→            <span className="text-sm font-medium text-slate-300">{item.label}</span>
   438→            <span className={`text-sm font-semibold tabular-nums ${item.textColor}`}>
   439→              {item.percentage}%
   440→            </span>
   441→          </div>
   442→          {/* Bar track */}
   443→          <div className="h-3 w-full rounded-full bg-white/[0.06] overflow-hidden">
   444→            <div
   445→              className={`h-full rounded-full ${item.color} transition-all duration-700 ease-out`}
   446→              style={{ width: `${item.percentage}%` }}
   447→            />
   448→          </div>
   449→        </div>
   450→      ))}
   451→
   452→      {/* Summary */}
   453→      <div
   454→        className="mt-1 rounded-lg p-3"
   455→        style={{
   456→          backgroundColor: 'rgba(255,255,255,0.02)',
   457→          border: '1px solid rgba(255,255,255,0.06)',
   458→        }}
   459→      >
   460→        <div className="flex items-center justify-between">
   461→          <span className="text-xsB0B0C0">إجمالي الطلاب المسجلين</span>
   462→          <span className="text-sm font-bold text-white tabular-nums">{total} طالب</span>
   463→        </div>
   464→      </div>
   465→    </div>
   466→  );
   467→}
