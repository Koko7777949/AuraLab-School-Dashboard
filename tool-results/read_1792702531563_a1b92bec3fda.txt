   255→function getStatusColor(status: SessionStatus): string {
   256→  switch (status) {
   257→    case 'upcoming': return '#0ea5e9';
   258→    case 'in-progress': return '#D4A574';
   259→    case 'completed': return '#64748b';

   263→function getEngagementColor(score: number): string {
   264→  if (score >= 80) return '#10b981';
   265→  if (score >= 60) return '#f59e0b';
   266→  return '#ef4444';
   267→}

   269→function formatDateArabic(dateStr: string): string {
   270→  const date = new Date(dateStr);
   271→  return date.toLocaleDateString('ar-SA', {
   272→    weekday: 'long',
   273→    year: 'numeric',
   274→    month: 'long',
   275→    day: 'numeric',
   276→  });
   277→}
   278→
   279→function formatTime(time: string | null): string {
   280→  if (!time) return '—';
   281→  return time;
   282→}
   283→
   284→function calcDuration(start: string | null, end: string | null): string {
   285→  if (!start || !end) return '—';
   286→  const [sh, sm] = start.split(':').map(Number);
   287→  const [eh, em] = end.split(':').map(Number);
   288→  const diff = (eh * 60 + em) - (sh * 60 + sm);
   289→  if (diff <= 0) return '—';
   290→  const h = Math.floor(diff / 60);
   291→  const m = diff % 60;
   292→  if (h === 0) return `${m} دقيقة`;
   293→  if (m === 0) return `${h} ساعة`;
   294→  return `${h} ساعة و ${m} دقيقة`;
   295→}
   296→
   297→function getTodayLabel(dateStr: string): string {
   298→  const today = new Date().toISOString().split('T')[0];
   299→  const tomorrow = new Date(Date.now() + 86400000).toISOString().split('T')[0];
   300→  const yesterday = new Date(Date.now() - 86400000).toISOString().split('T')[0];
   301→  if (dateStr === today) return 'اليوم';
   302→  if (dateStr === tomorrow) return 'غداً';
   303→  if (dateStr === yesterday) return 'أمس';
   304→  return formatDateArabic(dateStr);
   305→}

   311→/** Circular engagement indicator using SVG */
   312→function EngagementRing({
   313→  score,
   314→  size = 52,
   315→  strokeWidth = 4,
   316→}: {
   317→  score: number;
   318→  size?: number;
   319→  strokeWidth?: number;
   320→}) {
   321→  const radius = (size - strokeWidth) / 2;
   322→  const circumference = radius * 2 * Math.PI;
   323→  const color = score > 0 ? getEngagementColor(score) : 'rgba(255,255,255,0.1)';
   324→
   325→  return (
   326→    <div className="relative inline-flex items-center justify-center" style={{ width: size, height: size }}>
   327→      <svg width={size} height={size} className="-rotate-90">
   328→        <circle
   329→          cx={size / 2}
   330→          cy={size / 2}
   331→          r={radius}
   332→          fill="none"
   333→          stroke="rgba(255,255,255,0.06)"
   334→          strokeWidth={strokeWidth}
   335→        />
   336→        {score > 0 && (
   337→          <circle
   338→            cx={size / 2}
   339→            cy={size / 2}
   340→            r={radius}
   341→            fill="none"
   342→            stroke={color}
   343→            strokeWidth={strokeWidth}
   344→            strokeLinecap="round"
   345→            strokeDasharray={`${circumference} ${circumference}`}
   346→            style={{
   347→              strokeDashoffset: circumference - (score / 100) * circumference,
   348→              transition: 'stroke-dashoffset 1s ease-out',
   349→            }}
   350→          />
   351→        )}
   352→      </svg>
   353→      <span className="absolute text-xs font-bold text-white tabular-nums select-none">
   354→        {score > 0 ? score : '—'}
   355→      </span>
   356→    </div>
   357→  );
   358→}
