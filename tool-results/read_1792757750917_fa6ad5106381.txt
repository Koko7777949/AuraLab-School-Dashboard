   169→  const statuses: AttendanceStatus[] = ['present', 'present', 'present', 'present', 'present', 'absent', 'late', 'excused', 'present', 'present'];

   171→    present: ['', '', ''],
   172→    absent: ['لم يحضر بدون عذر', 'غياب مرضي'],

   175→  };
   176→
   177→  const records: AttendanceRecord[] = [];
   178→  const today = new Date();

   180→  for (let dayOffset = 0; dayOffset < 15; dayOffset++) {
   181→    const date = new Date(today);
   182→    date.setDate(date.getDate() - dayOffset);
   183→
   184→    // Skip weekends
   185→    const dayOfWeek = date.getDay();

   189→    const shuffled = [...students].sort(() => Math.random() - 0.5);
   190→    const dailyStudents = shuffled.slice(0, 6 + Math.floor(Math.random() * 3));

   192→    for (const student of dailyStudents) {

   217→function getClassName(classId: string | null): string {
   218→  if (!classId) return '—';

   223→function formatDate(dateStr: string): string {
   224→  const date = new Date(dateStr);
   225→  return date.toLocaleDateString('ar-SA', {
   226→    year: 'numeric',
   227→    month: 'short',
   228→    day: 'numeric',
   229→  });
   230→}

   450→function EditFormContent({
   451→  record,
   452→  onClose,
   453→}: {
   454→  record: AttendanceRecord | null;
   455→  onClose: () => void;
   456→}) {
   457→  const { updateAttendanceRecord } = useAttendanceStore();
   458→  const [status, setStatus] = useState<AttendanceStatus>(record?.status ?? 'present');
   459→  const [notes, setNotes] = useState(record?.notes ?? '');
   460→
   461→  const handleSubmit = (e: React.FormEvent) => {
   462→    e.preventDefault();
   463→    if (!record) return;
   464→
   465→    updateAttendanceRecord(record.id, { status, notes });
   466→    toast.success('تم تحديث سجل الحضور');
   467→    onClose();
   468→  };
   469→
   470→  return (
   471→    <form onSubmit={handleSubmit} className="flex flex-col gap-5 mt-2">
   472→      {/* Status radio buttons */}
   473→      <div className="flex flex-col gap-2">
   474→        <Label className="text-slate-300 text-sm">حالة الحضور</Label>
   475→        <div className="grid grid-cols-2 gap-2">
   476→          {STATUS_OPTIONS.map((opt) => {
   477→            const config = STATUS_CONFIG[opt.value];
   478→            const Icon = config.icon;
   479→            const isSelected = status === opt.value;
   480→            return (
   481→              <button
   482→                key={opt.value}
   483→                type="button"
   484→                onClick={() => setStatus(opt.value)}
   485→                className={cn(
   486→                  'flex items-center justify-center gap-2 rounded-xl py-3 px-4 border transition-all duration-200',
   487→                  isSelected
   488→                    ? config.bgClass + ' ' + config.color + ' font-semibold'
   489→                    : 'border-white/[0.06] text-slate-400 hover:bg-white/[0.04] hover:text-slate-300'
   490→                )}
   491→              >
   492→                <Icon className="h-4 w-4" />
   493→                <span className="text-sm">{opt.label}</span>
   494→              </button>
   495→            );
   496→          })}
   497→        </div>
   498→      </div>
   499→
   500→      {/* Notes */}
   501→      <div className="flex flex-col gap-2">
   502→        <Label className="text-slate-300 text-sm">ملاحظات</Label>
   503→        <Textarea
   504→          value={notes}
   505→          onChange={(e) => setNotes(e.target.value)}
   506→          placeholder="أضف ملاحظات حول الحضور..."
   507→          className="bg-white/[0.03] border-white/[0.08] text-slate-200 placeholder:text-muted-foreground/50 focus:border-[#D4A574]/50 focus:ring-[#D4A574]/20 min-h-[80px] resize-none"
   508→        />
   509→      </div>
   510→
   511→      <DialogFooter className="flex-row gap-3 sm:justify-start pt-2">
   512→        <Button
   513→          type="submit"
   514→          className="bg-[#D4A574] hover:bg-[#B8895A] text-slate-900 font-semibold rounded-lg px-6 gap-2"
   515→        >
   516→          <Save className="h-4 w-4" />
   517→          حفظ التغييرات
   518→        </Button>
   519→        <Button
   520→          type="button"
   521→          variant="outline"
   522→          onClick={onClose}
   523→          className="border-white/[0.08] text-slate-300 hover:bg-white/[0.05] hover:text-white rounded-lg px-6"
   524→        >
   525→          إلغاء
   526→        </Button>
   527→      </DialogFooter>
   528→    </form>
   529→  );
   530→}

   810→function AttendanceHeatmap({
   811→  records,
   812→  currentMonth,
   813→  currentYear,
   814→  onMonthChange,
   815→  onYearChange,
   816→}: {
   817→  records: AttendanceRecord[];
   818→  currentMonth: number;
   819→  currentYear: number;
   820→  onMonthChange: (month: number) => void;
   821→  onYearChange: (year: number) => void;
   822→}) {
   823→  const daysInMonth = new Date(currentYear, currentMonth + 1, 0).getDate();
   824→  const firstDayOfMonth = new Date(currentYear, currentMonth, 1).getDay();
   825→
   826→  // Group records by date for this month
   827→  const dateMap = new Map<string, AttendanceRecord[]>();
   828→  for (const rec of records) {
   829→    const d = new Date(rec.date);
   830→    if (d.getMonth() === currentMonth && d.getFullYear() === currentYear) {
   831→      const key = d.getDate().toString();
   832→      if (!dateMap.has(key)) dateMap.set(key, []);
   833→      dateMap.get(key)!.push(rec);
   834→    }
   835→  }
   836→
   837→  const getCellColor = (day: number) => {
   838→    const dayRecords = dateMap.get(day.toString());
   839→    if (!dayRecords || dayRecords.length === 0) return 'bg-white/[0.02]';
   840→    const presentCount = dayRecords.filter((r) => r.status === 'present').length;
   841→    const ratio = presentCount / dayRecords.length;
   842→    if (ratio >= 0.9) return 'bg-emerald-500/40';
   843→    if (ratio >= 0.7) return 'bg-emerald-500/25';
   844→    if (ratio >= 0.5) return 'bg-amber-500/25';
   845→    if (ratio > 0) return 'bg-rose-500/25';
   846→    return 'bg-rose-500/40';
   847→  };
   848→
   849→  const getDaySummary = (day: number) => {
   850→    const dayRecords = dateMap.get(day.toString());
   851→    if (!dayRecords || dayRecords.length === 0) return null;
   852→    const present = dayRecords.filter((r) => r.status === 'present').length;
   853→    const absent = dayRecords.filter((r) => r.status === 'absent').length;
   854→    const late = dayRecords.filter((r) => r.status === 'late').length;
   855→    const excused = dayRecords.filter((r) => r.status === 'excused').length;
   856→    return { total: dayRecords.length, present, absent, late, excused };
   857→  };
   858→
   859→  const handlePrevMonth = () => {
   860→    if (currentMonth === 0) {
   861→      onMonthChange(11);
   862→      onYearChange(currentYear - 1);
   863→    } else {
   864→      onMonthChange(currentMonth - 1);
   865→    }
   866→  };
   867→
   868→  const handleNextMonth = () => {
   869→    if (currentMonth === 11) {
   870→      onMonthChange(0);
   871→      onYearChange(currentYear + 1);
   872→    } else {
   873→      onMonthChange(currentMonth + 1);
   874→    }
   875→  };
   876→
   877→  // Build calendar grid (6 rows × 7 columns)
   878→  const cells: (number | null)[] = [];
   879→  // Add empty cells for offset (Saturday-first week: shift by (firstDay + 1) % 7)
   880→  const offset = (firstDayOfMonth + 1) % 7;
   881→  for (let i = 0; i < offset; i++) cells.push(null);
   882→  for (let d = 1; d <= daysInMonth; d++) cells.push(d);
   883→  while (cells.length % 7 !== 0) cells.push(null);
   884→
   885→  return (
   886→    <div
   887→      className="rounded-xl p-4"
   888→      style={{
   889→        backgroundColor: 'rgba(255,255,255,0.02)',
   890→        border: '1px solid rgba(255,255,255,0.06)',
   891→      }}
   892→    >
   893→      {/* Month navigation */}
   894→      <div className="flex items-center justify-between mb-4">
   895→        <h3 className="text-sm font-semibold text-slate-300 flex items-center gap-2">
   896→          <BarChart3 className="h-4 w-4 text-[#D4A574]" />
   897→          خريطة الحضور الشهرية
   898→        </h3>
   899→        <div className="flex items-center gap-2">
   900→          <Button
   901→            variant="ghost"
   902→            size="icon"
   903→            onClick={handlePrevMonth}
   904→            className="h-7 w-7 text-slate-400 hover:text-white hover:bg-white/[0.05] rounded-lg"
   905→          >
   906→            <ChevronRight className="h-4 w-4" />
   907→          </Button>
   908→          <span className="text-sm font-medium text-slate-200 min-w-[120px] text-center">
   909→            {ARABIC_MONTHS[currentMonth]} {currentYear}
   910→          </span>
   911→          <Button
   912→            variant="ghost"
   913→            size="icon"
   914→            onClick={handleNextMonth}
   915→            className="h-7 w-7 text-slate-400 hover:text-white hover:bg-white/[0.05] rounded-lg"
   916→          >
   917→            <ChevronLeft className="h-4 w-4" />
   918→          </Button>
   919→        </div>
   920→      </div>
   921→
   922→      {/* Day headers */}
   923→      <div className="grid grid-cols-7 gap-1 mb-1">
   924→        {ARABIC_DAYS.map((day) => (
   925→          <div key={day} className="text-center text-[10px] font-semibold text-muted-foreground/60 py-1">
   926→            {day}
   927→          </div>
   928→        ))}
   929→      </div>
   930→
   931→      {/* Calendar cells */}
   932→      <div className="grid grid-cols-7 gap-1">
   933→        {cells.map((day, i) => {
   934→          if (day === null) {
   935→            return <div key={`empty-${i}`} className="aspect-square" />;
   936→          }
   937→
   938→          const summary = getDaySummary(day);
   939→          const isToday =
   940→            day === new Date().getDate() &&
   941→            currentMonth === new Date().getMonth() &&
   942→            currentYear === new Date().getFullYear();
   943→
   944→          return (
   945→            <div
   946→              key={day}
   947→              className={cn(
   948→                'aspect-square rounded-lg flex flex-col items-center justify-center text-xs relative transition-all duration-200 cursor-default',
   949→                getCellColor(day),
   950→                isToday && 'ring-1 ring-[#D4A574]/60 ring-offset-1 ring-offset-[#121218]'
   951→              )}
   952→              title={summary ? `حاضر: ${summary.present}, غائب: ${summary.absent}, متأخر: ${summary.late}, مستأذن: ${summary.excused}` : 'لا توجد بيانات'}
   953→            >
   954→              <span className={cn(
   955→                'font-medium tabular-nums',
   956→                summary ? 'text-slate-200' : 'text-muted-foreground/40'
   957→              )}>
   958→                {day}
   959→              </span>
   960→              {summary && summary.total > 0 && (
   961→                <span className="text-[9px] text-muted-foreground/50 mt-0.5">
   962→                  {summary.total}
   963→                </span>
   964→              )}
   965→            </div>
   966→          );
   967→        })}
   968→      </div>
   969→
   970→      {/* Legend */}
   971→      <div className="flex items-center justify-center gap-4 mt-3 pt-3 border-t border-white/[0.04]">
   972→        <div className="flex items-center gap-1.5">
   973→          <div className="h-3 w-3 rounded bg-emerald-500/40" />
   974→          <span className="text-[10px] text-muted-foreground">ممتاز</span>
   975→        </div>
   976→        <div className="flex items-center gap-1.5">
   977→          <div className="h-3 w-3 rounded bg-emerald-500/25" />
   978→          <span className="text-[10px] text-muted-foreground">جيد</span>
   979→        </div>
   980→        <div className="flex items-center gap-1.5">
   981→          <div className="h-3 w-3 rounded bg-amber-500/25" />
   982→          <span className="text-[10px] text-muted-foreground">متوسط</span>
   983→        </div>
   984→        <div className="flex items-center gap-1.5">
   985→          <div className="h-3 w-3 rounded bg-rose-500/25" />
