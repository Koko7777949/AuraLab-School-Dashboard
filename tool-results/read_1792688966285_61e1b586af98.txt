   129→function getScoreColor(score: number | null, maxScore: number): { bar: string; text: string; bg: string } {
   130→  if (score === null) return { bar: 'bg-slate-500', text: 'text-slate-400', bg: '' };
   131→  const pct = (score / maxScore) * 100;
   132→  if (pct >= 70) return { bar: 'bg-emerald-500', text: 'text-emerald-400', bg: 'bg-emerald-500/10' };
   133→  if (pct >= 50) return { bar: 'bg-amber-500', text: 'text-amber-400', bg: 'bg-amber-500/10' };
   134→  return { bar: 'bg-rose-500', text: 'text-rose-400', bg: 'bg-rose-500/10' };
   135→}
   136→
   137→function getScoreLabel(score: number | null, maxScore: number): string {
   138→  if (score === null) return 'لم يُحدد';
   139→  const pct = (score / maxScore) * 100;
   140→  if (pct >= 90) return 'ممتاز';
   141→  if (pct >= 70) return 'جيد جداً';
   142→  if (pct >= 50) return 'مقبول';
   143→  return 'يحتاج تحسين';
   144→}

   150→interface SubjectAvg {
   151→  subject: string;
   152→  avg: number;
   153→  count: number;

   156→
   157→interface AnalysisStats {
   158→  highest: { pct: number; subject: string };
   159→  lowest: { pct: number; subject: string };
   160→  overallAvg: number;
   161→  passRate: number;
   162→}
   163→
   164→function getBarColorHex(avg: number): string {
   165→  if (avg >= 85) return '#10b981';
   166→  if (avg >= 70) return '#f59e0b';
   167→  if (avg >= 50) return '#0ea5e9';
   168→  return '#f43f5e';
   169→}
   170→
   171→function computeSubjectAverages(exams: Exam[]): SubjectAvg[] {
   172→  const groups: Record<string, { total: number; count: number }> = {};
   173→
   174→  for (const exam of exams) {
   175→    if (exam.score === null) continue;
   176→    const pct = (exam.score / Math.max(0.001, exam.maxScore)) * 100;
   177→    if (!groups[exam.subject]) {
   178→      groups[exam.subject] = { total: 0, count: 0 };
   179→    }
   180→    groups[exam.subject] = {
   181→      total: groups[exam.subject].total + pct,
   182→      count: groups[exam.subject].count + 1,
   183→    };
   184→  }
   185→
   186→  return Object.entries(groups)
   187→    .map(([subject, data]) => ({
   188→      subject,
   189→      avg: Math.round((data.total / Math.max(0.001, data.count)) * 10) / 10,
   190→      count: data.count,
   191→      color: getBarColorHex(data.total / Math.max(0.001, data.count)),
   192→    }))
   193→    .sort((a, b) => b.avg - a.avg);
   194→}
   195→
   196→function computeAnalysisStats(exams: Exam[]): AnalysisStats {
   197→  const scoredExams = exams.filter((e) => e.score !== null);
   198→
   199→  if (scoredExams.length === 0) {
   200→    return {
   201→      highest: { pct: 0, subject: '—' },
   202→      lowest: { pct: 0, subject: '—' },
   203→      overallAvg: 0,
   204→      passRate: 0,
   205→    };
   206→  }
   207→
   208→  const result = scoredExams.reduce(
   209→    (acc, exam) => {
   210→      const pct = (exam.score! / Math.max(0.001, exam.maxScore)) * 100;
   211→      return {
   212→        totalPct: acc.totalPct + pct,
   213→        passCount: acc.passCount + (pct >= 50 ? 1 : 0),
   214→        highestPct: Math.max(acc.highestPct, pct),
   215→        highestSubject: pct > acc.highestPct ? exam.subject : acc.highestSubject,
   216→        lowestPct: Math.min(acc.lowestPct, pct),
   217→        lowestSubject: pct < acc.lowestPct ? exam.subject : acc.lowestSubject,
   218→      };
   219→    },
   220→    { totalPct: 0, passCount: 0, highestPct: -1, highestSubject: '', lowestPct: 101, lowestSubject: '' },
   221→  );
   222→
   223→  const count = Math.max(0.001, scoredExams.length);
   224→  return {
   225→    highest: { pct: Math.round(result.highestPct * 10) / 10, subject: result.highestSubject },
   226→    lowest: { pct: Math.round(result.lowestPct * 10) / 10, subject: result.lowestSubject },
   227→    overallAvg: Math.round((result.totalPct / count) * 10) / 10,
   228→    passRate: Math.round(((result.passCount / count) * 100) * 10) / 10,
   229→  };
   230→}
